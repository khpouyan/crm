function normalizeIranPhoneLocal(raw?: string | null): string {
  const digits = normalizePersianDigits(String(raw || "")).replace(/\D/g, "");
  if (!digits) return "";
  if (digits.startsWith("0098") && digits.length >= 14) {
    return "0" + digits.slice(4, 14);
  }
  if (digits.startsWith("98") && digits.length >= 12) {
    return "0" + digits.slice(2, 12);
  }
  if (digits.startsWith("9") && digits.length === 10) {
    return "0" + digits;
  }
  return digits;
}

/**
 * ثبت سریع پرونده مراجع جدید در کلینیک از داخل ویزارد نوبت‌دهی
 * بررسی تکراری بودن صرفاً در مدل Client همین کلینیک انجام می‌شود.
 */
export async function quickCreateClientAction(
  input: QuickCreateClientInput
): Promise<{
  success: boolean;
  error?: string;
  client?: CreatedClientSummary;
}> {
  try {
    const ctx = await getCurrentClinicContext();
    if (!ctx) {
      return { success: false, error: "نشست کاربری معتبر نیست." };
    }

    if (
      !ctx.clinicUser.isOwner &&
      !hasPermission(ctx.clinicUser, "appointments.create") &&
      !hasPermission(ctx.clinicUser, "appointments.edit")
    ) {
      return {
        success: false,
        error: "شما مجوز ثبت پرونده مراجع جدید را ندارید.",
      };
    }

    const fullName = String(input.fullName || "").trim();
    if (!fullName) {
      return {
        success: false,
        error: "علت عدم ثبت: وارد کردن نام و نام خانوادگی مراجع الزامی است.",
      };
    }

    const cleanPhone = normalizeIranPhoneLocal(input.phone);
    const cleanNationalId =
      normalizePersianDigits(String(input.nationalId || "")).replace(
        /\D/g,
        ""
      ) || null;
    const cleanNickname = String(input.nickname || "").trim() || null;

    const existingClinicClients = await prisma.client.findMany({
      where: { clinicId: ctx.clinicId },
      include: {
        user: {
          select: {
            id: true,
            fullName: true,
            phone: true,
            nationalId: true,
          },
        },
      },
    });

    if (cleanNationalId) {
      const dupNat = existingClinicClients.find(
        (c) => c.user.nationalId && c.user.nationalId === cleanNationalId
      );
      if (dupNat) {
        return {
          success: false,
          error: `علت عدم ثبت: مراجعی با کد ملی ${normalizePersianDigits(
            cleanNationalId
          )} پیش‌تر با نام «${
            dupNat.user.fullName || "بدون نام"
          }» در این کلینیک پرونده دارد.`,
        };
      }
    }

    if (cleanPhone) {
      const dupPhone = existingClinicClients.find(
        (c) =>
          !c.user.phone.startsWith("NO_PHONE_") && c.user.phone === cleanPhone
      );
      if (dupPhone) {
        return {
          success: false,
          error: `علت عدم ثبت: مراجعی با شماره تماس ${normalizePersianDigits(
            cleanPhone
          )} پیش‌تر با نام «${
            dupPhone.user.fullName || "بدون نام"
          }» در این کلینیک پرونده دارد.`,
        };
      }
    }

    const maxFileRow = await prisma.client.findFirst({
      where: { clinicId: ctx.clinicId },
      orderBy: { fileNumber: "desc" },
      select: { fileNumber: true },
    });
    const nextFileNumber = (maxFileRow?.fileNumber ?? 0) + 1;

    const syntheticPhone =
      cleanPhone ||
      `NO_PHONE_${Date.now()}_${Math.random().toString(36).slice(2, 7)}`;

    const createdClient = await prisma.$transaction(async (tx) => {
      let targetUser = cleanPhone
        ? await tx.user.findUnique({ where: { phone: cleanPhone } })
        : null;

      if (!targetUser) {
        targetUser = await tx.user.create({
          data: {
            phone: syntheticPhone,
            fullName,
            nationalId: cleanNationalId,
          },
        });
      } else {
        targetUser = await tx.user.update({
          where: { id: targetUser.id },
          data: {
            fullName: targetUser.fullName || fullName,
            nationalId: targetUser.nationalId || cleanNationalId,
          },
        });
      }

      return tx.client.create({
        data: {
          clinicId: ctx.clinicId,
          userId: targetUser.id,
          fileNumber: nextFileNumber,
          nickname: cleanNickname,
        },
        include: {
          user: {
            select: {
              fullName: true,
              phone: true,
              nationalId: true,
            },
          },
        },
      });
    });

    revalidatePath(CALENDAR_PATH);
    revalidatePath("/clinic-dashboard/clients");

    return {
      success: true,
      client: {
        id: createdClient.id,
        fullName: createdClient.user.fullName || fullName,
        phone: createdClient.user.phone.startsWith("NO_PHONE_")
          ? ""
          : createdClient.user.phone,
        nationalId: createdClient.user.nationalId,
        nickname: createdClient.nickname,
        fileNumber: createdClient.fileNumber,
      },
    };
  } catch (err: unknown) {
    return {
      success: false,
      error:
        err instanceof Error
          ? `علت عدم ثبت پرونده: ${err.message}`
          : "علت عدم ثبت پرونده: خطای غیرمنتظره هنگام ایجاد پرونده مراجع رخ داد.",
    };
  }
}

export async function upsertAppointmentAction(
  input: UpsertAppointmentInput
): Promise<UpsertAppointmentResult> {
  try {
    const ctx = await getCurrentClinicContext();
    if (!ctx) {
      return { success: false, error: "نشست کاربری معتبر نیست." };
    }

    const isEdit = Boolean(input.id);
    const requiredPerm = isEdit ? "appointments.edit" : "appointments.create";

    if (!hasPermission(ctx.clinicUser, requiredPerm)) {
      return {
        success: false,
        error: isEdit
          ? "شما مجوز ویرایش نوبت را ندارید."
          : "شما مجوز ثبت نوبت جدید را ندارید.",
      };
    }

    if (!input.clientId || !input.serviceId) {
      return {
        success: false,
        error: "انتخاب مراجع و سرویس برای ثبت نوبت الزامی است.",
      };
    }

    if (
      !canAccessServiceCategory(ctx.clinicUser, requiredPerm, input.serviceId)
    ) {
      return {
        success: false,
        error: "شما دسترسی نوبت‌دهی برای این دسته‌بندی سرویس را ندارید.",
      };
    }

    const bookingSource =
      input.bookingSource === AppointmentBookingSource.AUTO
        ? AppointmentBookingSource.AUTO
        : AppointmentBookingSource.MANUAL;

    const [client, service, subService] = await Promise.all([
      prisma.client.findFirst({
        where: { id: input.clientId, clinicId: ctx.clinicId },
        include: { user: { select: { fullName: true, phone: true } } },
      }),
      prisma.service.findFirst({
        where: { id: input.serviceId, clinicId: ctx.clinicId },
        select: { id: true, title: true },
      }),
      input.subServiceId && input.subServiceId !== "NONE"
        ? prisma.subService.findFirst({
            where: {
              id: input.subServiceId,
              serviceId: input.serviceId,
            },
            select: { id: true, title: true },
          })
        : Promise.resolve(null),
    ]);

    if (!client) {
      return {
        success: false,
        error: "مراجع انتخاب‌شده در این کلینیک یافت نشد.",
      };
    }

    if (!service) {
      return {
        success: false,
        error: "سرویس انتخاب‌شده در این کلینیک یافت نشد.",
      };
    }

    const resolvedDate = resolveCalendarDate(input.dateInput);
    if (!resolvedDate) {
      return {
        success: false,
        error: "تاریخ واردشده برای نوبت معتبر نیست.",
      };
    }

    if (isEdit && input.id) {
      const existingAppt = await prisma.appointment.findFirst({
        where: { id: input.id, clinicId: ctx.clinicId },
        select: { id: true },
      });
      if (!existingAppt) {
        return {
          success: false,
          error: "نوبت مورد نظر برای ویرایش یافت نشد.",
        };
      }
    }

    const evaluation = await evaluateAppointmentSlot({
      clinicId: ctx.clinicId,
      clientId: client.id,
      serviceId: service.id,
      subServiceId: subService?.id ?? null,
      therapistId:
        input.therapistId && input.therapistId !== "NONE"
          ? input.therapistId
          : null,
      spaceId: input.spaceId && input.spaceId !== "NONE" ? input.spaceId : null,
      dateInput: resolvedDate.utcMidnight,
      startTime: input.startTime,
      bookingSource,
      allowManualOverride:
        bookingSource === AppointmentBookingSource.MANUAL &&
        Boolean(input.allowManualOverride),
      excludeAppointmentId: input.id,
    });

    if (!evaluation.valid || !evaluation.computed || !evaluation.timing) {
      const detailedError =
        evaluation.reasons.length > 0
          ? evaluation.reasons.join(" | ")
          : "امکان ثبت نوبت در این زمان وجود ندارد.";

      return {
        success: false,
        error: detailedError,
        reasons: evaluation.reasons,
        issues: evaluation.issues,
        canOverrideManually: evaluation.canOverrideManually,
      };
    }

    const hasBlockingIssues = evaluation.issues.some((i) => i.blocking);
    const isManualOverride =
      bookingSource === AppointmentBookingSource.MANUAL &&
      Boolean(input.allowManualOverride) &&
      hasBlockingIssues;

    const nextStatus = input.status ?? AppointmentStatus.CONFIRMED;
    const therapistId =
      input.therapistId && input.therapistId !== "NONE"
        ? input.therapistId
        : null;

    const savedId = await prisma.$transaction(async (tx) => {
      const payload = {
        clinicId: ctx.clinicId,
        clientId: client.id,
        serviceId: service.id,
        subServiceId: subService?.id ?? null,
        therapistId,
        spaceId: evaluation.resolvedSpaceId,
        bookingSource,
        status: nextStatus,
        date: resolvedDate.utcMidnight,
        arrivalTime: evaluation.computed!.arrivalTime,
        startTime: evaluation.computed!.startTime,
        endTime: evaluation.computed!.endTime,
        blockEndTime: evaluation.computed!.blockEndTime,
        timingMode: evaluation.timing!.timingMode,
        prepBeforeMinutes: evaluation.timing!.prepBeforeMinutes,
        durationMinutes: evaluation.timing!.durationMinutes,
        stage1Minutes: evaluation.timing!.stage1Minutes,
        freeGapMinutes: evaluation.timing!.freeGapMinutes,
        stage2Minutes: evaluation.timing!.stage2Minutes,
        restAfterMinutes: evaluation.timing!.restAfterMinutes,
        requiresPrivacy: evaluation.timing!.requiresPrivacy,
        isManualOverride,
        overrideReason: isManualOverride
          ? input.overrideReason?.trim() || "ثبت دستی با هماهنگی ویژه"
          : null,
        notes: input.notes?.trim() || null,
      };

      if (isEdit && input.id) {
        const updated = await tx.appointment.update({
          where: { id: input.id },
          data: payload,
          select: { id: true },
        });
        return updated.id;
      }

      const created = await tx.appointment.create({
        data: {
          ...payload,
          createdById: ctx.clinicUserId,
        },
        select: { id: true },
      });
      return created.id;
    });

    revalidatePath(CALENDAR_PATH);
    revalidatePath("/clinic-dashboard");

    return {
      success: true,
      appointmentId: savedId,
      reasons: evaluation.reasons,
      issues: evaluation.issues,
    };
  } catch (err: unknown) {
    console.error("upsertAppointmentAction error:", err);
    return {
      success: false,
      error:
        err instanceof Error
          ? err.message
          : "خطای غیرمنتظره هنگام ثبت نوبت رخ داد.",
    };
  }
}

  stepMinutes?: number;

      !hasPermission(ctx.clinicUser, "appointments.view") &&
      !hasPermission(ctx.clinicUser, "appointments.create")
    ) {
      return {
        success: false,
        slots: [],
        summaryReason: null,
        error: "شما مجوز مشاهده زمان‌های نوبت‌دهی را ندارید.",
      };
    }

    const res = await findAvailableSlotsForDate({
      clinicId: ctx.clinicId,
      serviceId: params.serviceId,
      subServiceId:
        params.subServiceId && params.subServiceId !== "NONE"
          ? params.subServiceId
          : null,
      therapistId:
        params.therapistId && params.therapistId !== "NONE"
          ? params.therapistId
          : null,
      spaceId:
        params.spaceId && params.spaceId !== "NONE" ? params.spaceId : null,
      dateInput: normalizePersianDigits(params.dateInput),
      stepMinutes: params.stepMinutes ?? 15,
      includeConditionalOnCall: params.includeConditionalOnCall ?? true,
    });

    const summaryReason =
      res.slots.length === 0
        ? res.blockingReasons[0] ||
          "در تاریخ انتخاب‌شده، زمان آزاد و معتبری برای این سرویس و شرایط یافت نشد."
        : null;

    return {
      success: true,
      slots: res.slots,
      summaryReason,
    };
  } catch (err: unknown) {
    return {
      success: false,
      slots: [],
      summaryReason: null,
      error:
        err instanceof Error
          ? err.message
          : "خطا در محاسبه بازه‌های خالی نوبت‌دهی.",
    };
  }
}

export async function updateAppointmentStatusAction(params: {

  status: AppointmentStatus;
  cancellationReason?: string | null;
}): Promise<{ success: boolean; error?: string }> {

function parseJalaliDateParam(raw?: string): {
  jy: number;
  jm: number;
  jd: number;
  utcMidnight: Date;
  jalaliStr: string;
} {
  const nowJ = toJalali(new Date());
  const norm = normalizePersianDigits(String(raw || "")).trim();
  const match = /^(\d{4})[/-](\d{1,2})[/-](\d{1,2})$/.exec(norm);

  if (match) {
    const jy = Number(match[1]);
    const jm = Number(match[2]);
    const jd = Number(match[3]);
    if (jy >= 1300 && jy <= 1500 && jm >= 1 && jm <= 12 && jd >= 1 && jd <= 31) {
      const g = toGregorian(jy, jm, jd);
      const check = toJalali(g);
      if (check.jy === jy && check.jm === jm && check.jd === jd) {
        return {
          jy,
          jm,
          jd,
          utcMidnight: new Date(
            Date.UTC(g.getFullYear(), g.getMonth(), g.getDate(), 0, 0, 0, 0)
          ),
          jalaliStr: `${jy}/${String(jm).padStart(2, "0")}/${String(
            jd
          ).padStart(2, "0")}`,
        };
      }
    }
  }

  const gNow = toGregorian(nowJ.jy, nowJ.jm, nowJ.jd);
  return {
    jy: nowJ.jy,
    jm: nowJ.jm,
    jd: nowJ.jd,
    utcMidnight: new Date(
      Date.UTC(gNow.getFullYear(), gNow.getMonth(), gNow.getDate(), 0, 0, 0, 0)
    ),
    jalaliStr: `${nowJ.jy}/${String(nowJ.jm).padStart(2, "0")}/${String(
      nowJ.jd
    ).padStart(2, "0")}`,
  };
}

function getJalaliMonthDaysCount(jy: number, jm: number): number {
  if (jm <= 6) return 31;
  if (jm <= 11) return 30;
  const g30 = toGregorian(jy, 12, 30);
  const j30 = toJalali(g30);
  return j30.jy === jy && j30.jm === 12 && j30.jd === 30 ? 30 : 29;
}

function utcDateToJalaliStr(dt: Date): string {
  const localRef = new Date(
    dt.getUTCFullYear(),
    dt.getUTCMonth(),
    dt.getUTCDate()
  );
  const j = toJalali(localRef);
  return `${j.jy}/${String(j.jm).padStart(2, "0")}/${String(j.jd).padStart(
    2,
    "0"
  )}`;
}

export default async function CalendarPage({
  searchParams,
}: {
  searchParams: Promise<{ date?: string; view?: string }>;
}) {
  const ctx = await getCurrentClinicContext();
  if (!ctx) {
    redirect("/clinic-dashboard/login");
  }

  const canView =
    hasPermission(ctx.clinicUser, "appointments.view") ||
    hasPermission(ctx.clinicUser, "appointments.create") ||
    hasPermission(ctx.clinicUser, "appointments.edit");

  if (!canView) {
    redirect("/clinic-dashboard");
  }

  const params = await searchParams;
  const selected = parseJalaliDateParam(params.date);

  const monthDays = getJalaliMonthDaysCount(selected.jy, selected.jm);
  const firstG = toGregorian(selected.jy, selected.jm, 1);
  const lastG = toGregorian(selected.jy, selected.jm, monthDays);

  const rangeStart = new Date(
    Date.UTC(firstG.getFullYear(), firstG.getMonth(), firstG.getDate(), 0, 0, 0, 0)
  );
  const rangeEnd = new Date(
    Date.UTC(
      lastG.getFullYear(),
      lastG.getMonth(),
      lastG.getDate(),
      23,
      59,
      59,
      999
    )
  );

  const [appointmentsRows, servicesRows, staffRows, spacesRows, clientsRows] =
    await Promise.all([
      prisma.appointment.findMany({
        where: {
          clinicId: ctx.clinicId,
          date: {
            gte: rangeStart,
            lte: rangeEnd,
          },
        },
        orderBy: [{ date: "asc" }, { startTime: "asc" }],
        include: {
          client: {
            include: {
              user: {
                select: {
                  fullName: true,
                  phone: true,
                },
              },
            },
          },
          service: {
            select: {
              id: true,
              title: true,
            },
          },
          subService: {
            select: {
              id: true,
              title: true,
            },
          },
          therapist: {
            include: {
              user: {
                select: {
                  fullName: true,
                  phone: true,
                },
              },
            },
          },
          space: {
            select: {
              id: true,
              name: true,
              spaceType: true,
            },
          },
          createdBy: {
            include: {
              user: {
                select: {
                  fullName: true,
                  phone: true,
                },
              },
            },
          },
        },
      }),
      prisma.service.findMany({
        where: { clinicId: ctx.clinicId },
        orderBy: { title: "asc" },
        select: {
          id: true,
          title: true,
          subServices: {
            orderBy: { title: "asc" },
            select: {
              id: true,
              title: true,
              price: true,
            },
          },
        },
      }),
      prisma.clinicUser.findMany({
        where: {
          clinicId: ctx.clinicId,
          deletedAt: null,
          isActive: true,
        },
        orderBy: [{ isOwner: "desc" }, { createdAt: "asc" }],
        include: {
          user: {
            select: {
              fullName: true,
              phone: true,
            },
          },
          assignedServices: {
            select: { id: true },
          },
        },
      }),
      prisma.clinicSpace.findMany({
        where: {
          clinicId: ctx.clinicId,
          isActive: true,
        },
        orderBy: [{ sortOrder: "asc" }, { name: "asc" }],
        select: {
          id: true,
          name: true,
          spaceType: true,
          allowedServices: {
            select: { id: true },
          },
        },
      }),
      prisma.client.findMany({
        where: { clinicId: ctx.clinicId },
        orderBy: [{ fileNumber: { sort: "desc", nulls: "last" } }],
        take: 1000,
        include: {
          user: {
            select: {
              fullName: true,
              phone: true,
              nationalId: true,
            },
          },
        },
      }),
    ]);

  const appointments: AppointmentRecord[] = appointmentsRows.map((a) => ({
    id: a.id,
    clientId: a.clientId,
    clientName: a.client.user.fullName || "مراجع بدون نام",
    clientPhone: a.client.user.phone.startsWith("NO_PHONE_")
      ? "بدون شماره"
      : a.client.user.phone,
    fileNumber: a.client.fileNumber,
    serviceId: a.serviceId,
    serviceTitle: a.service.title,
    subServiceId: a.subServiceId,
    subServiceTitle: a.subService?.title ?? null,
    therapistId: a.therapistId,
    therapistName:
      a.therapist?.user.fullName ||
      a.therapist?.user.phone ||
      a.therapist?.jobTitle ||
      "تعیین‌نشده",
    spaceId: a.spaceId,
    spaceName: a.space?.name ?? null,
    createdByName:
      a.createdBy?.user.fullName || a.createdBy?.user.phone || "سیستم",
    bookingSource: a.bookingSource,
    status: a.status,
    jalaliDate: utcDateToJalaliStr(a.date),
    arrivalTime: a.arrivalTime,
    startTime: a.startTime,
    endTime: a.endTime,
    blockEndTime: a.blockEndTime,
    timingMode: a.timingMode,
    freeGapMinutes: a.freeGapMinutes,
    isManualOverride: a.isManualOverride,
    overrideReason: a.overrideReason,
    notes: a.notes,
    cancellationReason: a.cancellationReason,
    createdAtIso: a.createdAt.toISOString(),
  }));

  const services: CalendarServiceOption[] = servicesRows.map((s) => ({
    id: s.id,
    title: s.title,
    subServices: s.subServices,
  }));

  const staff: CalendarStaffOption[] = staffRows.map((st) => ({
    id: st.id,
    fullName: st.user.fullName || st.user.phone,
    jobTitle: st.jobTitle,
    hasAllServicesAccess: st.isOwner || st.hasAllServicesAccess,
    assignedServiceIds: st.assignedServices.map((as) => as.id),
  }));

  const spaces: CalendarSpaceOption[] = spacesRows.map((sp) => ({
    id: sp.id,
    name: sp.name,
    spaceType: sp.spaceType,
    allowedServiceIds: sp.allowedServices.map((s) => s.id),
  }));

  const clients: CalendarClientOption[] = clientsRows.map((c) => ({
    id: c.id,
    fullName: c.user.fullName || "مراجع بدون نام",
    phone: c.user.phone.startsWith("NO_PHONE_") ? "" : c.user.phone,
    nationalId: c.user.nationalId,
    nickname: c.nickname,
    fileNumber: c.fileNumber,
  }));

  return (
    <CalendarClient
      clinicName={ctx.clinic.name}
      initialJalaliDate={selected.jalaliStr}
      initialView={
        params.view === "monthly" ||
        params.view === "table" ||
        params.view === "logs"
          ? params.view
          : "daily"
      }
      appointments={appointments}
      services={services}
      staff={staff}
      spaces={spaces}
      clients={clients}
      permissions={{
        canCreate: hasPermission(ctx.clinicUser, "appointments.create"),
        canEdit: hasPermission(ctx.clinicUser, "appointments.edit"),
        canDelete: hasPermission(ctx.clinicUser, "appointments.delete"),
      }}
    />
  );
}
