interface SubServiceFormDialogProps {
  open: boolean
  onOpenChange: (open: boolean) => void
  serviceId: string
  serviceTitle: string
  editingSubService?: {
    id: string
    title: string
    price: number
    attributes: unknown
    defaultProducts: {
      clinicProduct: { id: string }
      quantity: number
    }[]
  } | null
  clinicProducts: { id: string; name: string; unit: string }[]
}

export function SubServiceFormDialog({
  open,
  onOpenChange,
  serviceId,
  serviceTitle,
  editingSubService,
  clinicProducts,
}: SubServiceFormDialogProps) {
  const [title, setTitle] = useState("")
  const [price, setPrice] = useState<number>(0)
  const [attributes, setAttributes] = useState<ServiceAttributeItem[]>([])
  const [defaultProducts, setDefaultProducts] = useState<DefaultProductInput[]>(
    []
  )
  const [isPending, setIsPending] = useState(false)

  // نگاشت کالاها برای Base UI Select جهت نمایش نام فارسی به جای UUID
  const productSelectItems = useMemo(
    () =>
      clinicProducts.map((prod) => ({
        value: prod.id,
        label: `${prod.name} (${prod.unit})`,
      })),
    [clinicProducts]
  )

  useEffect(() => {
    if (open) {
      setTitle(editingSubService?.title ?? "")
      setPrice(editingSubService?.price ?? 0)

      const parsedAttrs = Array.isArray(editingSubService?.attributes)
        ? (editingSubService?.attributes as ServiceAttributeItem[])
        : []
      setAttributes(parsedAttrs)

      const mappedProducts =
        editingSubService?.defaultProducts.map((dp) => ({
          clinicProductId: dp.clinicProduct.id,
          quantity: dp.quantity,
        })) ?? []
      setDefaultProducts(mappedProducts)
    }
  }, [open, editingSubService])

  const addAttribute = () => {
    setAttributes((prev) => [...prev, { key: "", value: "" }])
  }

  const updateAttribute = (
    index: number,
    field: "key" | "value",
    val: string
  ) => {
    setAttributes((prev) =>
      prev.map((item, idx) =>
        idx === index ? { ...item, [field]: val } : item
      )
    )
  }

  const removeAttribute = (index: number) => {
    setAttributes((prev) => prev.filter((_, idx) => idx !== index))
  }

  const addDefaultProduct = () => {
    if (clinicProducts.length === 0) {
      toast.error("ابتدا در بخش انبار کالا، اقلام مصرفی را تعریف کنید.")
      return
    }
    setDefaultProducts((prev) => [
      ...prev,
      { clinicProductId: clinicProducts[0].id, quantity: 1 },
    ])
  }

  const updateDefaultProduct = (
    index: number,
    field: keyof DefaultProductInput,
    val: string | number
  ) => {
    setDefaultProducts((prev) =>
      prev.map((item, idx) =>
        idx === index ? { ...item, [field]: val } : item
      )
    )
  }

  const removeDefaultProduct = (index: number) => {
    setDefaultProducts((prev) => prev.filter((_, idx) => idx !== index))
  }

  const handleSubmit = async (e: React.FormEvent) => {

import { prisma } from "@/lib/prisma";
import { getCurrentClinicContext } from "@/lib/auth/session";
import { hasPermission } from "@/lib/permissions";
import { revalidatePath } from "next/cache";

export type ServiceAttributeItem = {

export type DefaultProductInput = {

/**
 * ایجاد یا ویرایش خدمت اصلی (همراه با انتساب پرسنل و کالاهای مصرفی پیش‌فرض کل خدمت)
 */
export async function upsertServiceAction(data: {
  id?: string;
  title: string;
  assignedStaffIds: string[];
  defaultProducts?: DefaultProductInput[];
}) {
  const ctx = await getCurrentClinicContext();
  if (!ctx) return { success: false, error: "دسترسی غیرمجاز" };

  const requiredPerm = data.id ? "services.edit" : "services.create";
  if (!hasPermission(ctx.clinicUser, requiredPerm)) {
    return {
      success: false,
      error: data.id
        ? "شما مجوز ویرایش سرویس را ندارید."
        : "شما مجوز ثبت سرویس جدید را ندارید.",
    };
  }

  if (!data.title.trim()) {
    return { success: false, error: "عنوان خدمت الزامی است" };
  }

  const productMap = new Map<string, number>();
  for (const p of data.defaultProducts ?? []) {
    if (p.clinicProductId && p.quantity > 0) {
      productMap.set(
        p.clinicProductId,
        (productMap.get(p.clinicProductId) ?? 0) + Number(p.quantity)
      );
    }
  }
  const cleanProducts = Array.from(productMap.entries()).map(
    ([clinicProductId, quantity]) => ({
      clinicProductId,
      quantity,
    })
  );

  try {
    const validStaff = await prisma.clinicUser.findMany({
      where: {
        id: { in: data.assignedStaffIds },
        clinicId: ctx.clinicId,
        isActive: true,
        deletedAt: null,
      },
      select: { id: true },
    });

    const staffIdSet = new Set(validStaff.map((s) => s.id));

    if (!ctx.clinicUser.isOwner && !ctx.clinicUser.hasAllServicesAccess) {
      staffIdSet.add(ctx.clinicUserId);
    }

    const finalStaffIds = Array.from(staffIdSet);

    await prisma.$transaction(async (tx) => {
      if (data.id) {
        const existing = await tx.service.findFirst({
          where: {
            id: data.id,
            clinicId: ctx.clinicId,
            ...(!ctx.clinicUser.isOwner && !ctx.clinicUser.hasAllServicesAccess
              ? { assignedStaff: { some: { id: ctx.clinicUserId } } }
              : {}),
          },
        });

        if (!existing) {
          throw new Error("خدمت مورد نظر یافت نشد یا دسترسی ندارید");
        }

        await tx.service.update({
          where: { id: data.id },
          data: {
            title: data.title.trim(),
            assignedStaff: {
              set: finalStaffIds.map((id) => ({ id })),
            },
          },
        });

        await tx.serviceCategoryDefaultProduct.deleteMany({
          where: { serviceId: data.id },
        });

        if (cleanProducts.length > 0) {
          await tx.serviceCategoryDefaultProduct.createMany({
            data: cleanProducts.map((p) => ({
              serviceId: data.id!,
              clinicProductId: p.clinicProductId,
              quantity: p.quantity,
            })),
          });
        }
      } else {
        await tx.service.create({
          data: {
            clinicId: ctx.clinicId,
            title: data.title.trim(),
            assignedStaff: {
              connect: finalStaffIds.map((id) => ({ id })),
            },
            defaultServiceProducts:
              cleanProducts.length > 0
                ? {
                    create: cleanProducts.map((p) => ({
                      clinicProductId: p.clinicProductId,
                      quantity: p.quantity,
                    })),
                  }
                : undefined,
          },
        });
      }
    });

    revalidatePath("/clinic-dashboard/services");
    return { success: true };
  } catch (error: unknown) {
    console.error("upsertServiceAction error:", error);
    return {
      success: false,
      error: error instanceof Error ? error.message : "خطا در ذخیره خدمت",
    };
  }
}

export default async function ServicesPage() {

  if (!ctx) {
    redirect("/clinic-dashboard/login");
  }

  const { clinicUser } = ctx;

  const canView =
    hasPermission(clinicUser, "services.view") ||
    hasPermission(clinicUser, "services.create") ||
    hasPermission(clinicUser, "services.edit");

  if (!canView) {
    redirect("/clinic-dashboard");
  }

  const canCreate = hasPermission(clinicUser, "services.create");
  const canEdit = hasPermission(clinicUser, "services.edit");
  const canDelete = hasPermission(clinicUser, "services.delete");

  const restrictToAssigned =
    !clinicUser.isOwner && !clinicUser.hasAllServicesAccess;

  const [services, activeStaff, clinicProducts] = await Promise.all([
    prisma.service.findMany({
      where: {
        clinicId: ctx.clinicId,
        ...(restrictToAssigned
          ? {
              assignedStaff: {
                some: { id: ctx.clinicUserId },
              },
            }
          : {}),
      },
      include: {
        assignedStaff: {
          where: { isActive: true, deletedAt: null },
          include: {
            user: { select: { id: true, fullName: true, phone: true } },
            role: { select: { name: true } },
          },
        },
        defaultServiceProducts: {
          include: {
            clinicProduct: { select: { id: true, name: true, unit: true } },
          },
        },
        subServices: {
          include: {
            defaultProducts: {
              include: {
                clinicProduct: { select: { id: true, name: true, unit: true } },
              },
            },
          },
          orderBy: { title: "asc" },
        },
      },
      orderBy: { title: "asc" },
    }),
    prisma.clinicUser.findMany({
      where: {
        clinicId: ctx.clinicId,
        isActive: true,
        deletedAt: null,
      },
      include: {
        user: { select: { id: true, fullName: true, phone: true } },
        role: { select: { name: true } },
      },
      orderBy: { createdAt: "asc" },
    }),
    prisma.clinicProduct.findMany({
      where: { clinicId: ctx.clinicId },
      select: { id: true, name: true, unit: true },
      orderBy: { name: "asc" },
    }),
  ]);

    hasAllServicesAccess: staff.isOwner || staff.hasAllServicesAccess,

  editingService,
  activeStaff,
  clinicProducts,
}: ServiceFormDialogProps) {
  const [title, setTitle] = useState("")
  const [selectedStaffIds, setSelectedStaffIds] = useState<string[]>([])
  const [defaultProducts, setDefaultProducts] = useState<DraftServiceProduct[]>(
    []
  )
  const [isPending, setIsPending] = useState(false)

  useEffect(() => {
    if (open) {
      setTitle(editingService?.title ?? "")
      setSelectedStaffIds(editingService?.assignedStaff.map((s) => s.id) ?? [])
      setDefaultProducts(
        editingService?.defaultServiceProducts?.map((dp) => ({
          uid: dp.id,
          clinicProductId: dp.clinicProduct.id,
          quantity: String(dp.quantity),
        })) ?? []
      )
    }
  }, [open, editingService])

  const productSelectItems = useMemo(
    () => [
      { value: "NONE", label: "انتخاب کالا از انبار..." },
      ...clinicProducts.map((p) => ({
        value: p.id,
        label: `${p.name} (${p.unit})`,
      })),
    ],
    [clinicProducts]
  )

  const toggleStaff = (staffId: string) => {
    setSelectedStaffIds((prev) =>
      prev.includes(staffId)
        ? prev.filter((id) => id !== staffId)
        : [...prev, staffId]
    )
  }

  const handleAddProductRow = () => {
    setDefaultProducts((prev) => [
      ...prev,
      {
        uid: crypto.randomUUID(),
        clinicProductId: "NONE",
        quantity: "1",
      },
    ])
  }

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault()
    if (!title.trim()) {
      toast.error("لطفاً عنوان خدمت را وارد کنید")
      return
    }

    const mappedProducts = defaultProducts
      .filter((p) => p.clinicProductId !== "NONE" && parseFloat(p.quantity) > 0)
      .map((p) => ({
        clinicProductId: p.clinicProductId,
        quantity: parseFloat(p.quantity),
      }))

    setIsPending(true)
    const res = await upsertServiceAction({
      id: editingService?.id,
      title,
      assignedStaffIds: selectedStaffIds,
      defaultProducts: mappedProducts,
    })
    setIsPending(false)

    if (res.success) {
      toast.success(
        editingService ? "خدمت با موفقیت ویرایش شد" : "خدمت جدید ایجاد شد"
      )
      onOpenChange(false)
    } else {
      toast.error(res.error || "خطایی رخ داد")
    }
  }

  return (
    <ResponsiveDialog
      open={open}
      onOpenChange={onOpenChange}
      drawerPosition="bottom"
    >
      <ResponsiveDialogTrigger
        className="sr-only"
        tabIndex={-1}
        aria-hidden="true"
      />
      <ResponsiveDialogPopup className="sm:max-w-lg">
        <ResponsiveDialogHeader>
          <ResponsiveDialogTitle>
            {editingService ? "ویرایش خدمت" : "افزودن خدمت جدید"}
          </ResponsiveDialogTitle>
          <ResponsiveDialogDescription>
            خدمت اصلی دسته‌بندی کلان است؛ مصرفی‌های مشترک این خدمت فقط یک‌بار به
            ازای کل خدمت در فاکتور لحاظ می‌شوند.
          </ResponsiveDialogDescription>
        </ResponsiveDialogHeader>

        <form onSubmit={handleSubmit}>
          <ResponsiveDialogPanel className="no-scrollbar max-h-[70vh] overflow-y-auto">
            <div className="px-5">
              <FieldGroup>
                <Field>
                  <FieldLabel htmlFor="service-title">عنوان خدمت *</FieldLabel>
                  <Input
                    id="service-title"
                    placeholder="مثلاً: لیزر موهای زائد، تزریق بوتاکس..."
                    value={title}
                    onChange={(e) => setTitle(e.target.value)}
                    autoFocus
                  />
                </Field>

                {/* بخش کالاهای مصرفی پیش‌فرض در سطح خدمت اصلی */}
                <Field>
                  <div className="flex items-center justify-between">
                    <FieldLabel className="flex items-center gap-1.5">
                      <Package className="size-4 text-primary" />
                      مصرفی دیفالت کل خدمت (اختیاری)
                    </FieldLabel>
                    <Button
                      type="button"
                      variant="outline"
                      size="sm"
                      className="h-7 gap-1 text-xs"
                      onClick={handleAddProductRow}
                    >
                      <Plus className="size-3.5" />
                      افزودن کالا
                    </Button>
                  </div>
                  <FieldDescription className="text-[11px]">
                    این اقلام هنگام ثبت فاکتور فقط یک‌بار برای کل خدمت (مثلاً
                    روتختی یکبارمصرف در لیزر) اضافه می‌شوند و با انتخاب چند
                    زیرسرویس تکرار نمی‌شوند.
                  </FieldDescription>

                  {defaultProducts.length > 0 && (
                    <div className="mt-2 space-y-2">
                      {defaultProducts.map((row) => {
                        const selectedProd = clinicProducts.find(
                          (cp) => cp.id === row.clinicProductId
                        )
                        return (
                          <div
                            key={row.uid}
                            className="flex items-center gap-2 rounded-lg border bg-muted/25 p-2"
                          >
                            <div className="flex-1">
                              <SearchableSelect
                                options={productSelectItems}
                                value={row.clinicProductId}
                                placeholder="جستجو و انتخاب کالا از انبار..."
                                emptyText="کالایی با این نام یافت نشد."
                                className="h-8 bg-background text-xs"
                                onValueChange={(val) =>
                                  setDefaultProducts((prev) =>
                                    prev.map((item) =>
                                      item.uid === row.uid
                                        ? {
                                            ...item,
                                            clinicProductId: val || "NONE",
                                          }
                                        : item
                                    )
                                  )
                                }
                              />
                            </div>

                            <Input
                              type="number"
                              step="any"
                              min="0.1"
                              className="h-8 w-20 bg-background text-center text-xs"
                              value={row.quantity}
                              onChange={(e) =>
                                setDefaultProducts((prev) =>
                                  prev.map((item) =>
                                    item.uid === row.uid
                                      ? { ...item, quantity: e.target.value }
                                      : item
                                  )
                                )
                              }
                            />

                            {selectedProd && (
                              <span className="shrink-0 text-xs text-muted-foreground">
                                {selectedProd.unit}
                              </span>
                            )}

                            <Button
                              type="button"
                              variant="ghost"
                              size="icon"
                              className="size-7 shrink-0 text-destructive"
                              onClick={() =>
                                setDefaultProducts((prev) =>
                                  prev.filter((item) => item.uid !== row.uid)
                                )
                              }
                            >
                              <Trash2 className="size-3.5" />
                            </Button>
                          </div>
                        )
                      })}
                    </div>
                  )}
                </Field>

                {/* بخش پرسنل مجاز */}
                <Field>
                  <FieldLabel className="flex items-center gap-1.5">
                    <Users className="size-4 text-muted-foreground" />
                    پرسنل دارای دسترسی اختصاصی به این خدمت
                  </FieldLabel>
                  <FieldDescription className="text-[11px]">
                    افرادی که در مدیریت پرسنل روی «همه خدمات» تنظیم شده‌اند، به
                    صورت خودکار به این خدمت دسترسی دارند.
                  </FieldDescription>

                  <div className="mt-1 max-h-48 divide-y overflow-y-auto rounded-lg border">
                    {activeStaff.length === 0 ? (
                      <p className="p-3 text-center text-xs text-muted-foreground">
                        پرسنل فعالی ثبت نشده است.
                      </p>
                    ) : (
                      activeStaff.map((staff) => {
                        const checked = selectedStaffIds.includes(staff.id)
                        return (
                          <label
                            key={staff.id}
                            className="flex cursor-pointer items-center justify-between gap-2 p-2.5 transition-colors hover:bg-muted/50"
                          >
                            <div className="flex min-w-0 items-center gap-2.5">
                              <Checkbox
                                checked={checked}
                                onCheckedChange={() => toggleStaff(staff.id)}
                              />
                              <div className="min-w-0 text-sm">
                                <div className="flex flex-wrap items-center gap-1.5">
                                  <span className="truncate font-medium">
                                    {staff.user.fullName || staff.user.phone}
                                  </span>
                                  {staff.hasAllServicesAccess && (
                                    <Badge
                                      variant="secondary"
                                      className="h-4 px-1.5 text-[10px] font-normal"
                                    >
                                      دسترسی همه خدمات
                                    </Badge>
                                  )}
                                </div>
                                <div className="text-xs text-muted-foreground">
                                  {staff.role.name}
                                </div>
                              </div>
                            </div>
                            {checked && (
                              <Check className="size-4 shrink-0 text-primary" />
                            )}
                          </label>
                        )
                      })
                    )}
                  </div>
                </Field>
              </FieldGroup>
            </div>
          </ResponsiveDialogPanel>

          <ResponsiveDialogFooter>
            <Button type="submit" disabled={isPending}>
              {isPending && <Loader2 className="ml-2 size-4 animate-spin" />}
              {editingService ? "ذخیره تغییرات" : "ایجاد خدمت"}
            </Button>
            <ResponsiveDialogDesktopOnly>
              <ResponsiveDialogClose
                render={
                  <Button type="button" variant="outline" disabled={isPending}>
                    انصراف
                  </Button>
                }
              />
            </ResponsiveDialogDesktopOnly>
          </ResponsiveDialogFooter>
        </form>
      </ResponsiveDialogPopup>
    </ResponsiveDialog>
  )
}
</file>

</files>
