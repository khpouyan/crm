// app/clinic-dashboard/(dashboard)/inventory/low-stock/page.tsx
import { redirect } from "next/navigation";
import { prisma } from "@/lib/prisma";
import { getCurrentClinicContext } from "@/lib/auth/session";
import { hasPermission } from "@/lib/permissions";
import { parsePaginationParams } from "@/lib/inventory/pagination-helper";
import { InventoryLowStockClient } from "@/components/clinic-dashboard/inventory-low-stock-client";

export const metadata = {
  title: "کالاهای تمام‌شده و نزدیک به اتمام | انبار",
};

export default async function InventoryLowStockPage({
  searchParams,
}: {
  searchParams: Promise<{ page?: string; pageSize?: string }>;
}) {
  const ctx = await getCurrentClinicContext();
  if (!ctx) redirect("/clinic-dashboard/login");

  const canManage = hasPermission(ctx.clinicUser, "finance.inventory");
  const canView =
    canManage || hasPermission(ctx.clinicUser, "tools.inventory_view");
  if (!canView) redirect("/clinic-dashboard");

  const params = await searchParams;
  const { page, pageSize, skip, take } = parsePaginationParams(params);

  // محاسبه موجودی کل هر کالا و فیلتر موارد کمتر یا مساوی alertThreshold
  const allProducts = await prisma.clinicProduct.findMany({
    where: { clinicId: ctx.clinicId },
    include: {
      batches: { where: { quantity: { gt: 0 } }, select: { quantity: true } },
    },
    orderBy: { name: "asc" },
  });

  const lowStockItems = allProducts
    .map((p) => ({
      id: p.id,
      name: p.name,
      code: p.code,
      unit: p.unit,
      alertThreshold: p.alertThreshold,
      totalStock: p.batches.reduce((s, b) => s + b.quantity, 0),
    }))
    .filter((p) => p.totalStock <= p.alertThreshold)
    .sort((a, b) => a.totalStock - b.totalStock);

  const totalCount = lowStockItems.length;
  const paginatedItems = lowStockItems.slice(skip, skip + take);

  return (
    <InventoryLowStockClient
      items={paginatedItems}
      totalCount={totalCount}
      page={page}
      pageSize={pageSize}
      canManage={canManage}
    />
  );
}

import { redirect } from "next/navigation";
import { prisma } from "@/lib/prisma";
import { getCurrentClinicContext } from "@/lib/auth/session";
import { hasPermission } from "@/lib/permissions";
import { parsePaginationParams } from "@/lib/inventory/pagination-helper";
import { checkClinicExportCapability } from "@/lib/export/data-export";
import { InventoryProductsClient } from "@/components/clinic-dashboard/inventory-products-client";

export const metadata = {
  title: "تعریف کالا و مصرفی | انبار",
};

export default async function InventoryProductsPage({
  searchParams,
}: {
  searchParams: Promise<{ page?: string; pageSize?: string; q?: string }>;
}) {
  const ctx = await getCurrentClinicContext();
  if (!ctx) redirect("/clinic-dashboard/login");

  const canManage = hasPermission(ctx.clinicUser, "finance.inventory");
  const canView =
    canManage || hasPermission(ctx.clinicUser, "tools.inventory_view");
  if (!canView) redirect("/clinic-dashboard");

  const params = await searchParams;
  const { page, pageSize, skip, take } = parsePaginationParams(params);
  const query = params.q?.trim() || "";

  const whereClause = {
    clinicId: ctx.clinicId,
    ...(query
      ? {
          OR: [
            { name: { contains: query, mode: "insensitive" as const } },
            { code: { contains: query, mode: "insensitive" as const } },
          ],
        }
      : {}),
  };

  const [totalCount, products, canExport] = await Promise.all([
    prisma.clinicProduct.count({ where: whereClause }),
    prisma.clinicProduct.findMany({
      where: whereClause,
      include: {
        batches: {
          where: { quantity: { gt: 0 } },
          select: { quantity: true },
        },
      },
      orderBy: { name: "asc" },
      skip,
      take,
    }),
    checkClinicExportCapability(ctx.clinicId),
  ]);

  const serializedProducts = products.map((p) => ({
    id: p.id,
    name: p.name,
    code: p.code,
    unit: p.unit,
    packageUnit: p.packageUnit,
    unitsPerPackage: p.unitsPerPackage,
    alertThreshold: p.alertThreshold,
    totalStock: p.batches.reduce((sum, b) => sum + b.quantity, 0),
  }));

  return (
    <InventoryProductsClient
      products={serializedProducts}
      totalCount={totalCount}
      page={page}
      pageSize={pageSize}
      initialQuery={query}
      canManage={canManage}
      canExport={canExport}
    />
  );
}

// app/clinic-dashboard/(dashboard)/inventory/sales-report/page.tsx
import { redirect } from "next/navigation";
import { prisma } from "@/lib/prisma";
import { getCurrentClinicContext } from "@/lib/auth/session";
import { hasPermission } from "@/lib/permissions";
import { parsePaginationParams } from "@/lib/inventory/pagination-helper";
import { checkClinicExportCapability } from "@/lib/export/data-export";
import { InventorySalesReportClient } from "@/components/clinic-dashboard/inventory-sales-report-client";

export const metadata = {
  title: "گزارش فروش و هزینه کالاها | انبار",
};

export default async function InventorySalesReportPage({
  searchParams,
}: {
  searchParams: Promise<{
    page?: string;
    pageSize?: string;
    from?: string;
    to?: string;
    serviceId?: string;
    staffId?: string;
  }>;
}) {
  const ctx = await getCurrentClinicContext();
  if (!ctx) redirect("/clinic-dashboard/login");

  const canView =
    hasPermission(ctx.clinicUser, "finance.inventory") ||
    hasPermission(ctx.clinicUser, "finance.accounting");
  if (!canView) redirect("/clinic-dashboard");

  const params = await searchParams;
  const { page, pageSize, skip, take } = parsePaginationParams(params);

  const dateFilter: { gte?: Date; lte?: Date } = {};
  if (params.from) {
    const d = new Date(params.from);
    d.setHours(0, 0, 0, 0);
    if (!isNaN(d.getTime())) dateFilter.gte = d;
  }
  if (params.to) {
    const d = new Date(params.to);
    d.setHours(23, 59, 59, 999);
    if (!isNaN(d.getTime())) dateFilter.lte = d;
  }
  const hasDate = Boolean(dateFilter.gte || dateFilter.lte);

  const [usages, stockOuts, services, staffs, canExport] = await Promise.all([
    prisma.usedConsumable.findMany({
      where: {
        clinicProduct: { clinicId: ctx.clinicId },
        providedService: {
          ...(hasDate ? { createdAt: dateFilter } : {}),
          ...(params.serviceId
            ? { subService: { serviceId: params.serviceId } }
            : {}),
          ...(params.staffId ? { therapistId: params.staffId } : {}),
        },
      },
      include: {
        clinicProduct: { select: { name: true, unit: true } },
        providedService: {
          select: {
            createdAt: true,
            therapistNameSnapshot: true,
            subService: {
              select: { title: true, service: { select: { title: true } } },
            },
          },
        },
      },
      orderBy: { providedService: { createdAt: "desc" } },
    }),
    // خروج دستی زمانی که فیلتر خدمت خاص انتخاب نشده باشد
    params.serviceId
      ? Promise.resolve([])
      : prisma.inventoryStockOut.findMany({
          where: {
            clinicId: ctx.clinicId,
            ...(hasDate ? { createdAt: dateFilter } : {}),
            ...(params.staffId ? { createdById: params.staffId } : {}),
          },
          include: {
            clinicProduct: { select: { name: true, unit: true } },
            createdBy: { select: { user: { select: { fullName: true } } } },
          },
          orderBy: { createdAt: "desc" },
        }),
    prisma.service.findMany({
      where: { clinicId: ctx.clinicId },
      select: { id: true, title: true },
      orderBy: { title: "asc" },
    }),
    prisma.clinicUser.findMany({
      where: { clinicId: ctx.clinicId, isActive: true, deletedAt: null },
      select: { id: true, user: { select: { fullName: true } } },
    }),
    checkClinicExportCapability(ctx.clinicId),
  ]);

  // ترکیب اقلام کسرشده با خدمات و اقلام خروجی دستی
  const combinedItems = [
    ...usages.map((u) => {
      const isSale = Boolean(u.sellingPrice && u.sellingPrice > 0);
      const totalCost = u.unitCost * u.quantity;
      const totalRevenue = isSale ? (u.sellingPrice || 0) * u.quantity : 0;
      return {
        id: `usage-${u.id}`,
        source: "SERVICE" as const,
        sourceLabel: `${u.providedService.subService.service.title} - ${u.providedService.subService.title}`,
        productName: u.clinicProduct.name,
        unit: u.clinicProduct.unit,
        quantity: u.quantity,
        isSale,
        totalCost,
        totalRevenue,
        netProfit: isSale ? totalRevenue - totalCost : 0,
        staffName: u.providedService.therapistNameSnapshot || "—",
        createdAt: u.providedService.createdAt.toISOString(),
      };
    }),
    ...stockOuts.map((o) => {
      const isSale = Boolean(o.sellingPrice && o.sellingPrice > 0);
      const totalCost = o.unitCost * o.quantity;
      const totalRevenue = isSale ? (o.sellingPrice || 0) * o.quantity : 0;
      return {
        id: `out-${o.id}`,
        source: "MANUAL" as const,
        sourceLabel: o.reason || "خروج دستی",
        productName: o.clinicProduct.name,
        unit: o.clinicProduct.unit,
        quantity: o.quantity,
        isSale,
        totalCost,
        totalRevenue,
        netProfit: isSale ? totalRevenue - totalCost : 0,
        staffName: o.createdBy?.user.fullName || "—",
        createdAt: o.createdAt.toISOString(),
      };
    }),
  ].sort(
    (a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime()
  );

  // محاسبه آمار کل برای کارت‌های بالای صفحه:
  // ۱. کالاهایی که قیمت فروش نداشتند -> هزینه کالاهای مصرفی
  // ۲. کالاهایی که قیمت فروش داشتند -> درآمد و سود فروش کالا
  const totalConsumableExpense = combinedItems
    .filter((i) => !i.isSale)
    .reduce((sum, i) => sum + i.totalCost, 0);

  const totalSalesProfit = combinedItems
    .filter((i) => i.isSale)
    .reduce((sum, i) => sum + i.netProfit, 0);

  const totalSalesRevenue = combinedItems
    .filter((i) => i.isSale)
    .reduce((sum, i) => sum + i.totalRevenue, 0);

  const totalCount = combinedItems.length;
  const paginatedRows = combinedItems.slice(skip, skip + take);

  return (
    <InventorySalesReportClient
      rows={paginatedRows}
      totalCount={totalCount}
      page={page}
      pageSize={pageSize}
      summary={{
        totalConsumableExpense,
        totalSalesProfit,
        totalSalesRevenue,
      }}
      services={services}
      staffs={staffs.map((s) => ({
        id: s.id,
        fullName: s.user.fullName || "بدون نام",
      }))}
      filters={{
        from: params.from,
        to: params.to,
        serviceId: params.serviceId,
        staffId: params.staffId,
      }}
      canExport={canExport}
    />
  );
}

  searchParams,

  searchParams: Promise<{ page?: string; pageSize?: string }>;
}) {
  const ctx = await getCurrentClinicContext();
  if (!ctx) redirect("/clinic-dashboard/login");

  if (!hasPermission(ctx.clinicUser, "finance.inventory")) {
    redirect("/clinic-dashboard");
  }

  const params = await searchParams;
  const { skip, take } = parsePaginationParams(params);

  const orders = await prisma.b2BOrder.findMany({

    include: {

/**
 * ۴. خروج دستی کالا از انبار (ثبت در InventoryStockOut + کسر FIFO از بچ‌ها)
 */
export async function stockOutProductAction(input: {
  clinicProductId: string;
  quantity: number;
  sellingPrice?: number | null;
  reason?: string;
}) {
  try {
    const ctx = await requireInventoryWriteContext();

    const deductQty = Number(input.quantity);
    if (!input.clinicProductId || deductQty <= 0) {
      return { success: false, error: "مقدار خروج باید بیشتر از صفر باشد." };
    }

    await prisma.$transaction(async (tx) => {
      const batches = await tx.productBatch.findMany({
        where: {
          clinicProductId: input.clinicProductId,
          clinicProduct: { clinicId: ctx.clinicId },
          quantity: { gt: 0 },
        },
        orderBy: [
          { expirationDate: { sort: "asc", nulls: "last" } },
          { createdAt: "asc" },
        ],
      });

      const totalAvailable = batches.reduce((sum, b) => sum + b.quantity, 0);
      if (totalAvailable < deductQty) {
        throw new Error(
          `موجودی قابل برداشت کافی نیست (موجودی فعلی: ${totalAvailable.toLocaleString("fa-IR")}).`
        );
      }

      let remaining = deductQty;
      let totalCostAccumulated = 0;

      for (const batch of batches) {
        if (remaining <= 0) break;
        const take = Math.min(batch.quantity, remaining);
        totalCostAccumulated += take * batch.purchasePrice;

        await tx.productBatch.update({
          where: { id: batch.id },
          data: { quantity: batch.quantity - take },
        });
        remaining -= take;
      }

      const avgUnitCost = Math.round(totalCostAccumulated / deductQty);

      await tx.inventoryStockOut.create({
        data: {
          clinicId: ctx.clinicId,
          clinicProductId: input.clinicProductId,
          createdById: ctx.clinicUserId,
          quantity: deductQty,
          unitCost: avgUnitCost,
          sellingPrice:
            input.sellingPrice && input.sellingPrice > 0
              ? Number(input.sellingPrice)
              : null,
          reason: input.reason?.trim() || "خروج دستی از انبار",
        },
      });
    });

    revalidatePath("/clinic-dashboard/inventory", "layout");
    return { success: true };
  } catch (error: any) {
    return {
      success: false,
      error: error?.message || "خطا در ثبت خروج کالا",
    };
  }
}

/**
 * ۷. ویرایش رکورد خروج دستی کالا از انبار (تنظیم هوشمند موجودی بچ‌ها)
 */
export async function updateStockOutAction(input: {
  stockOutId: string;
  quantity: number;
  sellingPrice?: number | null;
  reason?: string;
}) {
  try {
    const ctx = await requireInventoryWriteContext();

    const nextQty = Number(input.quantity);
    if (!input.stockOutId || nextQty <= 0) {
      return { success: false, error: "مقدار خروجی باید بیشتر از صفر باشد." };
    }

    await prisma.$transaction(async (tx) => {
      const existing = await tx.inventoryStockOut.findFirst({
        where: { id: input.stockOutId, clinicId: ctx.clinicId },
        include: { clinicProduct: true },
      });

      if (!existing) {
        throw new Error("رکورد خروج کالا یافت نشد.");
      }

      const diff = nextQty - existing.quantity;
      let updatedUnitCost = existing.unitCost;

      if (diff > 0) {
        // نیاز به کسر مقدار بیشتر از بچ‌های موجود
        const batches = await tx.productBatch.findMany({
          where: {
            clinicProductId: existing.clinicProductId,
            quantity: { gt: 0 },
          },
          orderBy: [
            { expirationDate: { sort: "asc", nulls: "last" } },
            { createdAt: "asc" },
          ],
        });

        const available = batches.reduce((s, b) => s + b.quantity, 0);
        if (available < diff) {
          throw new Error(
            `موجودی انبار برای افزایش مقدار خروج کافی نیست (موجودی قابل برداشت: ${available.toLocaleString("fa-IR")} ${existing.clinicProduct.unit}).`
          );
        }

        let remaining = diff;
        let extraCost = 0;

        for (const b of batches) {
          if (remaining <= 0) break;
          const take = Math.min(b.quantity, remaining);
          extraCost += take * b.purchasePrice;
          await tx.productBatch.update({
            where: { id: b.id },
            data: { quantity: b.quantity - take },
          });
          remaining -= take;
        }

        const prevTotalCost = existing.quantity * existing.unitCost;
        updatedUnitCost = Math.round((prevTotalCost + extraCost) / nextQty);
      } else if (diff < 0) {
        // بازگرداندن مقدار اضافه‌کسرشده به بچ‌های کالا
        let toRestore = Math.abs(diff);
        const batches = await tx.productBatch.findMany({
          where: { clinicProductId: existing.clinicProductId },
          orderBy: { createdAt: "desc" },
        });

        if (batches.length > 0) {
          for (const b of batches) {
            if (toRestore <= 0) break;
            const maxCap =
              b.initialQuantity > 0 ? b.initialQuantity : b.quantity + toRestore;
            const room = Math.max(0, maxCap - b.quantity);
            if (room > 0) {
              const putBack = Math.min(room, toRestore);
              await tx.productBatch.update({
                where: { id: b.id },
                data: { quantity: b.quantity + putBack },
              });
              toRestore -= putBack;
            }
          }
          // اگر همچنان مقداری باقی ماند، به آخرین بچ اضافه شود
          if (toRestore > 0) {
            await tx.productBatch.update({
              where: { id: batches[0].id },
              data: { quantity: batches[0].quantity + toRestore },
            });
          }
        }
      }

      await tx.inventoryStockOut.update({
        where: { id: existing.id },
        data: {
          quantity: nextQty,
          unitCost: updatedUnitCost,
          sellingPrice:
            input.sellingPrice && input.sellingPrice > 0
              ? Number(input.sellingPrice)
              : null,
          reason: input.reason?.trim() || "خروج دستی از انبار",
        },
      });
    });

    revalidatePath("/clinic-dashboard/inventory", "layout");
    return { success: true };
  } catch (error: any) {
    return {
      success: false,
      error: error?.message || "خطا در ویرایش رکورد خروج کالا",
    };
  }
}

/**
 * ۸. حذف رکورد خروج دستی کالا و بازگرداندن موجودی آن به انبار
 */
export async function deleteStockOutAction(stockOutId: string) {
  try {
    const ctx = await requireInventoryWriteContext();

    await prisma.$transaction(async (tx) => {
      const existing = await tx.inventoryStockOut.findFirst({
        where: { id: stockOutId, clinicId: ctx.clinicId },
      });

      if (!existing) {
        throw new Error("رکورد خروج کالا یافت نشد.");
      }

      let toRestore = existing.quantity;
      const batches = await tx.productBatch.findMany({
        where: { clinicProductId: existing.clinicProductId },
        orderBy: { createdAt: "desc" },
      });

      if (batches.length > 0) {
        for (const b of batches) {
          if (toRestore <= 0) break;
          const maxCap =
            b.initialQuantity > 0 ? b.initialQuantity : b.quantity + toRestore;
          const room = Math.max(0, maxCap - b.quantity);
          if (room > 0) {
            const putBack = Math.min(room, toRestore);
            await tx.productBatch.update({
              where: { id: b.id },
              data: { quantity: b.quantity + putBack },
            });
            toRestore -= putBack;
          }
        }

        if (toRestore > 0) {
          await tx.productBatch.update({
            where: { id: batches[0].id },
            data: { quantity: batches[0].quantity + toRestore },
          });
        }
      }

      await tx.inventoryStockOut.delete({
        where: { id: existing.id },
      });
    });

    revalidatePath("/clinic-dashboard/inventory", "layout");
    return { success: true };
  } catch (error: any) {
    return {
      success: false,
      error: error?.message || "خطا در حذف رکورد خروج کالا",
    };
  }
}
