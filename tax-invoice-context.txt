  currentStep,
  onExit,
  onStepClick,

  return (
    <div className="flex flex-col gap-4 border-b bg-card px-4 py-4 sm:flex-row sm:items-center sm:justify-between sm:px-6">
      <div className="flex items-center justify-between sm:justify-start">

          variant="outline"
          size="sm"
          onClick={onExit}
          className="gap-1.5 rounded-lg"

      </div>

      {/* استپر دایره‌ای مرکزی مطابق طرح */}
      <div className="mx-auto flex w-full max-w-xl items-center justify-between">
        {STEPS.map((step, index) => {
          const isCompleted = currentStep > step.id;
          const isCurrent = currentStep === step.id;
          const canNavigate = onStepClick && step.id < currentStep;

          return (
            <div key={step.id} className="flex flex-1 items-center">
              <button
                type="button"
                disabled={!canNavigate}
                onClick={() => canNavigate && onStepClick(step.id)}
                className="group flex flex-col items-center gap-1.5 text-center disabled:cursor-default"
              >
                <div
                  className={`flex size-10 items-center justify-center rounded-full border-2 text-sm font-bold transition-all ${
                    isCompleted
                      ? "border-primary bg-primary text-primary-foreground"
                      : isCurrent
                      ? "border-primary bg-background text-primary ring-4 ring-primary/15"
                      : "border-muted-foreground/30 bg-muted/40 text-muted-foreground"
                  }`}
                >
                  {isCompleted ? (
                    <Check className="size-5" />
                  ) : (
                    step.id.toLocaleString("fa-IR")
                  )}
                </div>
                <div className="space-y-0.5">
                  <div
                    className={`text-xs font-bold ${
                      isCurrent || isCompleted
                        ? "text-primary"
                        : "text-muted-foreground"
                    }`}
                  >
                    {step.title}
                  </div>
                  <div className="hidden text-[10px] text-muted-foreground sm:block">
                    {step.subtitle}
                  </div>
                </div>
              </button>

              {index < STEPS.length - 1 && (
                <div
                  className={`mx-2 h-0.5 flex-1 rounded-full transition-colors ${

                  }`}
                />

// lib/finance/invoice-calculator.ts

export interface CreditProviderRule {
  id: string;
  title: string;
  commissionPercent: number;
  clientFeePercent: number;
}

export interface CalcServiceLine {
  finalPrice: number;
  consumables: {
    quantity: number;
    sellingPrice: number;
  }[];
}

export interface CalcPaymentLine {
  paymentMethod: "POS" | "CASH" | "CARD_TO_CARD" | "WALLET" | "CREDIT";
  amount: number; // برای CREDIT: اصل مبلغ فاکتور بدون کارمزد
  terminalId?: string | null;
  creditProviderId?: string | null;
}

export interface CalcInstallmentLine {
  amount: number;
}

export interface InvoiceCalculationInput {
  services: CalcServiceLine[];
  discountAmount: number;
  extraChargeAmount: number;
  depositAmount: number;
  depositServiceSelected: boolean;
  payments: CalcPaymentLine[];
  remainingMode: "DEBT" | "INSTALLMENT";
  installments: CalcInstallmentLine[];
  creditProviders: CreditProviderRule[];
  clientWalletBalance: number;
  hasActiveTerminals: boolean;
}

export interface ResolvedCreditLine {
  principal: number;
  clientFee: number;
  totalCharge: number;
  clinicCost: number;
  providerTitle: string;
}

export interface InvoiceCalculationResult {
  servicesSum: number;
  consumablesSellSum: number;
  subTotal: number;
  discountAmount: number;
  extraChargeAmount: number;
  depositAmount: number;
  baseTotalBeforeCreditFee: number;
  creditFeeTotal: number;
  finalTotal: number;
  paidSum: number;
  walletUsedTotal: number;
  remaining: number;
  excessOverpayment: number;
  installmentsSum: number;
  status: "PAID" | "PARTIAL" | "UNPAID" | "INSTALLMENT";
  validationError: string | null;
  creditBreakdown: Record<number, ResolvedCreditLine>;
}

export function calculateInvoiceFinancials(
  input: InvoiceCalculationInput
): InvoiceCalculationResult {
  let servicesSum = 0;
  let consumablesSellSum = 0;

  for (const srv of input.services) {
    servicesSum += Math.max(0, Math.round(Number(srv.finalPrice) || 0));
    for (const cons of srv.consumables) {
      const q = Number(cons.quantity) || 0;
      const sp = Number(cons.sellingPrice) || 0;
      if (q > 0 && sp > 0) {
        consumablesSellSum += Math.round(q * sp);
      }
    }
  }

  const subTotal = servicesSum + consumablesSellSum;
  const discountAmount = Math.max(0, Math.round(Number(input.discountAmount) || 0));
  const extraChargeAmount = Math.max(0, Math.round(Number(input.extraChargeAmount) || 0));
  const depositAmount = Math.max(0, Math.round(Number(input.depositAmount) || 0));

  const baseTotalBeforeCreditFee = Math.max(
    0,
    subTotal - discountAmount + extraChargeAmount + depositAmount
  );

  let creditFeeTotal = 0;
  let paidSum = 0;
  let walletUsedTotal = 0;
  const creditBreakdown: Record<number, ResolvedCreditLine> = {};

  input.payments.forEach((pay, idx) => {
    const rawAmt = Number(pay.amount) || 0;
    if (rawAmt <= 0) return;

    if (pay.paymentMethod === "WALLET") {
      walletUsedTotal += Math.round(rawAmt);
    }

    if (pay.paymentMethod === "CREDIT") {
      const principal = Math.round(rawAmt);
      const cp = input.creditProviders.find((c) => c.id === pay.creditProviderId);
      const clientFeePercent = cp?.clientFeePercent ?? 0;
      const commissionPercent = cp?.commissionPercent ?? 0;

      const clientFee = Math.round((principal * clientFeePercent) / 100);
      const totalCharge = principal + clientFee;
      const clinicCost = Math.round((totalCharge * commissionPercent) / 100);

      creditFeeTotal += clientFee;
      paidSum += totalCharge;

      creditBreakdown[idx] = {
        principal,
        clientFee,
        totalCharge,
        clinicCost,
        providerTitle: cp?.title || "سرویس اعتباری",
      };
    } else {
      paidSum += Math.round(rawAmt);
    }
  });

  const finalTotal = baseTotalBeforeCreditFee + creditFeeTotal;
  const remaining = Math.max(0, finalTotal - paidSum);
  const excessOverpayment = Math.max(0, paidSum - finalTotal);

  const installmentsSum =
    remaining > 0 && input.remainingMode === "INSTALLMENT"
      ? input.installments.reduce(
          (sum, ins) => sum + Math.max(0, Math.round(Number(ins.amount) || 0)),
          0
        )
      : 0;

  let status: InvoiceCalculationResult["status"] = "PAID";
  if (remaining > 0) {
    if (input.remainingMode === "INSTALLMENT") {
      status = "INSTALLMENT";
    } else if (paidSum > 0) {
      status = "PARTIAL";
    } else {
      status = "UNPAID";
    }
  }

  // اعتبارسنجی دقیق و شفاف
  let validationError: string | null = null;

  if (input.services.length === 0) {
    validationError = "حداقل یک زیرسرویس باید انتخاب شود.";
  } else if (excessOverpayment > 0) {
    validationError = `مجموع مبالغ پرداختی (${paidSum.toLocaleString("fa-IR")} تومان) به میزان ${excessOverpayment.toLocaleString("fa-IR")} تومان از مبلغ کل فاکتور (${finalTotal.toLocaleString("fa-IR")} تومان) بیشتر است. لطفاً مبالغ پرداختی را اصلاح کنید.`;
  } else if (walletUsedTotal > input.clientWalletBalance) {
    validationError = `مبلغ برداشتی از کیف پول (${walletUsedTotal.toLocaleString("fa-IR")} تومان) از موجودی کیف پول مراجع (${input.clientWalletBalance.toLocaleString("fa-IR")} تومان) بیشتر است.`;
  } else if (depositAmount > 0 && !input.depositServiceSelected) {
    validationError = "برای ثبت مبلغ بیعانه، انتخاب سرویس مربوطه الزامی است.";
  } else {
    for (let i = 0; i < input.payments.length; i++) {
      const p = input.payments[i];
      if ((Number(p.amount) || 0) < 0) {
        validationError = `مبلغ ردیف پرداخت شماره ${(i + 1).toLocaleString("fa-IR")} نمی‌تواند منفی باشد.`;
        break;
      }
      if (
        (Number(p.amount) || 0) > 0 &&
        p.paymentMethod === "POS" &&
        input.hasActiveTerminals &&
        (!p.terminalId || p.terminalId === "NONE")
      ) {
        validationError = `لطفاً دستگاه کارت‌خوان ردیف پرداخت شماره ${(i + 1).toLocaleString("fa-IR")} را انتخاب کنید.`;
        break;
      }
    }
  }

  if (!validationError && remaining > 0 && input.remainingMode === "INSTALLMENT") {
    if (input.installments.length === 0) {
      validationError = "لطفاً حداقل یک قسط تعریف کنید.";
    } else if (installmentsSum !== remaining) {
      validationError = `مجموع اقساط (${installmentsSum.toLocaleString("fa-IR")} تومان) با مانده حساب (${remaining.toLocaleString("fa-IR")} تومان) برابر نیست.`;
    }
  }

  return {
    servicesSum,
    consumablesSellSum,
    subTotal,
    discountAmount,
    extraChargeAmount,
    depositAmount,
    baseTotalBeforeCreditFee,
    creditFeeTotal,
    finalTotal,
    paidSum,
    walletUsedTotal,
    remaining,
    excessOverpayment,
    installmentsSum,
    status,
    validationError,
    creditBreakdown,
  };
}

function parseAttributes(raw: unknown): AttributeEntry[] {
  if (!Array.isArray(raw)) return [];
  return raw
    .filter(
      (item): item is { key?: unknown; value?: unknown } =>
        typeof item === "object" && item !== null
    )
    .map((item) => ({
      key: String(item.key ?? ""),
      value: String(item.value ?? ""),
    }))
    .filter((item) => item.key !== "");
}

export default async function NewClientInvoicePage({
  params,
}: {

function getCurrentTimeHHMM() {
  const now = new Date();
  return `${String(now.getHours()).padStart(2, "0")}:${String(
    now.getMinutes()
  ).padStart(2, "0")}`;
}

function formatDateToHHMM(isoOrDate: string | Date) {
  const dt = isoOrDate instanceof Date ? isoOrDate : new Date(isoOrDate);
  if (Number.isNaN(dt.getTime())) return getCurrentTimeHHMM();
  return `${String(dt.getHours()).padStart(2, "0")}:${String(
    dt.getMinutes()
  ).padStart(2, "0")}`;
}

            </Field>

            <Field>
              <FieldLabel>ساعت ثبت پرداخت *</FieldLabel>
              <ResponsiveTimePicker
                open={timePickerOpen}
                onOpenChange={setTimePickerOpen}
              >
                <InputGroup>
                  <InputGroupInput
                    dir="ltr"
                    inputMode="numeric"
                    placeholder="۱۴:۳۰"
                    value={paymentTimeText}
                    onChange={(e) => setPaymentTimeText(e.target.value)}
                    onBlur={(e) => commitTypedPaymentTime(e.target.value)}
                    onKeyDown={(e) => {
                      if (e.key === "Enter") {
                        e.preventDefault();
                        commitTypedPaymentTime(paymentTimeText);
                      }
                    }}
                    className="text-right tabular-nums"
                  />
                  <InputGroupAddon align="inline-end">
                    <ResponsiveTimePickerTrigger
                      render={
                        <InputGroupButton
                          type="button"
                          size="icon-xs"
                          variant="ghost"
                          aria-label="انتخاب ساعت از لیست"
                        >
                          <Clock className="size-4" />
                        </InputGroupButton>
                      }
                    />
                  </InputGroupAddon>
                </InputGroup>

                <ResponsiveTimePickerContent
                  value={parsedPaymentTime}
                  onValueChange={(next) => {
                    const formatted = `${String(next.hour).padStart(2, "0")}:${String(next.minute).padStart(2, "0")}`;
                    onPaymentTimeChange(formatted);
                    setPaymentTimeText(toPersianDigits(formatted));
                  }}
                  hourFormat="24"
                  digits="fa"
                  drawerTitle="انتخاب ساعت پرداخت"
                  drawerConfirmLabel="تایید ساعت"
                  onDrawerConfirm={() => setTimePickerOpen(false)}
                />
              </ResponsiveTimePicker>
            </Field>
          </div>

          <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
            <Field>
              <FieldLabel>مبلغ تخفیف (تومان)</FieldLabel>
              <PriceInput
                locale="fa-IR"
                disabled={!canApplyDiscount}
                value={discountAmount}
                onValueChange={(v) => onDiscountAmountChange(v || 0)}
              />
              {discountAmount > 0 && (
                <Input
                  placeholder="دلیل تخفیف (اختیاری)"
                  value={discountReason}
                  onChange={(e) => onDiscountReasonChange(e.target.value)}
                />
              )}
            </Field>

            <Field>
              <FieldLabel>اضافه پرداخت / هزینه جانبی (تومان)</FieldLabel>
              <PriceInput
                locale="fa-IR"
                value={extraChargeAmount}
                onValueChange={(v) => onExtraChargeAmountChange(v || 0)}
              />
              {extraChargeAmount > 0 && (
                <Input
                  placeholder="بابت چه موردی؟ (اختیاری)"
                  value={extraChargeReason}
                  onChange={(e) => onExtraChargeReasonChange(e.target.value)}
                />
              )}
            </Field>
          </div>
        </FieldGroup>
      </div>

      {/* بخش بیعانه سرویس آینده (شارژ کیف پول همراه با فاکتور) */}
      <div className="space-y-3 rounded-2xl border border-primary/25 bg-primary/5 p-5">
        <div className="flex flex-wrap items-center justify-between gap-2">
          <div className="flex items-center gap-2 text-sm font-bold text-primary">
            <BookmarkCheck className="size-4" />
            پرداخت بیعانه بابت سرویس آینده (شارژ کیف پول در همین فاکتور)
          </div>
          {calculation.depositAmount > 0 && (
            <Badge className="bg-emerald-600 text-white">
              +{calculation.depositAmount.toLocaleString("fa-IR")} تومان واریز
              به کیف پول
            </Badge>
          )}
        </div>

        <FieldGroup>
          <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
            <Field>
              <FieldLabel className="text-xs">
                انتخاب سرویس (جهت توضیحات بیعانه)
              </FieldLabel>
              <Select
                value={depositServiceId}
                items={allSubServiceOptions}
                onValueChange={(val) => {
                  onDepositServiceIdChange(val || "NONE")
                  if (!val || val === "NONE") {
                    const prevDep = Number(depositAmount) || 0
                    onDepositAmountChange(0)
                    if (prevDep > 0) {
                      onPaymentsChange((prev) => {
                        if (
                          prev.length === 1 &&
                          prev[0].paymentMethod !== "CREDIT"
                        ) {
                          return [
                            {
                              ...prev[0],
                              amount: Math.max(
                                0,
                                (Number(prev[0].amount) || 0) - prevDep
                              ),
                            },
                          ]
                        }
                        return prev
                      })
                    }
                  }
                }}
              >
                <SelectTrigger className="bg-background">
                  <SelectValue placeholder="انتخاب سرویس..." />
                </SelectTrigger>
                <SelectContent>
                  {allSubServiceOptions.map((opt) => (
                    <SelectItem key={opt.value} value={opt.value}>
                      {opt.label}
                    </SelectItem>
                  ))}
                </SelectContent>
              </Select>
            </Field>

            <Field>
              <FieldLabel className="text-xs">مبلغ بیعانه (تومان)</FieldLabel>
              <PriceInput
                locale="fa-IR"
                value={depositAmount}
                onValueChange={(v) => {
                  const nextDep = v || 0
                  const diff = nextDep - (Number(depositAmount) || 0)
                  onDepositAmountChange(nextDep)
                  onPaymentsChange((prev) => {
                    if (
                      prev.length === 1 &&
                      prev[0].paymentMethod !== "CREDIT"
                    ) {
                      return [
                        {
                          ...prev[0],
                          amount: Math.max(
                            0,
                            (Number(prev[0].amount) || 0) + diff
                          ),
                        },
                      ]
                    }
                    return prev
                  })
                }}
              />
            </Field>
          </div>
        </FieldGroup>
      </div>

      {/* روش‌های پرداخت */}
      <div className="space-y-4 rounded-2xl border bg-card p-5 shadow-2xs">
        <div className="flex flex-wrap items-center justify-between gap-2">
          <div className="flex items-center gap-2 text-sm font-bold">
            <CreditCard className="size-4 text-primary" />
            روش‌های پرداخت
          </div>
          <div className="flex items-center gap-2">
            <Badge variant="outline" className="gap-1.5 text-xs">
              <Wallet className="size-3.5 text-emerald-600" />
              موجودی کیف پول: {clientWalletBalance.toLocaleString("fa-IR")}{" "}
              تومان
            </Badge>
            <Button
              type="button"
              variant="ghost"
              size="sm"
              className="h-7 gap-1 text-xs"
              nativeButton={false}
              render={<Link href="/clinic-dashboard/settings/financial" />}
            >
              <Settings className="size-3.5" />
              تنظیمات کارمزد
            </Button>
          </div>
        </div>

        <div className="space-y-3">
          {payments.map((pay, idx) => {
            const creditInfo = calculation.creditBreakdown[idx]
            const selectedCreditConfig =
              pay.paymentMethod === "CREDIT"
                ? creditProviders.find((cp) => cp.id === pay.creditProviderId)
                : undefined

            return (
              <div
                key={pay.uid}
                className="space-y-3 rounded-xl border bg-muted/25 p-4"
              >
                <div className="flex items-center justify-between">
                  <Badge variant="secondary">
                    {(idx + 1).toLocaleString("fa-IR")}
                  </Badge>
                  {payments.length > 1 && (
                    <Button
                      type="button"
                      variant="ghost"
                      size="icon"
                      className="size-7 text-destructive"
                      onClick={() =>
                        onPaymentsChange((prev) =>
                          prev.filter((r) => r.uid !== pay.uid)
                        )
                      }
                    >
                      <Trash2 className="size-4" />
                    </Button>
                  )}
                </div>

                <FieldGroup>
                  <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
                    <Field>
                      <FieldLabel className="text-xs">روش پرداخت</FieldLabel>
                      <Select
                        value={pay.paymentMethod}
                        items={PAYMENT_METHOD_OPTIONS}
                        onValueChange={(val) =>
                          onPaymentsChange((prev) =>
                            prev.map((row) =>
                              row.uid === pay.uid
                                ? {
                                    ...row,
                                    paymentMethod:
                                      (val as DraftPaymentRow["paymentMethod"]) ||
                                      "POS",
                                  }
                                : row
                            )
                          )
                        }
                      >
                        <SelectTrigger className="bg-background">
                          <SelectValue />
                        </SelectTrigger>
                        <SelectContent>
                          {PAYMENT_METHOD_OPTIONS.map((opt) => (
                            <SelectItem key={opt.value} value={opt.value}>
                              {opt.label}
                            </SelectItem>
                          ))}
                        </SelectContent>
                      </Select>
                    </Field>

                    <Field>
                      <FieldLabel className="text-xs">
                        {pay.paymentMethod === "CREDIT"
                          ? "اصل مبلغ اعتباری (بدون کارمزد)"
                          : "مبلغ (تومان)"}
                      </FieldLabel>
                      <PriceInput
                        locale="fa-IR"
                        value={pay.amount}
                        onValueChange={(v) =>
                          onPaymentsChange((prev) =>
                            prev.map((row) =>
                              row.uid === pay.uid
                                ? { ...row, amount: v || 0 }
                                : row
                            )
                          )
                        }
                      />
                    </Field>
                  </div>

                  {pay.paymentMethod === "POS" && (
                    <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
                      <Field>
                        <FieldLabel className="text-xs">
                          دستگاه کارت‌خوان (POS)
                        </FieldLabel>
                        <Select
                          value={pay.terminalId}
                          items={[
                            { value: "NONE", label: "انتخاب کنید" },
                            ...terminals,
                          ]}
                          onValueChange={(val) =>
                            onPaymentsChange((prev) =>
                              prev.map((row) =>
                                row.uid === pay.uid
                                  ? { ...row, terminalId: val || "NONE" }
                                  : row
                              )
                            )
                          }
                        >
                          <SelectTrigger className="bg-background">
                            <SelectValue />
                          </SelectTrigger>
                          <SelectContent>
                            <SelectItem value="NONE">انتخاب کنید</SelectItem>
                            {terminals.map((t) => (
                              <SelectItem key={t.value} value={t.value}>
                                {t.label}
                              </SelectItem>
                            ))}
                          </SelectContent>
                        </Select>
                      </Field>

                      <Field>
                        <FieldLabel className="text-xs">
                          شماره رهگیری (اختیاری)
                        </FieldLabel>
                        <Input
                          dir="ltr"
                          className="bg-background"
                          value={pay.trackingCode}
                          onChange={(e) =>
                            onPaymentsChange((prev) =>
                              prev.map((row) =>
                                row.uid === pay.uid
                                  ? { ...row, trackingCode: e.target.value }
                                  : row
                              )
                            )
                          }
                        />
                      </Field>
                    </div>
                  )}

                  {pay.paymentMethod === "CREDIT" && (
                    <div className="space-y-3">
                      <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
                        <Field>
                          <FieldLabel className="text-xs">
                            سرویس اعتباری *
                          </FieldLabel>
                          <Select
                            value={pay.creditProviderId}
                            items={creditProviderOptions}
                            onValueChange={(val) =>
                              onPaymentsChange((prev) =>
                                prev.map((row) =>
                                  row.uid === pay.uid
                                    ? {
                                        ...row,
                                        creditProviderId: val || "CUSTOM",
                                      }
                                    : row
                                )
                              )
                            }
                          >
                            <SelectTrigger className="bg-background">
                              <SelectValue />
                            </SelectTrigger>
                            <SelectContent>
                              {creditProviderOptions.map((opt) => (
                                <SelectItem key={opt.value} value={opt.value}>
                                  {opt.label}
                                </SelectItem>
                              ))}
                            </SelectContent>
                          </Select>
                        </Field>

                        <Field>
                          <FieldLabel className="text-xs">
                            شماره قرارداد / رهگیری
                          </FieldLabel>
                          <Input
                            dir="ltr"
                            className="bg-background"
                            value={pay.trackingCode}
                            onChange={(e) =>
                              onPaymentsChange((prev) =>
                                prev.map((row) =>
                                  row.uid === pay.uid
                                    ? { ...row, trackingCode: e.target.value }
                                    : row
                                )
                              )
                            }
                          />
                        </Field>
                      </div>

                      {pay.creditProviderId === "CUSTOM" && (
                        <Input
                          className="bg-background text-xs"
                          placeholder="نام سرویس اعتباری را وارد کنید..."
                          value={pay.customCreditProvider}
                          onChange={(e) =>
                            onPaymentsChange((prev) =>
                              prev.map((row) =>
                                row.uid === pay.uid
                                  ? {
                                      ...row,
                                      customCreditProvider: e.target.value,
                                    }
                                  : row
                              )
                            )
                          }
                        />
                      )}

                      <div className="flex flex-col gap-3 rounded-xl border border-primary/25 bg-primary/5 p-3 text-xs sm:flex-row sm:items-center sm:justify-between">
                        <div className="space-y-1">
                          <div className="flex items-center gap-1.5 font-bold text-primary">
                            <Percent className="size-3.5" />
                            محاسبه کارمزد{" "}
                            {selectedCreditConfig?.title || "سرویس اعتباری"}
                          </div>
                          <div className="flex flex-wrap gap-x-4 gap-y-1 text-muted-foreground">
                            <span>
                              کارمزد مراجع (
                              {selectedCreditConfig?.clientFeePercent ?? 0}%):{" "}
                              <strong className="text-amber-600">
                                +
                                {(creditInfo?.clientFee ?? 0).toLocaleString(
                                  "fa-IR"
                                )}{" "}
                                تومان
                              </strong>
                            </span>
                            <span>
                              جمع قابل کسر از اعتبار:{" "}
                              <strong className="text-emerald-700">
                                {(creditInfo?.totalCharge ?? 0).toLocaleString(
                                  "fa-IR"
                                )}{" "}
                                تومان
                              </strong>
                            </span>
                          </div>
                        </div>

                        <div className="flex flex-wrap gap-2">
                          <Button
                            type="button"
                            size="sm"
                            variant="secondary"
                            className="h-8 text-xs"
                            onClick={() =>
                              handleSettleRemainingWithCredit(pay.uid)
                            }
                          >
                            تسویه باقیمانده با اعتبار
                          </Button>
                          <Button
                            type="button"
                            size="sm"
                            variant="outline"
                            className="h-8 text-xs"
                            onClick={() => handlePayFullWithCredit(pay.uid)}
                          >
                            پرداخت کل فاکتور با اعتبار
                          </Button>
                        </div>
                      </div>
                    </div>
                  )}
                </FieldGroup>
              </div>
            )
          })}
        </div>

        <Button
          type="button"
          variant="outline"
          className="w-full border-dashed text-emerald-700 hover:bg-emerald-500/10"
          onClick={() =>
            onPaymentsChange((prev) => [
              ...prev,
              {
                uid: crypto.randomUUID(),
                paymentMethod: "CASH",
                amount: calculation.remaining,
                terminalId: terminals[0]?.value || "NONE",
                creditProviderId: creditProviders[0]?.id || "CUSTOM",
                customCreditProvider: "",
                trackingCode: "",
              },
            ])
          }
        >
          <Plus className="ml-1.5 size-4" />
          افزودن روش پرداخت
        </Button>
      </div>

      {/* بخش مانده حساب و اقساط */}
      {calculation.remaining > 0 && (
        <div className="space-y-4 rounded-2xl border bg-card p-5 shadow-2xs">
          <div className="flex items-center justify-between">
            <div>
              <div className="text-base font-extrabold text-amber-600">
                {calculation.remaining.toLocaleString("fa-IR")} تومان
              </div>
              <div className="text-xs text-muted-foreground">
                مبلغ مانده — گزینه پرداخت مبلغ مانده را انتخاب کنید:
              </div>
            </div>
          </div>

          <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
            <button
              type="button"
              onClick={() => onRemainingModeChange("DEBT")}
              className={`rounded-xl border p-3.5 text-right transition-all ${
                remainingMode === "DEBT"
                  ? "border-primary bg-primary text-primary-foreground shadow-xs"
                  : "bg-background hover:bg-muted"
              }`}
            >
              <div className="font-bold">بدهکار</div>
              <div className="text-xs opacity-80">پرداخت در آینده</div>
            </button>

            <button
              type="button"
              onClick={() => {
                onRemainingModeChange("INSTALLMENT")
                if (installments.length === 0) {
                  handleSplitInstallmentsEqually(1)
                }
              }}
              className={`rounded-xl border p-3.5 text-right transition-all ${
                remainingMode === "INSTALLMENT"
                  ? "border-primary bg-primary text-primary-foreground shadow-xs"
                  : "bg-background hover:bg-muted"
              }`}
            >
              <div className="font-bold">اقساط</div>
              <div className="text-xs opacity-80">پرداخت قسطی / چک</div>
            </button>
          </div>

          {remainingMode === "INSTALLMENT" && (
            <div className="space-y-4 rounded-xl border border-primary/20 bg-primary/5 p-4">
              <div className="flex flex-wrap items-center justify-between gap-2">
                <div className="flex items-center gap-2 text-sm font-bold">
                  <CalendarIcon className="size-4 text-primary" />
                  تنظیم اقساط
                </div>

                <div className="flex items-center gap-2">
                  <Button
                    type="button"
                    size="sm"
                    onClick={() =>
                      handleSplitInstallmentsEqually(installments.length || 1)
                    }
                  >
                    تقسیم مساوی
                  </Button>
                  <Button
                    type="button"
                    size="sm"
                    variant="outline"
                    onClick={() =>
                      handleSplitInstallmentsEqually(installments.length + 1)
                    }
                  >
                    + افزودن قسط جدید
                  </Button>
                </div>
              </div>

              <div className="space-y-3">
                {installments.map((ins, idx) => (
                  <div
                    key={ins.uid}
                    className="space-y-3 rounded-xl border bg-background p-3.5"
                  >
                    <div className="flex items-center justify-between">
                      <Badge variant="secondary">
                        قسط {(idx + 1).toLocaleString("fa-IR")}
                      </Badge>
                      {installments.length > 1 && (
                        <Button
                          type="button"
                          variant="ghost"
                          size="icon"
                          className="size-7 text-destructive"
                          onClick={() =>
                            onInstallmentsChange((prev) =>
                              prev.filter((r) => r.uid !== ins.uid)
                            )
                          }
                        >
                          <Trash2 className="size-3.5" />
                        </Button>
                      )}
                    </div>

                    <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
                      <Field>
                        <FieldLabel className="text-xs">مبلغ قسط</FieldLabel>
                        <PriceInput
                          locale="fa-IR"
                          value={ins.amount}
                          onValueChange={(v) =>
                            onInstallmentsChange((prev) =>
                              prev.map((r) =>
                                r.uid === ins.uid ? { ...r, amount: v || 0 } : r
                              )
                            )
                          }
                        />
                      </Field>

                      <Field>
                        <FieldLabel className="text-xs">
                          تاریخ پرداخت
                        </FieldLabel>
                        <DatePicker
                          calendarType="shamsi"
                          value={ins.dueDate}
                          onValueChange={(d) =>
                            onInstallmentsChange((prev) =>
                              prev.map((r) =>
                                r.uid === ins.uid ? { ...r, dueDate: d } : r
                              )
                            )
                          }
                        />
                      </Field>
                    </div>

                    <div className="flex flex-wrap items-center gap-3">
                      <div className="flex items-center gap-2">
                        <Checkbox
                          id={`chk-${ins.uid}`}
                          checked={ins.isCheck}
                          onCheckedChange={(checked) =>
                            onInstallmentsChange((prev) =>
                              prev.map((r) =>
                                r.uid === ins.uid
                                  ? { ...r, isCheck: Boolean(checked) }
                                  : r
                              )
                            )
                          }
                        />
                        <Label
                          htmlFor={`chk-${ins.uid}`}
                          className="cursor-pointer text-xs"
                        >
                          این قسط چک است
                        </Label>
                      </div>

                      {ins.isCheck && (
                        <Input
                          dir="ltr"
                          className="h-8 w-44 text-xs"
                          placeholder="شماره چک..."
                          value={ins.checkNumber}
                          onChange={(e) =>
                            onInstallmentsChange((prev) =>
                              prev.map((r) =>
                                r.uid === ins.uid
                                  ? { ...r, checkNumber: e.target.value }
                                  : r
                              )
                            )
                          }
                        />
                      )}
                    </div>
                  </div>
                ))}
              </div>

              <div className="grid grid-cols-3 gap-2 rounded-xl border bg-background p-3 text-center text-xs">
                <div>
                  <div className="text-muted-foreground">مانده کل</div>
                  <strong className="mt-0.5 block">
                    {calculation.remaining.toLocaleString("fa-IR")} تومان
                  </strong>
                </div>
                <div>
                  <div className="text-muted-foreground">مجموع اقساط</div>
                  <strong className="mt-0.5 block text-primary">
                    {calculation.installmentsSum.toLocaleString("fa-IR")} تومان
                  </strong>
                </div>
                <div>
                  <div className="text-muted-foreground">تفاوت</div>
                  <strong
                    className={`mt-0.5 block ${
                      calculation.remaining - calculation.installmentsSum === 0
                        ? "text-emerald-600"
                        : "text-destructive"
                    }`}
                  >
                    {(
                      calculation.remaining - calculation.installmentsSum
                    ).toLocaleString("fa-IR")}{" "}
                    تومان
                  </strong>
                </div>
              </div>
            </div>
          )}
        </div>
      )}

      {/* نمایش خطای اعتبارسنجی فاکتور قبل از ثبت نهایی */}
      {calculation.validationError && (
        <div
          role="alert"
          className="flex items-start gap-2.5 rounded-xl border-2 border-red-500/60 bg-red-500/10 p-4 text-xs text-red-700 dark:text-red-300"
        >
          <AlertTriangle className="mt-0.5 size-4 shrink-0 text-red-600" />
          <div className="space-y-0.5">
            <div className="font-bold">
              مشکلی در مقادیر فاکتور وجود دارد (دکمه ثبت نهایی غیرفعال است):
            </div>
            <div>{calculation.validationError}</div>
          </div>
        </div>
      )}

      {/* دکمه‌های پایین صفحه */}
      <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
        <Button
          type="button"
          size="lg"
          disabled={isSubmitting || Boolean(calculation.validationError)}
          onClick={onSubmitInvoice}
          className="w-full"
        >
          {isSubmitting
            ? "در حال پردازش..."
            : isEditMode
              ? "ذخیره و به‌روزرسانی نهایی فاکتور"
              : "ثبت فاکتور"}
        </Button>

        <Button
          type="button"
          size="lg"
          variant="outline"
          disabled={isSubmitting}
          onClick={onBackStep}
          className="w-full"
        >
          بازگشت
        </Button>
      </div>
    </div>
  )
}

function parseAttributes(raw: unknown): AttributeEntry[] {
  if (!Array.isArray(raw)) return [];
  return raw
    .filter(
      (item): item is { key?: unknown; value?: unknown } =>
        typeof item === "object" && item !== null
    )
    .map((item) => ({
      key: String(item.key ?? ""),
      value: String(item.value ?? ""),
    }))
    .filter((item) => item.key !== "");
}

export default async function EditClientInvoicePage({
  params,
  searchParams,
}: {
  params: Promise<{ id: string; invoiceId: string }>;
  searchParams: Promise<{ step?: string }>;
}) {
  const ctx = await getCurrentClinicContext();
  if (!ctx) redirect("/clinic-dashboard/login");

  const { id, invoiceId } = await params;
  const { step } = await searchParams;
  const requestedStep = step === "3" ? 3 : 1;

  if (requestedStep !== 3 && !hasPermission(ctx.clinicUser, "services.edit")) {
    redirect(`/clinic-dashboard/clients/${id}`);
  }

  const [client, clinic, operator, invoice] = await Promise.all([
    prisma.client.findFirst({
      where: { id, clinicId: ctx.clinicId },
      include: {
        user: {
          select: { fullName: true, phone: true, nationalId: true },
        },
      },
    }),
    prisma.clinic.findUnique({
      where: { id: ctx.clinicId },
      select: { name: true },
    }),
    prisma.clinicUser.findUnique({
      where: { id: ctx.clinicUserId },
      include: { user: { select: { fullName: true } } },
    }),
    prisma.invoice.findFirst({
      where: {
        id: invoiceId,
        clientId: id,
        clinicId: ctx.clinicId,
      },
      include: {
        providedServices: {
          include: {
            subService: {
              include: { service: { select: { id: true, title: true } } },
            },
            assistants: { select: { id: true } },
            usedConsumables: {
              include: {
                clinicProduct: { select: { name: true, unit: true } },
              },
            },
          },
          orderBy: { createdAt: "asc" },
        },
        payments: {
          orderBy: { createdAt: "asc" },
        },
        installments: {
          orderBy: { installmentNumber: "asc" },
        },
      },
    }),
  ]);

  if (!client || !clinic || !invoice) notFound();

  const serviceScopeFilter = buildServiceScopeFilter(ctx.clinicUser);

  const [
    catalogServices,
    products,
    staffMembers,
    clientReferrers,
    terminals,
    creditProviders,
  ] = await Promise.all([
    prisma.service.findMany({
      where: {
        clinicId: ctx.clinicId,
        ...serviceScopeFilter,
      },
      include: {
        defaultServiceProducts: {
          include: {
            clinicProduct: {
              select: { id: true, name: true, unit: true },
            },
          },
        },
        subServices: {
          include: {
            defaultProducts: {
              include: {
                clinicProduct: {
                  select: { id: true, name: true, unit: true },
                },
              },
            },
          },
          orderBy: { title: "asc" },
        },
      },
      orderBy: { title: "asc" },
    }),

    prisma.clinicProduct.findMany({
      where: { clinicId: ctx.clinicId },
      include: {
        batches: {
          where: { quantity: { gt: 0 } },
          orderBy: [{ expirationDate: "asc" }, { createdAt: "asc" }],
        },
      },
      orderBy: { name: "asc" },
    }),

    prisma.clinicUser.findMany({
      where: {
        clinicId: ctx.clinicId,
        deletedAt: null,
        isActive: true,
      },
      include: {
        user: { select: { fullName: true } },
        assignedServices: { select: { id: true } },
      },
      orderBy: { createdAt: "asc" },
    }),

    prisma.client.findMany({
      where: {
        clinicId: ctx.clinicId,
        id: { not: client.id },
      },
      include: {
        user: { select: { fullName: true } },
      },
      orderBy: { createdAt: "desc" },
      take: 200,
    }),

    prisma.paymentTerminal.findMany({
      where: { clinicId: ctx.clinicId, isActive: true },
      include: {
        bankAccount: { select: { bankName: true } },
      },
    }),

    prisma.clinicCreditProvider.findMany({
      where: { clinicId: ctx.clinicId, isActive: true },
      orderBy: { createdAt: "asc" },
    }),
  ]);

  const blocksByServiceId = new Map<string, DraftServiceAccordionBlock>();
  for (const ps of invoice.providedServices) {
    const sId = ps.subService.service.id;
    const existingBlock = blocksByServiceId.get(sId);

    const subSelection = {
      subServiceId: ps.subServiceId,
      subServiceTitle: ps.subService.title,
      finalPrice: ps.finalPrice,
      attributes: parseAttributes(ps.attributes),
    };

    const mappedConsumables = ps.usedConsumables.map((uc) => ({
      uid: uc.id,
      clinicProductId: uc.clinicProductId,
      productName: uc.clinicProduct.name,
      unit: uc.clinicProduct.unit,
      preferredBatchId: uc.productBatchId || "AUTO",
      quantity: uc.quantity,
      sellingPrice: uc.sellingPrice || 0,
      isDefault: false,
    }));

    if (existingBlock) {
      existingBlock.allowMultiSubServices = true;
      existingBlock.selectedSubServices.push(subSelection);
      existingBlock.consumables.push(...mappedConsumables);
    } else {
      blocksByServiceId.set(sId, {
        uid: ps.id,
        serviceId: sId,
        allowMultiSubServices: true,
        selectedSubServices: [subSelection],
        expandedSubServiceId: ps.subServiceId,
        consumables: mappedConsumables,
        selectedNewProductId: "NONE",
        selectedBatchForNewProduct: "AUTO",
        description: ps.description || "",
      });
    }
  }

  let resolvedDepositSubId = "NONE";
  if (invoice.depositAmount > 0 && invoice.depositServiceTitle) {
    for (const srv of catalogServices) {
      for (const sub of srv.subServices) {
        if (invoice.depositServiceTitle.includes(sub.title)) {
          resolvedDepositSubId = sub.id;
          break;
        }
      }
    }
  }

  const firstPs = invoice.providedServices[0];
  let refType: "NONE" | "STAFF" | "CLIENT" = "NONE";
  let refId = "NONE";
  if (firstPs?.referrerId) {
    refType = "STAFF";
    refId = firstPs.referrerId;
  } else if (firstPs?.referrerClientId) {
    refType = "CLIENT";
    refId = firstPs.referrerClientId;
  }

  const firstPayment = invoice.payments[0];
  const storedPaidAt =
    firstPayment?.paidAt ?? firstPayment?.createdAt ?? invoice.createdAt;

  const initialEditData: InitialInvoiceEditData = {
    invoiceId: invoice.id,
    invoiceNumber: invoice.invoiceNumber,
    performedAt: (firstPs?.performedAt || invoice.createdAt).toISOString(),
    paidAt: storedPaidAt.toISOString(),
    therapistId: firstPs?.therapistId || "NONE",
    assistantId: firstPs?.assistants[0]?.id || "NONE",
    referrerType: refType,
    referrerId: refId,
    discountAmount: invoice.discountAmount,
    discountReason: invoice.discountReason || "",
    extraChargeAmount: invoice.extraChargeAmount,
    extraChargeReason: invoice.extraChargeReason || "",
    depositServiceId: resolvedDepositSubId,
    depositAmount: invoice.depositAmount,
    remainingMode:
      invoice.status === "INSTALLMENT" ? "INSTALLMENT" : "DEBT",
    serviceBlocks: Array.from(blocksByServiceId.values()),
    payments: invoice.payments.map((p) => ({
      uid: p.id,
      paymentMethod: p.paymentMethod as
        | "POS"
        | "CASH"
        | "CARD_TO_CARD"
        | "WALLET"
        | "CREDIT",
      amount:
        p.paymentMethod === "CREDIT" && p.creditPrincipal > 0
          ? p.creditPrincipal
          : p.amount,
      terminalId: p.terminalId || "NONE",
      creditProviderId: p.creditProviderId || "CUSTOM",
      customCreditProvider: p.creditProvider || "",
      trackingCode: p.trackingCode || "",
    })),
    installments: invoice.installments.map((ins) => ({
      uid: ins.id,
      amount: ins.amount,
      dueDate: ins.dueDate,
      isCheck: ins.isCheck,
      checkNumber: ins.checkNumber || "",
    })),
  };

  const walletSpentBefore = invoice.payments
    .filter((p) => p.paymentMethod === "WALLET")
    .reduce((sum, p) => sum + p.amount, 0);
  const editableWalletBalance = Math.max(
    0,
    client.walletBalance + walletSpentBefore - (invoice.depositAmount || 0)
  );

  const isNoPhone = client.user.phone.startsWith("NO_PHONE_");

  return (
    <InvoiceWizardContainer
      clinicName={clinic.name}
      operatorName={operator?.user?.fullName || operator?.jobTitle || "اپراتور"}
      client={{
        id: client.id,
        fullName: client.user.fullName || "مراجعه‌کننده",
        phone: isNoPhone ? null : client.user.phone,
        nationalId: client.user.nationalId,
        gender: client.gender,
        walletBalance: editableWalletBalance,
      }}
      catalogServices={catalogServices.map((s) => ({
        id: s.id,
        title: s.title,
        defaultServiceProducts: s.defaultServiceProducts.map((dp) => ({
          clinicProductId: dp.clinicProductId,
          productName: dp.clinicProduct.name,
          unit: dp.clinicProduct.unit,
          quantity: dp.quantity,
        })),
        subServices: s.subServices.map((sub) => ({
          id: sub.id,
          title: sub.title,
          price: sub.price,
          attributes: parseAttributes(sub.attributes),
          defaultProducts: sub.defaultProducts.map((dp) => ({
            clinicProductId: dp.clinicProductId,
            productName: dp.clinicProduct.name,
            unit: dp.clinicProduct.unit,
            quantity: dp.quantity,
          })),
        })),
      }))}
      products={products.map((p) => ({
        id: p.id,
        name: p.name,
        unit: p.unit,
        totalStock: p.batches.reduce((sum, b) => sum + b.quantity, 0),
        batches: p.batches.map((b) => ({
          id: b.id,
          quantity: b.quantity,
          purchasePrice: b.purchasePrice,
          sellingPrice: b.sellingPrice,
          expirationDate: b.expirationDate
            ? b.expirationDate.toISOString()
            : null,
          exclusiveSubServiceId: b.exclusiveSubServiceId,
          exclusiveTherapistId: b.exclusiveTherapistId,
        })),
      }))}
      therapists={staffMembers
        .filter((s) => s.canBeTherapist)
        .map((s) => ({
          id: s.id,
          label: s.user.fullName || s.jobTitle,
          hasAllServicesAccess: s.isOwner || s.hasAllServicesAccess,
          assignedServiceIds: s.assignedServices.map((as) => as.id),
        }))}
      assistants={staffMembers
        .filter((s) => s.canBeAssistant)
        .map((s) => ({
          value: s.id,
          label: s.user.fullName || s.jobTitle,
        }))}
      staffReferrers={staffMembers
        .filter((s) => s.canBeReferrer)
        .map((s) => ({
          value: s.id,
          label: s.user.fullName || s.jobTitle,
        }))}
      clientReferrers={clientReferrers.map((c) => ({
        value: c.id,
        label: `${c.user.fullName || "بدون نام"}${
          c.fileNumber ? ` (پرونده ${c.fileNumber})` : ""
        }`,
      }))}
      terminals={terminals.map((t) => ({
        value: t.id,
        label: `${t.name} (${t.bankAccount.bankName})`,
      }))}
      creditProviders={creditProviders.map((cp) => ({
        id: cp.id,
        title: cp.title,
        commissionPercent: cp.commissionPercent,
        clientFeePercent: cp.clientFeePercent,
      }))}
      canApplyDiscount={
        ctx.clinicUser.isOwner ||
        hasPermission(ctx.clinicUser, "other.discounts")
      }
      initialEditData={initialEditData}
      initialStep={requestedStep}
    />
  );
}

  return {

    serviceId: "NONE",

export function Step1Services({
  performedDate,
  onPerformedDateChange,
  performedTime,
  onPerformedTimeChange,
  therapistId,
  onTherapistIdChange,
  assistantId,
  onAssistantIdChange,
  referrerType,
  onReferrerTypeChange,
  referrerId,

  therapists,
  assistants,
  staffReferrers,
  clientReferrers,
  catalogServices,
  products,
  serviceBlocks,
  onServiceBlocksChange,
  openBlockUid,
  onOpenBlockUidChange,
  onNextStep,
}: Step1ServicesProps) {
  // فیلتر هوشمند خدمات بر اساس سطح دسترسی درمانگر انتخاب‌شده
  const allowedServices = useMemo(() => {
    if (!therapistId || therapistId === "NONE") return catalogServices
    const therapist = therapists.find((t) => t.id === therapistId)
    if (!therapist || therapist.hasAllServicesAccess) return catalogServices
    return catalogServices.filter((s) =>
      therapist.assignedServiceIds.includes(s.id)
    )
  }, [therapistId, therapists, catalogServices])

  const serviceOptions: SelectOption[] = useMemo(
    () => [
      { value: "NONE", label: "انتخاب خدمت..." },
      ...allowedServices.map((s) => ({ value: s.id, label: s.title })),
    ],
    [allowedServices]
  )

  const [timePickerOpen, setTimePickerOpen] = useState(false)
  const [timeText, setTimeText] = useState(() =>
    toPersianDigits(performedTime || "12:00")
  )

  // همگام‌سازی متن با تغییر بیرونی یا انتخاب از چرخنده
  useEffect(() => {
    setTimeText(toPersianDigits(performedTime || "12:00"))
  }, [performedTime])

  const parsedPerformedTime = useMemo(() => {
    const [hh, mm] = (performedTime || "00:00").split(":").map(Number)
    return {
      hour: Number.isFinite(hh) ? hh : 0,
      minute: Number.isFinite(mm) ? mm : 0,
    }
  }, [performedTime])

  function commitTypedTime(rawInput: string) {
    const clean = normalizePersianDigits(rawInput).replace(/[^\d:]/g, "")
    let hh = 0
    let mm = 0

    if (clean.includes(":")) {
      const [hStr, mStr] = clean.split(":")
      hh = parseInt(hStr || "0", 10)
      mm = parseInt(mStr || "0", 10)
    } else if (clean.length <= 2) {
      hh = parseInt(clean || "0", 10)
      mm = 0
    } else {
      const padded = clean.padStart(4, "0").slice(0, 4)
      hh = parseInt(padded.slice(0, 2), 10)
      mm = parseInt(padded.slice(2, 4), 10)
    }

    hh = Number.isFinite(hh) ? Math.min(23, Math.max(0, hh)) : 0
    mm = Number.isFinite(mm) ? Math.min(59, Math.max(0, mm)) : 0

    const formatted = `${String(hh).padStart(2, "0")}:${String(mm).padStart(2, "0")}`
    onPerformedTimeChange(formatted)
    setTimeText(toPersianDigits(formatted))
  }

  const therapistOptions: SelectOption[] = useMemo(
    () => [
      { value: "NONE", label: "انتخاب درمانگر..." },
      ...therapists.map((t) => ({ value: t.id, label: t.label })),
    ],
    [therapists]
  )

  const assistantOptions: SelectOption[] = useMemo(
    () => [{ value: "NONE", label: "انتخاب کنید" }, ...assistants],
    [assistants]
  )

  const referrerOptions: SelectOption[] = useMemo(
    () => [
      { value: "NONE", label: "انتخاب معرف..." },
      ...(referrerType === "STAFF"
        ? staffReferrers
        : referrerType === "CLIENT"
          ? clientReferrers
          : []),
    ],
    [referrerType, staffReferrers, clientReferrers]
  )

  const productOptions: SelectOption[] = useMemo(
    () => [
      { value: "NONE", label: "انتخاب کالا از انبار..." },
      ...products.map((p) => ({
        value: p.id,
        label: `${p.name} (موجودی: ${p.totalStock.toLocaleString("fa-IR")} ${p.unit})`,
      })),
    ],
    [products]
  )

  // هنگام تغییر درمانگر، خدماتی که درمانگر به آن‌ها دسترسی ندارد پاک‌سازی می‌شوند
  function handleTherapistChange(newTherapistId: string) {
    onTherapistIdChange(newTherapistId)
    if (!newTherapistId || newTherapistId === "NONE") return

    const therapist = therapists.find((t) => t.id === newTherapistId)
    if (!therapist || therapist.hasAllServicesAccess) return

    let removedCount = 0
    onServiceBlocksChange((prev) =>
      prev.map((block) => {
        if (
          block.serviceId !== "NONE" &&
          !therapist.assignedServiceIds.includes(block.serviceId)
        ) {
          removedCount++
          return createEmptyServiceBlock()
        }
        return block
      })
    )

    if (removedCount > 0) {
      toast.warning(
        "برخی خدمات انتخاب‌شده به دلیل عدم دسترسی درمانگر جدید بازنشانی شدند."
      )
    }
  }

  function updateBlock(
    uid: string,
    updater: (prev: DraftServiceAccordionBlock) => DraftServiceAccordionBlock
  ) {
    onServiceBlocksChange((prev) =>
      prev.map((b) => (b.uid === uid ? updater(b) : b))
    )
  }

  function handleToggleSubService(
    block: DraftServiceAccordionBlock,
    service: WizardCatalogService,
    subId: string
  ) {
    const sub = service.subServices.find((ss) => ss.id === subId)
    if (!sub) return

    const alreadySelected = block.selectedSubServices.some(
      (s) => s.subServiceId === sub.id
    )

    // ساخت مصرفی‌های دیفالت زیرسرویس و خدمت اصلی
    const buildDefaultConsumables = (subIds: string[]) => {
      const map = new Map<
        string,
        {
          clinicProductId: string
          productName: string
          unit: string
          quantity: number
          sellingPrice: number
        }
      >()

      // ۱. مصرفی‌های دیفالت سطح خدمت اصلی (فقط یک‌بار به ازای کل خدمت)
      if (subIds.length > 0) {
        for (const sdp of service.defaultServiceProducts ?? []) {
          const prod = products.find((p) => p.id === sdp.clinicProductId)
          const sp = prod?.batches[0]?.sellingPrice || 0
          const prevItem = map.get(sdp.clinicProductId)
          if (prevItem) {
            prevItem.quantity += sdp.quantity
          } else {
            map.set(sdp.clinicProductId, {
              clinicProductId: sdp.clinicProductId,
              productName: sdp.productName,
              unit: sdp.unit,
              quantity: sdp.quantity,
              sellingPrice: sp,
            })
          }
        }
      }

      // ۲. مصرفی‌های دیفالت اختصاصی هر زیرسرویس انتخاب‌شده
      for (const id of subIds) {
        const targetSub = service.subServices.find((x) => x.id === id)
        if (!targetSub) continue
        for (const dp of targetSub.defaultProducts) {
          const prod = products.find((p) => p.id === dp.clinicProductId)
          const sp = prod?.batches[0]?.sellingPrice || 0
          const prevItem = map.get(dp.clinicProductId)
          if (prevItem) {
            prevItem.quantity += dp.quantity
          } else {
            map.set(dp.clinicProductId, {
              clinicProductId: dp.clinicProductId,
              productName: dp.productName,
              unit: dp.unit,
              quantity: dp.quantity,
              sellingPrice: sp,
            })
          }
        }
      }

      return Array.from(map.values()).map((item) => ({
        uid: crypto.randomUUID(),
        clinicProductId: item.clinicProductId,
        productName: item.productName,
        unit: item.unit,
        preferredBatchId: "AUTO",
        quantity: item.quantity,
        sellingPrice: item.sellingPrice,
        isDefault: true,
      }))
    }

    if (!block.allowMultiSubServices) {
      // حالت رادیو باتن (تک انتخابی)
      const nextSelected: DraftSubServiceSelection[] = [
        {
          subServiceId: sub.id,
          subServiceTitle: sub.title,
          finalPrice: sub.price,
          attributes: sub.attributes.map((a) => ({ ...a })),
        },
      ]
      const manualConsumables = block.consumables.filter((c) => !c.isDefault)
      updateBlock(block.uid, (b) => ({
        ...b,
        selectedSubServices: nextSelected,

              subServiceId: sub.id,

        ...b,
        selectedSubServices: nextSelected,
        expandedSubServiceId: alreadySelected ? "" : sub.id,
        consumables: [
          ...buildDefaultConsumables(nextSelected.map((x) => x.subServiceId)),

      }))
    }

        (s, item) => s + (Number(item.finalPrice) || 0),

      const consSum = block.consumables.reduce(
        (c, item) =>
          c + (Number(item.quantity) || 0) * (Number(item.sellingPrice) || 0),
        0
      )
      return sum + subSum + consSum
    }, 0)
  }, [serviceBlocks])

                                    value={cons.quantity}
                                    onChange={(e) => {
                                      const q = parseFloat(e.target.value) || 0
                                      updateBlock(block.uid, (b) => ({
                                        ...b,
                                        consumables: b.consumables.map(
                                          (item) =>
                                            item.uid === cons.uid
                                              ? { ...item, quantity: q }
                                              : item
                                        ),
                                      }))
                                    }}
                                  />
                                  <span>{cons.unit}</span>
                                  <Button
                                    type="button"
                                    variant="ghost"
                                    size="icon"
                                    className="size-7 text-destructive"
                                    onClick={() =>
                                      updateBlock(block.uid, (b) => ({
                                        ...b,
                                        consumables: b.consumables.filter(
                                          (item) => item.uid !== cons.uid
                                        ),
                                      }))
                                    }
                                  >
                                    <Trash2 className="size-3.5" />
                                  </Button>
                                </div>
                              </div>
                            )
                          })}
                        </div>
                      )}
                    </div>

                    {/* باکس توضیحات */}
                    <Field>
                      <FieldLabel className="text-xs">توضیحات</FieldLabel>
                      <Textarea
                        rows={2}
                        placeholder="توضیحات مربوط به این خدمت..."
                        value={block.description}
                        onChange={(e) =>
                          updateBlock(block.uid, (b) => ({
                            ...b,
                            description: e.target.value,
                          }))
                        }
                      />
                    </Field>
                  </FieldGroup>
                </div>
              )}
            </div>
          )
        })}

        {/* دکمه افزودن آکاردیون خدمت جدید */}
        <Button
          type="button"
          variant="outline"
          className="w-full rounded-xl border-dashed py-5 text-primary hover:bg-primary/5"
          onClick={() => {
            const newBlock = createEmptyServiceBlock()
            onServiceBlocksChange((prev) => [...prev, newBlock])
            onOpenBlockUidChange(newBlock.uid)
          }}
        >
          <Plus className="ml-2 size-4" />
          افزودن خدمت دیگر به این فاکتور
        </Button>
      </div>

      {/* نوار جمع کل و رفتن به مرحله پرداخت */}
      <div className="flex flex-col items-center justify-between gap-3 rounded-2xl border bg-card p-4 shadow-2xs sm:flex-row">
        <div className="text-sm">
          جمع کل خدمات انتخاب‌شده:{" "}
          <strong className="text-lg font-extrabold text-primary">
            {grandTotalStep1.toLocaleString("fa-IR")} تومان
          </strong>
        </div>

        <Button
          type="button"
          size="lg"
          onClick={handleValidateAndContinue}
          className="w-full gap-2 sm:w-auto"
        >
          ادامه و ثبت پرداختی
          <ArrowLeft className="size-4" />
        </Button>
      </div>
    </div>
  )
}
</file>
