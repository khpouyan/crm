"use server";

import { revalidatePath } from "next/cache";
import { ServiceTimingMode, StaffScheduleStatus } from "@prisma/client";
import { prisma } from "@/lib/prisma";
import { getCurrentClinicContext } from "@/lib/auth/session";
import { hasPermission } from "@/lib/permissions";
import { normalizePersianDigits } from "@/lib/normalize-persian-digits";
import { toGregorian } from "@/lib/jalali";

const STAFF_SCHEDULE_PATH =
  "/clinic-dashboard/settings/appointments/staff-schedule";

interface ClinicShiftWindow {
  startTime: string;
  endTime: string;
}

interface ClinicWorkingDayPolicy {
  isClinicHoliday: boolean;
  holidayNote: string | null;
  shifts: ClinicShiftWindow[];
}

function isClinicScheduleAdmin(
  ctx: NonNullable<Awaited<ReturnType<typeof getCurrentClinicContext>>>
) {
  return (
    ctx.clinicUser.isOwner ||
    hasPermission(ctx.clinicUser, "appointments.settings") ||
    hasPermission(ctx.clinicUser, "management.settings")
  );
}

function canEditTargetStaff(
  ctx: NonNullable<Awaited<ReturnType<typeof getCurrentClinicContext>>>,
  targetClinicUserId: string
) {
  if (isClinicScheduleAdmin(ctx)) return true;
  return (
    ctx.clinicUser.id === targetClinicUserId &&
    hasPermission(ctx.clinicUser, "appointments.self_schedule")
  );
}

function toSafeMinutes(val: unknown, fallback: number, max = 600): number {
  const n = Number(normalizePersianDigits(String(val ?? fallback)).trim());
  if (!Number.isFinite(n) || n < 0) return fallback;
  return Math.min(max, Math.floor(n));
}

function isValidTimeFormat(val: string): boolean {
  return /^([01]\d|2[0-3]):([0-5]\d)$/.test(val);
}

function timeToMinutes(hhmm: string): number {
  const [h, m] = hhmm.split(":").map(Number);
  return h * 60 + m;
}

function isJalaliLeapYear(jy: number): boolean {
  const rem = (((jy - 474) % 2820) + 2820) % 2820;
  return (((rem + 38) * 682) % 2816) < 682;
}

function getJalaliMonthDaysCount(jy: number, jm: number): number {
  if (jm <= 6) return 31;
  if (jm <= 11) return 30;
  return isJalaliLeapYear(jy) ? 30 : 29;
}

function jalaliToGregorianUtc(jy: number, jm: number, jd: number): Date {
  const g = toGregorian(jy, jm, jd);
  return new Date(
    Date.UTC(g.getFullYear(), g.getMonth(), g.getDate(), 0, 0, 0, 0)
  );
}

function parseJalaliDateString(raw: string): {
  jy: number;
  jm: number;
  jd: number;
  utcDate: Date;
  weekday: number;
} | null {
  const normalized = normalizePersianDigits(raw || "").trim();
  const parts = normalized.split(/[/-]/).map((p) => Number(p.trim()));
  if (parts.length !== 3) return null;
  const [jy, jm, jd] = parts;
  if (
    !Number.isInteger(jy) ||
    !Number.isInteger(jm) ||
    !Number.isInteger(jd) ||
    jy < 1300 ||
    jy > 1500 ||
    jm < 1 ||
    jm > 12 ||
    jd < 1 ||
    jd > 31
  ) {
    return null;
  }

  const maxDayInMonth = getJalaliMonthDaysCount(jy, jm);
  if (jd > maxDayInMonth) return null;

  const utcDate = jalaliToGregorianUtc(jy, jm, jd);
  const weekday = (utcDate.getUTCDay() + 1) % 7;

  return { jy, jm, jd, utcDate, weekday };
}

function parseWorkingDayRow(row: {
  isClinicHoliday: boolean;
  holidayNote: string | null;
  shifts: unknown;
}): ClinicWorkingDayPolicy {
  const rawList = Array.isArray(row.shifts) ? row.shifts : [];
  const shifts: ClinicShiftWindow[] = rawList
    .map((item) => {
      if (!item || typeof item !== "object") return null;
      const obj = item as { startTime?: unknown; endTime?: unknown };
      const st = typeof obj.startTime === "string" ? obj.startTime.trim() : "";
      const en = typeof obj.endTime === "string" ? obj.endTime.trim() : "";
      if (!isValidTimeFormat(st) || !isValidTimeFormat(en) || st >= en) {
        return null;
      }
      return { startTime: st, endTime: en };
    })
    .filter((s): s is ClinicShiftWindow => Boolean(s));

  return {
    isClinicHoliday: row.isClinicHoliday,
    holidayNote: row.holidayNote,
    shifts,
  };
}

function validateStaffSlotAgainstClinicWorkingDay(
  policy: ClinicWorkingDayPolicy | undefined,
  jalaliLabel: string,
  startTime: string,
  endTime: string
): { ok: true } | { ok: false; error: string } {
  if (!policy) {
    return {
      ok: false,
      error: `برای تاریخ ${jalaliLabel} در بخش «ساعات کاری کلینیک» ساعات کاری ثبت نشده است. ابتدا ساعات کاری کلینیک را تعیین کنید.`,
    };
  }

  if (policy.isClinicHoliday) {
    const reason = policy.holidayNote ? ` (${policy.holidayNote})` : "";
    return {
      ok: false,
      error: `تاریخ ${jalaliLabel} تعطیل رسمی/کلینیک${reason} است و امکان ثبت زمان حضور پرسنل در روز تعطیل وجود ندارد.`,
    };
  }

  if (policy.shifts.length === 0) {
    return {
      ok: false,
      error: `در تاریخ ${jalaliLabel} هیچ بازه کاری فعالی برای کلینیک ثبت نشده است.`,
    };
  }

  const reqStart = timeToMinutes(startTime);
  const reqEnd = timeToMinutes(endTime);

  const fitsInsideAnyClinicShift = policy.shifts.some((sh) => {
    const shStart = timeToMinutes(sh.startTime);
    const shEnd = timeToMinutes(sh.endTime);
    return reqStart >= shStart && reqEnd <= shEnd;
  });

  if (!fitsInsideAnyClinicShift) {
    const allowedText = policy.shifts
      .map((s) => `${s.startTime} تا ${s.endTime}`)
      .join(" و ");
    return {
      ok: false,
      error: `بازه انتخاب‌شده (${startTime} تا ${endTime}) خارج از ساعات کاری کلینیک در روز ${jalaliLabel} است. بازه‌های مجاز کلینیک: ${allowedText}.`,
    };
  }

  return { ok: true };
}

export interface UpsertStaffServiceTimingInput {
  clinicUserId: string;
  serviceId: string;
  timingMode: ServiceTimingMode;
  prepBeforeMinutes: number;
  durationMinutes: number;
  stage1Minutes: number;
  freeGapMinutes: number;
  stage2Minutes: number;
  restAfterMinutes: number;
  requiresPrivacy: boolean;
}

export async function upsertStaffServiceTimingAction(
  input: UpsertStaffServiceTimingInput
) {
  try {
    const ctx = await getCurrentClinicContext();
    if (!ctx) {
      return { success: false, error: "نشست کاربری معتبر نیست." };
    }

    if (!canEditTargetStaff(ctx, input.clinicUserId)) {
      return {
        success: false,
        error: "شما مجوز ویرایش زمان‌بندی سرویس‌های این پرسنل را ندارید.",
      };
    }

    const targetStaff = await prisma.clinicUser.findFirst({
      where: {
        id: input.clinicUserId,
        clinicId: ctx.clinicId,
        deletedAt: null,
      },
      select: {
        id: true,
        isOwner: true,
        hasAllServicesAccess: true,
        assignedServices: { select: { id: true } },
      },
    });

    if (!targetStaff) {
      return { success: false, error: "پرسنل مورد نظر یافت نشد." };
    }

    const service = await prisma.service.findFirst({
      where: {
        id: input.serviceId,
        clinicId: ctx.clinicId,
      },
      select: { id: true },
    });

    if (!service) {
      return { success: false, error: "سرویس مورد نظر یافت نشد." };
    }

    const hasServiceAccess =
      targetStaff.isOwner ||
      targetStaff.hasAllServicesAccess ||
      targetStaff.assignedServices.some((s) => s.id === service.id);

    if (!hasServiceAccess) {
      return {
        success: false,
        error: "این سرویس به پرسنل انتخاب‌شده اختصاص داده نشده است.",
      };
    }

    const timingMode =
      input.timingMode === ServiceTimingMode.SPLIT_STAGE
        ? ServiceTimingMode.SPLIT_STAGE
        : ServiceTimingMode.SINGLE_STAGE;

    const prepBeforeMinutes = toSafeMinutes(input.prepBeforeMinutes, 0, 240);
    const durationMinutes = Math.max(
      5,
      toSafeMinutes(input.durationMinutes, 30, 480)
    );
    const stage1Minutes = Math.max(
      5,
      toSafeMinutes(input.stage1Minutes, 15, 240)
    );
    const freeGapMinutes = toSafeMinutes(input.freeGapMinutes, 0, 360);
    const stage2Minutes = Math.max(
      5,
      toSafeMinutes(input.stage2Minutes, 15, 240)
    );
    const restAfterMinutes = toSafeMinutes(input.restAfterMinutes, 0, 180);
    const requiresPrivacy = Boolean(input.requiresPrivacy);

    await prisma.clinicStaffServiceTiming.upsert({
      where: {
        clinicUserId_serviceId: {
          clinicUserId: targetStaff.id,
          serviceId: service.id,
        },
      },
      create: {
        clinicId: ctx.clinicId,
        clinicUserId: targetStaff.id,
        serviceId: service.id,
        timingMode,
        prepBeforeMinutes,
        durationMinutes,
        stage1Minutes,
        freeGapMinutes,
        stage2Minutes,
        restAfterMinutes,
        requiresPrivacy,
      },
      update: {
        timingMode,
        prepBeforeMinutes,
        durationMinutes,
        stage1Minutes,
        freeGapMinutes,
        stage2Minutes,
        restAfterMinutes,
        requiresPrivacy,
      },
    });

    revalidatePath(STAFF_SCHEDULE_PATH);
    return { success: true };
  } catch (err: unknown) {
    return {
      success: false,
      error:
        err instanceof Error
          ? err.message
          : "خطا در ذخیره زمان‌بندی سرویس پرسنل.",
    };
  }
}

export interface UpsertStaffScheduleInput {
  id?: string;
  clinicUserId: string;
  spaceId: string | null;
  jalaliDate: string;
  startTime: string;
  endTime: string;
  status: StaffScheduleStatus;
  blocksClinicBook: boolean;
  note?: string | null;
  isActive: boolean;
  allowedServiceIds: string[];
}

const VALID_SCHEDULE_STATUSES = new Set<StaffScheduleStatus>([
  StaffScheduleStatus.IN_CLINIC,
  StaffScheduleStatus.ON_CALL,
  StaffScheduleStatus.RESIDENT,
  StaffScheduleStatus.HOSPITAL_SHIFT,
  StaffScheduleStatus.DAY_OFF,
]);

export async function upsertStaffScheduleAction(
  input: UpsertStaffScheduleInput
) {
  try {
    const ctx = await getCurrentClinicContext();
    if (!ctx) {
      return { success: false, error: "نشست کاربری معتبر نیست." };
    }

    if (!canEditTargetStaff(ctx, input.clinicUserId)) {
      return {
        success: false,
        error: "شما مجوز ویرایش برنامه حضور این پرسنل را ندارید.",
      };
    }

    const parsedDate = parseJalaliDateString(input.jalaliDate);
    if (!parsedDate) {
      return {
        success: false,
        error: "تاریخ شمسی انتخاب‌شده معتبر نیست.",
      };
    }

    const startTime = normalizePersianDigits(input.startTime || "").trim();
    const endTime = normalizePersianDigits(input.endTime || "").trim();

    if (!isValidTimeFormat(startTime) || !isValidTimeFormat(endTime)) {
      return { success: false, error: "فرمت ساعت شروع یا پایان معتبر نیست." };
    }

    if (startTime >= endTime) {
      return {
        success: false,
        error: "ساعت پایان شیفت باید بعد از ساعت شروع باشد.",
      };
    }

    if (!VALID_SCHEDULE_STATUSES.has(input.status)) {
      return { success: false, error: "وضعیت حضور انتخاب‌شده معتبر نیست." };
    }

    const workingDayRow = await prisma.clinicWorkingDay.findUnique({
      where: {
        clinicId_date: {
          clinicId: ctx.clinicId,
          date: parsedDate.utcDate,
        },
      },
      select: {
        isClinicHoliday: true,
        holidayNote: true,
        shifts: true,
      },
    });

    const policy = workingDayRow ? parseWorkingDayRow(workingDayRow) : undefined;
    const jalaliLabel = `${parsedDate.jy}/${String(parsedDate.jm).padStart(
      2,
      "0"
    )}/${String(parsedDate.jd).padStart(2, "0")}`;

    const slotCheck = validateStaffSlotAgainstClinicWorkingDay(
      policy,
      jalaliLabel,
      startTime,
      endTime
    );
    if (!slotCheck.ok) {
      return { success: false, error: slotCheck.error };
    }

    const targetStaff = await prisma.clinicUser.findFirst({
      where: {
        id: input.clinicUserId,
        clinicId: ctx.clinicId,
        deletedAt: null,
      },
      select: { id: true },
    });

    if (!targetStaff) {
      return { success: false, error: "پرسنل مورد نظر یافت نشد." };
    }

    let verifiedSpaceId: string | null = null;
    if (input.spaceId && input.spaceId !== "NONE") {
      const space = await prisma.clinicSpace.findFirst({
        where: {
          id: input.spaceId,
          clinicId: ctx.clinicId,
        },
        select: { id: true },
      });
      if (!space) {
        return { success: false, error: "مکان انتخاب‌شده یافت نشد." };
      }
      verifiedSpaceId = space.id;
    }

    const uniqueServiceIds = Array.from(
      new Set((input.allowedServiceIds || []).filter(Boolean))
    );
    const validServices =
      uniqueServiceIds.length > 0
        ? await prisma.service.findMany({
            where: {
              clinicId: ctx.clinicId,
              id: { in: uniqueServiceIds },
            },
            select: { id: true },
          })
        : [];
    const verifiedServiceIds = validServices.map((s) => s.id);

    const blocksClinicBook =
      input.status === StaffScheduleStatus.HOSPITAL_SHIFT ||
      input.status === StaffScheduleStatus.RESIDENT ||
      input.status === StaffScheduleStatus.DAY_OFF
        ? true
        : Boolean(input.blocksClinicBook);

    if (input.id) {
      const existing = await prisma.clinicStaffSchedule.findFirst({
        where: {
          id: input.id,
          clinicId: ctx.clinicId,
          clinicUserId: targetStaff.id,
        },
        select: { id: true },
      });

      if (!existing) {
        return { success: false, error: "برنامه زمانی مورد نظر یافت نشد." };
      }

      await prisma.clinicStaffSchedule.update({
        where: { id: existing.id },
        data: {
          spaceId: verifiedSpaceId,
          weekday: parsedDate.weekday,
          specificDate: parsedDate.utcDate,
          startTime,
          endTime,
          status: input.status,
          blocksClinicBook,
          note: input.note?.trim() || null,
          isActive: Boolean(input.isActive),
          allowedServices: {
            set: verifiedServiceIds.map((id) => ({ id })),
          },
        },
      });
    } else {
      await prisma.clinicStaffSchedule.create({
        data: {
          clinicId: ctx.clinicId,
          clinicUserId: targetStaff.id,
          spaceId: verifiedSpaceId,
          weekday: parsedDate.weekday,
          specificDate: parsedDate.utcDate,
          startTime,
          endTime,
          status: input.status,
          blocksClinicBook,
          note: input.note?.trim() || null,
          isActive: Boolean(input.isActive),
          allowedServices: {
            connect: verifiedServiceIds.map((id) => ({ id })),
          },
        },
      });
    }

    revalidatePath(STAFF_SCHEDULE_PATH);
    return { success: true };
  } catch (err: unknown) {
    return {
      success: false,
      error:
        err instanceof Error
          ? err.message
          : "خطا در ثبت برنامه حضور پرسنل.",
    };
  }
}

export interface GenerateBulkStaffScheduleInput {
  clinicUserId: string;
  fromJalaliDate: string;
  toJalaliDate: string;
  weekdays: number[];
  spaceId: string | null;
  startTime: string;
  endTime: string;
  status: StaffScheduleStatus;
  blocksClinicBook: boolean;
  note?: string | null;
  allowedServiceIds: string[];
  replaceOnTargetDays: boolean;
}

export async function generateBulkStaffSchedulesAction(
  input: GenerateBulkStaffScheduleInput
) {
  try {
    const ctx = await getCurrentClinicContext();
    if (!ctx) {
      return { success: false, error: "نشست کاربری معتبر نیست." };
    }

    if (!canEditTargetStaff(ctx, input.clinicUserId)) {
      return {
        success: false,
        error: "شما مجوز ایجاد برنامه بازه‌ای برای این پرسنل را ندارید.",
      };
    }

    const fromParsed = parseJalaliDateString(input.fromJalaliDate);
    const toParsed = parseJalaliDateString(input.toJalaliDate);

    if (!fromParsed || !toParsed) {
      return {
        success: false,
        error: "تاریخ شروع یا پایان بازه شمسی معتبر نیست.",
      };
    }

    if (fromParsed.utcDate.getTime() > toParsed.utcDate.getTime()) {
      return {
        success: false,
        error: "تاریخ پایان بازه باید مساوی یا بعد از تاریخ شروع باشد.",
      };
    }

    const startTime = normalizePersianDigits(input.startTime || "").trim();
    const endTime = normalizePersianDigits(input.endTime || "").trim();
    if (!isValidTimeFormat(startTime) || !isValidTimeFormat(endTime)) {
      return { success: false, error: "فرمت ساعت شروع یا پایان معتبر نیست." };
    }
    if (startTime >= endTime) {
      return {
        success: false,
        error: "ساعت پایان شیفت باید بعد از ساعت شروع باشد.",
      };
    }

    if (!VALID_SCHEDULE_STATUSES.has(input.status)) {
      return { success: false, error: "وضعیت حضور انتخاب‌شده معتبر نیست." };
    }

    const targetWeekdays = new Set(
      (input.weekdays || [])
        .map((d) => Number(d))
        .filter((d) => Number.isInteger(d) && d >= 0 && d <= 6)
    );

    if (targetWeekdays.size === 0) {
      return {
        success: false,
        error: "حداقل یک روز از هفته را انتخاب کنید.",
      };
    }

    const targetStaff = await prisma.clinicUser.findFirst({
      where: {
        id: input.clinicUserId,
        clinicId: ctx.clinicId,
        deletedAt: null,
      },
      select: { id: true },
    });

    if (!targetStaff) {
      return { success: false, error: "پرسنل مورد نظر یافت نشد." };
    }

    let verifiedSpaceId: string | null = null;
    if (input.spaceId && input.spaceId !== "NONE") {
      const space = await prisma.clinicSpace.findFirst({
        where: { id: input.spaceId, clinicId: ctx.clinicId },
        select: { id: true },
      });
      if (!space) {
        return { success: false, error: "مکان انتخاب‌شده یافت نشد." };
      }
      verifiedSpaceId = space.id;
    }

    const uniqueServiceIds = Array.from(
      new Set((input.allowedServiceIds || []).filter(Boolean))
    );
    const validServices =
      uniqueServiceIds.length > 0
        ? await prisma.service.findMany({
            where: { clinicId: ctx.clinicId, id: { in: uniqueServiceIds } },
            select: { id: true },
          })
        : [];
    const verifiedServiceIds = validServices.map((s) => s.id);

    const blocksClinicBook =
      input.status === StaffScheduleStatus.HOSPITAL_SHIFT ||
      input.status === StaffScheduleStatus.RESIDENT ||
      input.status === StaffScheduleStatus.DAY_OFF
        ? true
        : Boolean(input.blocksClinicBook);

    const candidateDates: Array<{
      date: Date;
      weekday: number;
      jalaliLabel: string;
    }> = [];
    let curJy = fromParsed.jy;
    let curJm = fromParsed.jm;
    let curJd = fromParsed.jd;

    for (let guard = 0; guard < 370; guard++) {
      const curUtc = jalaliToGregorianUtc(curJy, curJm, curJd);
      if (curUtc.getTime() > toParsed.utcDate.getTime()) break;

      const weekday = (curUtc.getUTCDay() + 1) % 7;
      if (targetWeekdays.has(weekday)) {
        candidateDates.push({
          date: curUtc,
          weekday,
          jalaliLabel: `${curJy}/${String(curJm).padStart(2, "0")}/${String(
            curJd
          ).padStart(2, "0")}`,
        });
      }

      const daysInCurMonth = getJalaliMonthDaysCount(curJy, curJm);
      if (curJd < daysInCurMonth) {
        curJd += 1;
      } else {
        curJd = 1;
        if (curJm < 12) {
          curJm += 1;
        } else {
          curJm = 1;
          curJy += 1;
        }
      }
    }

    if (candidateDates.length === 0) {
      return {
        success: false,
        error: "هیچ روزی در بازه انتخاب‌شده با روزهای هفته مشخص‌شده منطبق نبود.",
      };
    }

    const workingDaysRows = await prisma.clinicWorkingDay.findMany({
      where: {
        clinicId: ctx.clinicId,
        date: {
          gte: fromParsed.utcDate,
          lte: toParsed.utcDate,
        },
      },
      select: {
        date: true,
        isClinicHoliday: true,
        holidayNote: true,
        shifts: true,
      },

function gregorianUtcToJalaliKey(date: Date): string {
  const fn = (
    jalaliLib as {
      gregorianToJalali?: (d: Date) => { year: number; month: number; day: number };
    }
  ).gregorianToJalali;

  if (typeof fn === "function") {
    const j = fn(date);
    if (j?.year && j?.month && j?.day) {
      return normalizePersianDigits(
        `${j.year}/${String(j.month).padStart(2, "0")}/${String(j.day).padStart(
          2,
          "0"
        )}`
      );
    }
  }

  const gy = date.getUTCFullYear();
  const gm = date.getUTCMonth() + 1;
  const gd = date.getUTCDate();

  const gMonthDays = [0, 31, 59, 90, 120, 151, 181, 212, 243, 273, 304, 334];
  const gy2 = gm > 2 ? gy + 1 : gy;
  let days =
    355666 +
    365 * gy +
    Math.floor((gy2 + 3) / 4) -
    Math.floor((gy2 + 99) / 100) +
    Math.floor((gy2 + 399) / 400) +
    gd +
    gMonthDays[gm - 1];

  let jy = -1595 + 33 * Math.floor(days / 12053);
  days %= 12053;

  jy += 4 * Math.floor(days / 1461);
  days %= 1461;

  if (days > 365) {
    jy += Math.floor((days - 1) / 365);
    days = (days - 1) % 365;
  }

  const jm =
    days < 186 ? 1 + Math.floor(days / 31) : 7 + Math.floor((days - 186) / 30);
  const jd = 1 + (days < 186 ? days % 31 : (days - 186) % 30);

  return `${jy}/${String(jm).padStart(2, "0")}/${String(jd).padStart(2, "0")}`;
}

function jalaliToGregorianUtc(jy: number, jm: number, jd: number): Date {
  const fn = (
    jalaliLib as {
      jalaliToGregorian?: (y: number, m: number, d: number) => Date
    }
  ).jalaliToGregorian
  if (typeof fn === "function") {
    const res = fn(jy, jm, jd)
    if (res instanceof Date && !Number.isNaN(res.getTime())) {
      return new Date(
        Date.UTC(res.getFullYear(), res.getMonth(), res.getDate(), 0, 0, 0, 0)
      )
    }
  }

  const jyAdj = jy - 979
  const jmAdj = jm - 1
  const jdAdj = jd - 1

  let jDayNo =
    365 * jyAdj +
    Math.floor(jyAdj / 33) * 8 +
    Math.floor(((jyAdj % 33) + 3) / 4)
  const monthDays = [31, 31, 31, 31, 31, 31, 30, 30, 30, 30, 30, 29]
  for (let i = 0; i < jmAdj; ++i) {
    jDayNo += monthDays[i]
  }
  jDayNo += jdAdj

  let gDayNo = jDayNo + 79

  let gy = 1600 + 400 * Math.floor(gDayNo / 146097)
  gDayNo = gDayNo % 146097

  let leap = true
  if (gDayNo >= 36525) {
    gDayNo--
    gy += 100 * Math.floor(gDayNo / 36524)
    gDayNo = gDayNo % 36524

    if (gDayNo >= 365) {
      gDayNo++
    } else {
      leap = false
    }
  }

  gy += 4 * Math.floor(gDayNo / 1461)
  gDayNo %= 1461

  if (gDayNo >= 366) {
    leap = false
    gDayNo -= 366
    gy += Math.floor(gDayNo / 365)
    gDayNo = gDayNo % 365
  }

  const gMonthDays = [
    31,
    leap ? 29 : 28,
    31,
    30,
    31,
    30,
    31,
    31,
    30,
    31,
    30,
    31,
  ]
  let gm = 0
  for (; gm < 12 && gDayNo >= gMonthDays[gm]; gm++) {
    gDayNo -= gMonthDays[gm]
  }
  const gd = gDayNo + 1

  return new Date(Date.UTC(gy, gm, gd, 0, 0, 0, 0))
}

function getCurrentJalaliYearMonth(): { year: number; month: number } {
  const fn = (
    jalaliLib as {
      getCurrentJalaliDate?: () => { year: number; month: number }
    }
  ).getCurrentJalaliDate
  if (typeof fn === "function") {
    const cur = fn()
    if (cur?.year && cur?.month) {
      return { year: cur.year, month: cur.month }
    }
  }

  try {
    const parts = new Intl.DateTimeFormat("fa-IR-u-ca-persian", {
      year: "numeric",
      month: "numeric",
      timeZone: "Asia/Tehran",
    }).formatToParts(new Date())

    const yearStr = parts.find((p) => p.type === "year")?.value || "1405"
    const monthStr = parts.find((p) => p.type === "month")?.value || "9"
    const normalized = normalizePersianDigits(`${yearStr}/${monthStr}`)
    const [y, m] = normalized.split("/").map(Number)
    if (y >= 1350 && y <= 1480 && m >= 1 && m <= 12) {
      return { year: y, month: m }
    }
  } catch {
    // fallback below
  }
  return { year: 1405, month: 9 }
}

function formatJalaliKey(jy: number, jm: number, jd: number): string {
  const fn = (
    jalaliLib as {
      formatJalaliDate?: (y: number, m: number, d: number) => string
    }
  ).formatJalaliDate
  if (typeof fn === "function") {
    return normalizePersianDigits(fn(jy, jm, jd))
  }
  return `${jy}/${String(jm).padStart(2, "0")}/${String(jd).padStart(2, "0")}`
}

function SplitTimeSelect({
  value,
  onChange,
}: {
  value: string
  onChange: (next: string) => void
}) {
  const [hh, mm] = (value || "09:00").split(":")
  return (
    <div className="flex items-center gap-1.5" dir="ltr">
      <Select
        value={hh || "09"}
        onValueChange={(newH) => onChange(`${newH || "09"}:${mm || "00"}`)}
      >
        <SelectTrigger className="w-20 bg-background">
          <SelectValue>{toFa(hh || "09")}</SelectValue>
        </SelectTrigger>
        <SelectContent>
          {HOUR_OPTIONS.map((h) => (
            <SelectItem key={h} value={h}>
              {toFa(h)}
            </SelectItem>
          ))}
        </SelectContent>
      </Select>
      <span className="font-bold text-muted-foreground">:</span>
      <Select
        value={mm || "00"}
        onValueChange={(newM) => onChange(`${hh || "09"}:${newM || "00"}`)}
      >
        <SelectTrigger className="w-20 bg-background">
          <SelectValue>{toFa(mm || "00")}</SelectValue>
        </SelectTrigger>
        <SelectContent>
          {MINUTE_OPTIONS.map((m) => (
            <SelectItem key={m} value={m}>
              {toFa(m)}
            </SelectItem>
          ))}
        </SelectContent>
      </Select>
    </div>
  )
}

                  type="button"
                  variant="outline"
                  size="icon-xs"
                  onClick={handlePrevMonth}
                  title="ماه قبل"
                >
                  <ChevronRight className="size-4" />
                </Button>

                <div className="flex items-center gap-2">
                  <Select
                    value={String(selectedMonth)}
                    onValueChange={(v) => setSelectedMonth(Number(v ?? 1))}
                  >
                    <SelectTrigger className="w-32 bg-background">
                      <SelectValue>
                        {JALALI_MONTHS[selectedMonth - 1]}
                      </SelectValue>
                    </SelectTrigger>
                    <SelectContent>
                      {JALALI_MONTHS.map((mName, idx) => (
                        <SelectItem key={mName} value={String(idx + 1)}>
                          {mName}
                        </SelectItem>
                      ))}
                    </SelectContent>
                  </Select>

                  <Select
                    value={String(selectedYear)}
                    onValueChange={(v) =>
                      setSelectedYear(Number(v ?? initialYm.year))
                    }
                  >
                    <SelectTrigger className="w-24 bg-background">
                      <SelectValue>{toFa(selectedYear)}</SelectValue>
                    </SelectTrigger>
                    <SelectContent>
                      {[
                        initialYm.year - 1,
                        initialYm.year,
                        initialYm.year + 1,
                        initialYm.year + 2,
                      ].map((yr) => (
                        <SelectItem key={yr} value={String(yr)}>
                          {toFa(yr)}
                        </SelectItem>
                      ))}
                    </SelectContent>
                  </Select>
                </div>

                <Button
                  type="button"
                  variant="outline"
                  size="icon-xs"
                  onClick={handleNextMonth}
                  title="ماه بعد"
                >
                  <ChevronLeft className="size-4" />
                </Button>
              </div>

              <Button
                type="button"
                variant="ghost"
                size="xs"
                onClick={handleGoCurrentMonth}
              >
                برو به ماه جاری
              </Button>
            </CardHeader>

            <CardContent className="p-3 sm:p-4">
              <div className="mb-2 hidden grid-cols-7 gap-2 text-center text-xs font-bold text-muted-foreground md:grid">
                {WEEKDAYS.map((w) => (
                  <div
                    key={w.index}
                    className={cn(
                      "rounded-lg bg-muted/50 py-2",
                      w.index === 6 && "text-destructive"
                    )}
                  >
                    {w.label}
                  </div>
                ))}
              </div>

              <div className="grid grid-cols-1 gap-2.5 sm:grid-cols-2 md:grid-cols-7">
                {calendarCells.map((cell, idx) => {
                  if (!cell.day || !cell.key) {
                    return (
                      <div
                        key={`empty-${idx}`}
                        className="hidden min-h-32 rounded-xl border border-dashed bg-muted/10 md:block"
                      />
                    )
                  }

                  const isFriday = cell.weekday === 6
                  const isOfficialHoliday = Boolean(cell.holidayTitle)
                  const dayLabel =
                    WEEKDAYS.find((w) => w.index === cell.weekday)?.label || ""

                  return (
                    <div
                      key={cell.key}
                      className={cn(
                        "flex min-h-32 flex-col justify-between rounded-xl border bg-background p-2.5 transition-colors",
                        cell.isDayBlockedForStaff
                          ? "border-destructive/30 bg-destructive/5"
                          : "hover:border-primary/40"
                      )}
                    >
                      <div>
                        <div className="mb-1.5 flex items-center justify-between gap-1">
                          <div className="flex items-center gap-1.5">
                            <span
                              className={cn(
                                "inline-flex size-6 items-center justify-center rounded-full text-xs font-bold",
                                isFriday ||
                                  isOfficialHoliday ||
                                  cell.isDayBlockedForStaff
                                  ? "bg-destructive/15 text-destructive"
                                  : "bg-primary/10 text-primary"
                              )}
                            >
                              {toFa(cell.day)}
                            </span>
                            <span className="text-[11px] font-medium text-muted-foreground md:hidden">
                              {dayLabel}
                            </span>
                          </div>

                          {!cell.isDayBlockedForStaff ? (
                            <Button
                              type="button"
                              variant="ghost"
                              size="icon-xs"
                              onClick={() =>
                                openCreateScheduleForDate(cell.key!)
                              }
                              title={`ثبت شیفت برای ${toFa(cell.key)}`}
                            >
                              <Plus className="size-3.5" />
                            </Button>
                          ) : (
                            <span
                              className="inline-flex items-center text-destructive/80"
                              title="تعطیل یا فاقد ساعت کاری ثبت‌شده در کلینیک"
                            >
                              <Lock className="size-3.5" />
                            </span>
                          )}
                        </div>

                        {cell.holidayTitle && (
                          <div className="mb-1 rounded-md bg-destructive/10 px-1.5 py-0.5 text-[10px] font-medium text-destructive">
                            {cell.holidayTitle}
                          </div>
                        )}

                        {!cell.clinicDay ? (
                          <div className="mb-1.5 rounded-md bg-muted px-1.5 py-1 text-[10px] text-muted-foreground">
                            ساعت کاری کلینیک ثبت نشده
                          </div>
                        ) : cell.clinicDay.isClinicHoliday ? (
                          <div className="mb-1.5 rounded-md bg-destructive/10 px-1.5 py-1 text-[10px] font-medium text-destructive">
                            تعطیل کلینیک{" "}
                            {cell.clinicDay.holidayNote
                              ? `(${cell.clinicDay.holidayNote})`
                              : ""}
                          </div>
                        ) : cell.clinicDay.shifts.length > 0 ? (
                          <div className="mb-1.5 space-y-0.5 rounded-md bg-emerald-500/10 px-1.5 py-1 text-[10px] text-emerald-800 dark:text-emerald-300">
                            <div className="font-semibold">ساعت کلینیک:</div>
                            {cell.clinicDay.shifts.map((sh, sIdx) => (
                              <div key={sIdx} dir="ltr" className="font-mono">
                                {toFa(sh.startTime)} - {toFa(sh.endTime)}
                              </div>
                            ))}
                          </div>
                        ) : null}

                        <div className="space-y-1.5">
                          {cell.slots.map((slot) => {
                            const meta = STATUS_META[slot.status]
                            return (
                              <div
                                key={slot.id}
                                className="space-y-1 rounded-lg border bg-muted/35 p-1.5 text-[11px]"
                              >
                                <div className="flex items-center justify-between gap-1">
                                  <Badge
                                    variant={meta.badgeVariant}
                                    className="h-4 px-1.5 text-[10px]"
                                  >
                                    {meta.shortLabel}
                                  </Badge>
                                  <div className="flex items-center gap-0.5">
                                    <button
                                      type="button"
                                      disabled={isSaving}
                                      onClick={() => openEditSchedule(slot)}
                                      className="rounded p-0.5 hover:bg-background"
                                      title="ویرایش"
                                    >
                                      <Edit3 className="size-3" />
                                    </button>
                                    <button
                                      type="button"
                                      disabled={isSaving}
                                      onClick={() =>
                                        void handleDeleteSchedule(slot.id)
                                      }
                                      className="rounded p-0.5 text-destructive hover:bg-destructive/10"
                                      title="حذف"
                                    >
                                      <Trash2 className="size-3" />
                                    </button>
                                  </div>
                                </div>

                                <div
                                  dir="ltr"
                                  className="font-mono text-[10px] font-semibold"
                                >
                                  {toFa(slot.startTime)} - {toFa(slot.endTime)}
                                </div>

                                {slot.space && (
                                  <div className="flex items-center gap-1 truncate text-[10px] text-muted-foreground">
                                    {slot.space.spaceType ===
                                    ClinicSpaceType.HOSPITAL ? (
                                      <Hospital className="size-3 shrink-0 text-emerald-600" />
                                    ) : (
                                      <DoorOpen className="size-3 shrink-0 text-primary" />
                                    )}
                                    <span className="truncate">
                                      {slot.space.name}
                                    </span>
                                  </div>
                                )}

                                {slot.blocksClinicBook && (
                                  <div className="flex items-center gap-0.5 text-[10px] font-medium text-destructive">
                                    <AlertTriangle className="size-2.5 shrink-0" />
                                    <span>نوبت مطب بسته</span>
                                  </div>
                                )}
                              </div>
                            )
                          })}
                        </div>
                      </div>
                    </div>
                  )
                })}
              </div>
            </CardContent>
          </Card>
        </TabsContent>

        <TabsContent value="services" className="space-y-4">
          {activeStaff.services.length === 0 ? (
            <Card>
              <CardContent className="p-6 text-center text-sm text-muted-foreground">
                هنوز هیچ سرویسی به این پرسنل اختصاص داده نشده است. ابتدا از بخش «مدیریت پرسنل» خدمات مجاز این فرد را مشخص کنید.
              </CardContent>
            </Card>
          ) : (
            <div className="grid grid-cols-1 gap-4 md:grid-cols-2 lg:grid-cols-3">
              {activeStaff.services.map((srv) => {
                const isSplit =
                  srv.timingMode === ServiceTimingMode.SPLIT_STAGE
                const totalCycleMinutes = isSplit
                  ? srv.prepBeforeMinutes +
                    srv.stage1Minutes +
                    srv.freeGapMinutes +
                    srv.stage2Minutes +
                    srv.restAfterMinutes
                  : srv.prepBeforeMinutes +
                    srv.durationMinutes +
                    srv.restAfterMinutes

                return (
                  <Card key={srv.serviceId} className="flex flex-col">
                    <CardHeader className="p-4 pb-2">
                      <div className="flex items-start justify-between gap-2">
                        <div className="space-y-1">
                          <CardTitle className="text-base font-bold">
                            {srv.serviceTitle}
                          </CardTitle>
                          <div className="flex flex-wrap gap-1.5">
                            <Badge
                              variant={isSplit ? "default" : "secondary"}
                              className="gap-1 text-[11px]"
                            >
                              {isSplit ? (
                                <>
                                  <Split className="size-3" />
                                  چندمرحله‌ای (دارای تایم آزاد میانی)
                                </>
                              ) : (
                                <>
                                  <Clock className="size-3" />
                                  تک‌مرحله‌ای استاندارد
                                </>
                              )}
                            </Badge>
                            {srv.isCustomized && (
                              <Badge variant="outline" className="text-[11px]">
                                شخصی‌سازی‌شده
                              </Badge>
                            )}
                          </div>
                        </div>

                        <Button
                          type="button"
                          variant="outline"
                          size="xs"
                          onClick={() => openEditServiceTiming(srv)}
                        >
                          <Edit3 className="ml-1 size-3.5" />
                          تنظیم زمان
                        </Button>
                      </div>
                    </CardHeader>

                    <CardContent className="flex-1 space-y-2.5 p-4 pt-2 text-xs">
                      <div className="space-y-1.5 rounded-xl border bg-muted/20 p-3">
                        <div className="flex items-center justify-between">
                          <span className="text-muted-foreground">
                            حضور مراجع قبل از شروع:
                          </span>
                          <span className="font-semibold">
                            {toFa(srv.prepBeforeMinutes)} دقیقه
                          </span>
                        </div>

                        {isSplit ? (
                          <>
                            <div className="flex items-center justify-between">
                              <span className="text-muted-foreground">
                                مرحله ۱ (مثلاً خون‌گیری):
                              </span>
                              <span className="font-semibold">
                                {toFa(srv.stage1Minutes)} دقیقه
                              </span>
                            </div>
                            <div className="flex items-center justify-between text-emerald-700 dark:text-emerald-300">
                              <span className="flex items-center gap-1 font-medium">
                                <Sparkles className="size-3" />
                                تایم آزاد میانی (قابل رزرو برای دیگران):
                              </span>
                              <span className="font-bold">
                                {toFa(srv.freeGapMinutes)} دقیقه
                              </span>
                            </div>
                            <div className="flex items-center justify-between">
                              <span className="text-muted-foreground">
                                مرحله ۲ (مثلاً تزریق نهایی):
                              </span>
                              <span className="font-semibold">
                                {toFa(srv.stage2Minutes)} دقیقه
                              </span>
                            </div>
                          </>
                        ) : (
                          <div className="flex items-center justify-between">
                            <span className="text-muted-foreground">
                              مدت انجام سرویس:
                            </span>
                            <span className="font-semibold">
                              {toFa(srv.durationMinutes)} دقیقه
                            </span>
                          </div>
                        )}

                        <div className="flex items-center justify-between border-t pt-1.5">
                          <span className="flex items-center gap-1 text-muted-foreground">
                            <Coffee className="size-3" />
                            زمان استراحت / پاکسازی:
                          </span>
                          <span className="font-semibold">
                            {toFa(srv.restAfterMinutes)} دقیقه
                          </span>
                        </div>
                      </div>

                      <div className="flex items-center justify-between text-[11px] text-muted-foreground">
                        <span>کل بازه چرخه نوبت:</span>
                        <strong className="text-foreground">
                          {toFa(totalCycleMinutes)} دقیقه
                        </strong>
                      </div>

                      {srv.requiresPrivacy && (
                        <div className="flex items-center gap-1.5 rounded-lg bg-amber-500/10 px-2.5 py-1.5 text-[11px] font-medium text-amber-800 dark:text-amber-300">
                          <ShieldCheck className="size-3.5 shrink-0" />
                          نیازمند فضای خالی / رعایت کامل پرایوسی مراجع
                        </div>
                      )}
                    </CardContent>
                  </Card>
                )
              })}
            </div>
          )}
        </TabsContent>
      </Tabs>

      {/* مودال تنظیم زمان‌بندی سرویس */}
      <ResponsiveDialog
        open={timingDialogOpen}
        onOpenChange={(open) => !isSaving && setTimingDialogOpen(open)}
        drawerPosition="bottom"
      >
        <ResponsiveDialogTrigger
          className="sr-only"
          tabIndex={-1}
          aria-hidden="true"
        />
        <ResponsiveDialogContent className="sm:max-w-lg">
          <ResponsiveDialogHeader>
            <ResponsiveDialogTitle>
              تنظیم زمان‌بندی «{editingService?.serviceTitle}»
            </ResponsiveDialogTitle>
            <ResponsiveDialogDescription>
              مدت انجام سرویس، زمان حضور مراجع قبل از شروع، استراحت و در صورت نیاز تایم آزاد میانی (مانند PRP) را مشخص کنید.
            </ResponsiveDialogDescription>
          </ResponsiveDialogHeader>

          <form onSubmit={handleSaveServiceTiming} className="contents">
            <ResponsiveDialogPanel className="no-scrollbar max-h-[70vh] overflow-y-auto pt-2">
              <div className="px-5">
                <FieldGroup>
                  <Field>
                    <FieldLabel>مدل زمان‌بندی سرویس</FieldLabel>
                    <Select
                      value={timingMode}
                      onValueChange={(val) =>
                        setTimingMode(
                          (val as ServiceTimingMode) ||
                            ServiceTimingMode.SINGLE_STAGE
                        )
                      }
                    >
                      <SelectTrigger className="bg-background">
                        <SelectValue>
                          {timingMode === ServiceTimingMode.SPLIT_STAGE
                            ? "چندمرحله‌ای با تایم آزاد میانی (مثل PRP)"
                            : "تک‌مرحله‌ای استاندارد"}
                        </SelectValue>
                      </SelectTrigger>
                      <SelectContent>
                        <SelectItem value={ServiceTimingMode.SINGLE_STAGE}>
                          تک‌مرحله‌ای استاندارد
                        </SelectItem>
                        <SelectItem value={ServiceTimingMode.SPLIT_STAGE}>
                          چندمرحله‌ای با تایم آزاد میانی (مثل PRP)
                        </SelectItem>
                      </SelectContent>
                    </Select>
                  </Field>

                  <Field>
                    <FieldLabel htmlFor="prepBefore">
                      زمان حضور مراجع قبل از شروع (دقیقه)
                    </FieldLabel>
                    <Input
                      id="prepBefore"
                      type="number"
                      min={0}
                      max={240}
                      dir="ltr"
                      value={prepBeforeMinutes}
                      onChange={(e) => setPrepBeforeMinutes(e.target.value)}
                    />
                  </Field>

                  {timingMode === ServiceTimingMode.SINGLE_STAGE ? (
                    <Field>
                      <FieldLabel htmlFor="durationMin">
                        مدت انجام سرویس (دقیقه)
                      </FieldLabel>
                      <Input
                        id="durationMin"
                        type="number"
                        min={5}
                        max={480}
                        dir="ltr"
                        value={durationMinutes}
                        onChange={(e) => setDurationMinutes(e.target.value)}
                      />
                    </Field>
                  ) : (
                    <div className="space-y-3 rounded-xl border border-primary/30 bg-primary/5 p-3.5">
                      <p className="text-xs font-bold text-primary">
                        تنظیمات مراحل سرویس و تایم آزاد میانی
                      </p>

                      <div className="grid grid-cols-1 gap-3 sm:grid-cols-3">
                        <Field>
                          <FieldLabel htmlFor="stage1Min">
                            مرحله ۱ (دقیقه)
                          </FieldLabel>
                          <Input
                            id="stage1Min"
                            type="number"
                            min={5}
                            max={240}
                            dir="ltr"
                            value={stage1Minutes}
                            onChange={(e) => setStage1Minutes(e.target.value)}
                          />
                          <FieldDescription className="text-[10px]">
                            مثلاً خون‌گیری
                          </FieldDescription>
                        </Field>

                        <Field>
                          <FieldLabel htmlFor="freeGapMin">
                            تایم آزاد میانی (دقیقه)
                          </FieldLabel>
                          <Input
                            id="freeGapMin"
                            type="number"
                            min={0}
                            max={360}
                            dir="ltr"
                            value={freeGapMinutes}
                            onChange={(e) => setFreeGapMinutes(e.target.value)}
                          />
                          <FieldDescription className="text-[10px]">
                            قابل رزرو برای مراجع دیگر
                          </FieldDescription>
                        </Field>

                        <Field>
                          <FieldLabel htmlFor="stage2Min">
                            مرحله ۲ (دقیقه)
                          </FieldLabel>
                          <Input
                            id="stage2Min"
                            type="number"
                            min={5}
                            max={240}
                            dir="ltr"
                            value={stage2Minutes}
                            onChange={(e) => setStage2Minutes(e.target.value)}
                          />
                          <FieldDescription className="text-[10px]">
                            مثلاً تزریق نهایی
                          </FieldDescription>
                        </Field>
                      </div>
                    </div>
                  )}

                  <Field>
                    <FieldLabel htmlFor="restAfter">
                      زمان استراحت / پاکسازی بعد از سرویس (دقیقه)
                    </FieldLabel>
                    <Input
                      id="restAfter"
                      type="number"
                      min={0}
                      max={180}
                      dir="ltr"
                      value={restAfterMinutes}
                      onChange={(e) => setRestAfterMinutes(e.target.value)}
                    />
                  </Field>

                  <Field orientation="horizontal">
                    <Checkbox
                      id="srvPrivacy"
                      checked={serviceRequiresPrivacy}
                      onCheckedChange={(v) =>
                        setServiceRequiresPrivacy(Boolean(v))
                      }
                    />
                    <div className="space-y-0.5">
                      <FieldLabel
                        htmlFor="srvPrivacy"
                        className="cursor-pointer font-semibold"
                      >
                        این سرویس نیازمند اتاق خالی / رعایت پرایوسی است
                      </FieldLabel>
                    </div>
                  </Field>
                </FieldGroup>
              </div>
            </ResponsiveDialogPanel>

            <ResponsiveDialogFooter className="gap-2">
              <ResponsiveDialogDesktopOnly>
                <Button
                  type="button"
                  variant="outline"
                  disabled={isSaving}
                  onClick={() => setTimingDialogOpen(false)}
                >
                  <Ban className="ml-1.5 size-4" />
                  انصراف
                </Button>
              </ResponsiveDialogDesktopOnly>
              <Button
                type="submit"
                disabled={isSaving}
                className="w-full sm:w-auto"
              >
                <Save className="ml-1.5 size-4" />
                {isSaving ? "در حال ذخیره..." : "ذخیره زمان‌بندی"}
              </Button>
            </ResponsiveDialogFooter>
          </form>
        </ResponsiveDialogContent>
      </ResponsiveDialog>

      {/* مودال ثبت / ویرایش شیفت در یک روز شمسی مشخص */}
      <ResponsiveDialog
        open={scheduleDialogOpen}
        onOpenChange={(open) => !isSaving && setScheduleDialogOpen(open)}
        drawerPosition="bottom"
      >
        <ResponsiveDialogTrigger
          className="sr-only"
          tabIndex={-1}
          aria-hidden="true"
        />
        <ResponsiveDialogContent className="sm:max-w-xl">
          <ResponsiveDialogHeader>
            <ResponsiveDialogTitle>
              {editingScheduleId
                ? "ویرایش برنامه حضور روزانه"
                : "ثبت برنامه حضور در روز شمسی"}
            </ResponsiveDialogTitle>
            <ResponsiveDialogDescription>
              ساعات حضور پرسنل باید داخل بازه‌های کاریِ تعریف‌شده کلینیک در این تاریخ باشد.
            </ResponsiveDialogDescription>
          </ResponsiveDialogHeader>

          <form onSubmit={handleSaveSchedule} className="contents">
            <ResponsiveDialogPanel className="no-scrollbar max-h-[70vh] overflow-y-auto pt-2">
              <div className="px-5">
                <FieldGroup>
                  {selectedDayPolicy &&
                    !selectedDayPolicy.isClinicHoliday &&
                    selectedDayPolicy.shifts.length > 0 && (
                      <div className="rounded-xl border border-emerald-500/30 bg-emerald-500/10 p-3 text-xs text-emerald-900 dark:text-emerald-200">
                        <span className="font-bold">
                          بازه‌های کاری مجاز کلینیک در {toFa(targetJalaliDate)}:
                        </span>{" "}
                        <span dir="ltr" className="font-mono font-semibold">
                          {toFa(
                            selectedDayPolicy.shifts
                              .map((s) => `${s.startTime} - ${s.endTime}`)
                              .join(" | ")
                          )}
                        </span>
                      </div>
                    )}

                  <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
                    <Field>
                      <FieldLabel htmlFor="jDateInput">
                        تاریخ شمسی (YYYY/MM/DD)
                      </FieldLabel>
                      <Input
                        id="jDateInput"
                        dir="ltr"
                        required
                        value={targetJalaliDate}
                        onChange={(e) => setTargetJalaliDate(e.target.value)}
                        placeholder="1405/09/05"
                      />
                    </Field>

                    <Field>
                      <FieldLabel>مکان (اتاق / یونیت / بیمارستان)</FieldLabel>
                      <Select
                        value={spaceId}
                        onValueChange={(val) => setSpaceId(val || "NONE")}
                      >
                        <SelectTrigger className="bg-background">
                          <SelectValue>
                            {spaceId === "NONE"
                              ? "بدون اتصال به مکان خاص"
                              : spaces.find((s) => s.id === spaceId)?.name ||
                                "انتخاب مکان"}
                          </SelectValue>
                        </SelectTrigger>
                        <SelectContent>
                          <SelectItem value="NONE">
                            بدون اتصال به مکان خاص
                          </SelectItem>
                          {spaces.map((sp) => (
                            <SelectItem key={sp.id} value={sp.id}>
                              {sp.name}
                            </SelectItem>
                          ))}
                        </SelectContent>
                      </Select>
                    </Field>
                  </div>

                  <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
                    <Field>
                      <FieldLabel>ساعت شروع</FieldLabel>
                      <SplitTimeSelect
                        value={startTime}
                        onChange={setStartTime}
                      />
                    </Field>

                    <Field>
                      <FieldLabel>ساعت پایان</FieldLabel>
                      <SplitTimeSelect value={endTime} onChange={setEndTime} />
                    </Field>
                  </div>

                  <Field>
                    <FieldLabel>وضعیت حضور پرسنل</FieldLabel>
                    <Select
                      value={scheduleStatus}
                      onValueChange={(val) => {
                        const next =
                          (val as StaffScheduleStatus) ||
                          StaffScheduleStatus.IN_CLINIC
                        setScheduleStatus(next)
                        if (
                          next === StaffScheduleStatus.HOSPITAL_SHIFT ||
                          next === StaffScheduleStatus.RESIDENT ||
                          next === StaffScheduleStatus.DAY_OFF
                        ) {
                          setBlocksClinicBook(true)
                        } else if (next === StaffScheduleStatus.IN_CLINIC) {
                          setBlocksClinicBook(false)
                        }
                      }}
                    >
                      <SelectTrigger className="bg-background">
                        <SelectValue>
                          {STATUS_META[scheduleStatus].label}
                        </SelectValue>
                      </SelectTrigger>
                      <SelectContent>
                        {(
                          Object.keys(STATUS_META) as StaffScheduleStatus[]
                        ).map((st) => (
                          <SelectItem key={st} value={st}>
                            {STATUS_META[st].label}
                          </SelectItem>
                        ))}
                      </SelectContent>
                    </Select>
                    <FieldDescription className="text-[11px]">
                      {STATUS_META[scheduleStatus].description}
                    </FieldDescription>
                  </Field>

                  <Field orientation="horizontal">
                    <Checkbox
                      id="blocksClinic"
                      checked={blocksClinicBook}
                      onCheckedChange={(v) => setBlocksClinicBook(Boolean(v))}
                    />
                    <div className="space-y-0.5">
                      <FieldLabel
                        htmlFor="blocksClinic"
                        className="cursor-pointer font-semibold"
                      >
                        بستن نوبت‌دهی مطب در این بازه زمانی
                      </FieldLabel>
                    </div>
                  </Field>

                  <Field>
                    <FieldLabel>
                      سرویس‌های قابل انجام در این شیفت و مکان (
                      {toFa(scheduleServiceIds.length)} انتخاب‌شده)
                    </FieldLabel>
                    <div className="mt-1.5 grid max-h-40 grid-cols-1 gap-2 overflow-y-auto rounded-xl border p-2.5 sm:grid-cols-2">
                      {activeStaff.services.map((srv) => {
                        const checked = scheduleServiceIds.includes(
                          srv.serviceId
                        )
                        return (
                          <label
                            key={srv.serviceId}
                            className={cn(
                              "flex cursor-pointer items-center gap-2 rounded-lg border px-2.5 py-2 text-xs transition-colors",
                              checked
                                ? "border-primary bg-primary/10 font-medium text-primary"
                                : "bg-background hover:bg-muted/50"
                            )}
                          >
                            <Checkbox
                              checked={checked}
                              onCheckedChange={() =>
                                toggleScheduleService(srv.serviceId)
                              }
                            />
                            <span className="truncate">{srv.serviceTitle}</span>
                          </label>
                        )
                      })}
                    </div>
                  </Field>

                  <Field>
                    <FieldLabel htmlFor="schNote">
                      توضیحات شیفت / آنکال (اختیاری)
                    </FieldLabel>
                    <Textarea
                      id="schNote"
                      rows={2}
                      value={scheduleNote}
                      onChange={(e) => setScheduleNote(e.target.value)}
                    />
                  </Field>
                </FieldGroup>
              </div>
            </ResponsiveDialogPanel>

            <ResponsiveDialogFooter className="gap-2">
              <ResponsiveDialogDesktopOnly>
                <Button
                  type="button"
                  variant="outline"
                  disabled={isSaving}
                  onClick={() => setScheduleDialogOpen(false)}
                >
                  <Ban className="ml-1.5 size-4" />
                  انصراف
                </Button>
              </ResponsiveDialogDesktopOnly>
              <Button
                type="submit"
                disabled={isSaving}
                className="w-full sm:w-auto"
              >
                <Save className="ml-1.5 size-4" />
                {isSaving ? "در حال ذخیره..." : "ذخیره برنامه حضور"}
              </Button>
            </ResponsiveDialogFooter>
          </form>
        </ResponsiveDialogContent>
      </ResponsiveDialog>

      {/* مودال اعمال بازه‌ای / چندماهه / سالانه */}
      <ResponsiveDialog
        open={bulkDialogOpen}
        onOpenChange={(open) => !isSaving && setBulkDialogOpen(open)}
        drawerPosition="bottom"
      >
        <ResponsiveDialogTrigger
          className="sr-only"
          tabIndex={-1}
          aria-hidden="true"
        />
        <ResponsiveDialogContent className="sm:max-w-xl">
          <ResponsiveDialogHeader>
            <ResponsiveDialogTitle>
              اعمال بازه‌ای / چندماهه / سالانه برنامه حضور
            </ResponsiveDialogTitle>
            <ResponsiveDialogDescription>
              فقط در روزها و ساعات کاریِ فعال کلینیک ثبت می‌شود؛ روزهای تعطیل کلینیک یا ساعات خارج از ساعت کاری کلینیک اعمال نخواهند شد.
            </ResponsiveDialogDescription>
          </ResponsiveDialogHeader>

          <form onSubmit={handleGenerateBulk} className="contents">
            <ResponsiveDialogPanel className="no-scrollbar max-h-[70vh] overflow-y-auto pt-2">
              <div className="px-5">
                <FieldGroup>
                  <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
                    <Field>
                      <FieldLabel htmlFor="bulkFrom">
                        از تاریخ شمسی (YYYY/MM/DD)
                      </FieldLabel>
                      <Input
                        id="bulkFrom"
                        dir="ltr"
                        required
                        value={bulkFromDate}
                        onChange={(e) => setBulkFromDate(e.target.value)}
                        placeholder="1405/09/01"
                      />
                    </Field>

                    <Field>
                      <FieldLabel htmlFor="bulkTo">
                        تا تاریخ شمسی (YYYY/MM/DD)
                      </FieldLabel>
                      <Input
                        id="bulkTo"
                        dir="ltr"
                        required
                        value={bulkToDate}
                        onChange={(e) => setBulkToDate(e.target.value)}
                        placeholder="1405/12/29"
                      />
                    </Field>
                  </div>

                  <Field>
                    <FieldLabel>روزهای هفته در این بازه</FieldLabel>
                    <div className="mt-1 flex flex-wrap gap-1.5">
                      {WEEKDAYS.map((d) => {
                        const checked = bulkWeekdays.includes(d.index)
                        return (
                          <button
                            key={d.index}
                            type="button"
                            onClick={() => toggleBulkWeekday(d.index)}
                            className={cn(
                              "rounded-lg border px-3 py-1.5 text-xs font-medium transition-colors",
                              checked
                                ? "border-primary bg-primary text-primary-foreground"
                                : "bg-background hover:bg-muted"
                            )}
                          >
                            {d.label}
                          </button>
                        )
                      })}
                    </div>
                  </Field>

                  <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
                    <Field>
                      <FieldLabel>ساعت شروع</FieldLabel>
                      <SplitTimeSelect
                        value={bulkStartTime}
                        onChange={setBulkStartTime}
                      />
                    </Field>

                    <Field>
                      <FieldLabel>ساعت پایان</FieldLabel>
                      <SplitTimeSelect
                        value={bulkEndTime}
                        onChange={setBulkEndTime}
                      />
                    </Field>
                  </div>

                  <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
                    <Field>
                      <FieldLabel>وضعیت حضور</FieldLabel>
                      <Select
                        value={bulkStatus}
                        onValueChange={(val) => {
                          const next =
                            (val as StaffScheduleStatus) ||
                            StaffScheduleStatus.IN_CLINIC
                          setBulkStatus(next)
                          if (
                            next === StaffScheduleStatus.HOSPITAL_SHIFT ||
                            next === StaffScheduleStatus.RESIDENT ||
                            next === StaffScheduleStatus.DAY_OFF
                          ) {
                            setBulkBlocksClinic(true)
                          } else if (next === StaffScheduleStatus.IN_CLINIC) {
                            setBulkBlocksClinic(false)
                          }
                        }}
                      >
                        <SelectTrigger className="bg-background">
                          <SelectValue>
                            {STATUS_META[bulkStatus].label}
                          </SelectValue>
                        </SelectTrigger>
                        <SelectContent>
                          {(
                            Object.keys(STATUS_META) as StaffScheduleStatus[]
                          ).map((st) => (
                            <SelectItem key={st} value={st}>
                              {STATUS_META[st].label}
                            </SelectItem>
                          ))}
                        </SelectContent>
                      </Select>
                    </Field>

                    <Field>
                      <FieldLabel>مکان متصل</FieldLabel>
                      <Select
                        value={bulkSpaceId}
                        onValueChange={(val) => setBulkSpaceId(val || "NONE")}
                      >
                        <SelectTrigger className="bg-background">
                          <SelectValue>
                            {bulkSpaceId === "NONE"
                              ? "بدون اتصال به مکان خاص"
                              : spaces.find((s) => s.id === bulkSpaceId)
                                  ?.name || "انتخاب مکان"}
                          </SelectValue>
                        </SelectTrigger>
                        <SelectContent>
                          <SelectItem value="NONE">
                            بدون اتصال به مکان خاص
                          </SelectItem>
                          {spaces.map((sp) => (
                            <SelectItem key={sp.id} value={sp.id}>
                              {sp.name}
                            </SelectItem>
                          ))}
                        </SelectContent>
                      </Select>
                    </Field>
                  </div>

                  <Field orientation="horizontal">
                    <Checkbox
                      id="bulkBlocksClinic"
                      checked={bulkBlocksClinic}
                      onCheckedChange={(v) => setBulkBlocksClinic(Boolean(v))}
                    />
                    <FieldLabel
                      htmlFor="bulkBlocksClinic"
                      className="cursor-pointer font-semibold"
                    >
                      بستن نوبت‌دهی مطب در این بازه زمانی
                    </FieldLabel>
                  </Field>

                  <Field orientation="horizontal">
                    <Checkbox
                      id="bulkReplace"
                      checked={bulkReplaceExisting}
                      onCheckedChange={(v) => setBulkReplaceExisting(Boolean(v))}
                    />
                    <div className="space-y-0.5">
                      <FieldLabel
                        htmlFor="bulkReplace"
                        className="cursor-pointer font-semibold"
                      >
                        جایگزینی برنامه‌های قبلی در روزهای انتخاب‌شده
                      </FieldLabel>
                      <FieldDescription className="text-[11px]">
                        در صورت فعال بودن، شیفت‌های قبلی همان روزها در بازه پاک شده و الگوی جدید جایگزین می‌شود.
                      </FieldDescription>
                    </div>
                  </Field>

                  <Field>
                    <FieldLabel>
                      سرویس‌های مجاز ({toFa(bulkServiceIds.length)} انتخاب‌شده)
                    </FieldLabel>
                    <div className="mt-1.5 grid max-h-36 grid-cols-1 gap-2 overflow-y-auto rounded-xl border p-2.5 sm:grid-cols-2">
                      {activeStaff.services.map((srv) => {
                        const checked = bulkServiceIds.includes(srv.serviceId)
                        return (
                          <label
                            key={srv.serviceId}
                            className={cn(
                              "flex cursor-pointer items-center gap-2 rounded-lg border px-2.5 py-2 text-xs transition-colors",
                              checked
                                ? "border-primary bg-primary/10 font-medium text-primary"
                                : "bg-background hover:bg-muted/50"
                            )}
                          >
                            <Checkbox
                              checked={checked}
                              onCheckedChange={() =>
                                toggleBulkService(srv.serviceId)
                              }
                            />
                            <span className="truncate">{srv.serviceTitle}</span>
                          </label>
                        )
                      })}
                    </div>
                  </Field>

                  <Field>
                    <FieldLabel htmlFor="bulkNote">
                      یادداشت بازه (اختیاری)
                    </FieldLabel>
                    <Input
                      id="bulkNote"
                      value={bulkNote}
                      onChange={(e) => setBulkNote(e.target.value)}
                      placeholder="مثلاً: برنامه فصلی یا شیفت آنکال بیمارستان"
                    />
                  </Field>
                </FieldGroup>
              </div>
            </ResponsiveDialogPanel>

            <ResponsiveDialogFooter className="gap-2">
              <ResponsiveDialogDesktopOnly>
                <Button
                  type="button"
                  variant="outline"
                  disabled={isSaving}
                  onClick={() => setBulkDialogOpen(false)}
                >
                  <Ban className="ml-1.5 size-4" />
                  انصراف
                </Button>
              </ResponsiveDialogDesktopOnly>
              <Button
                type="submit"
                disabled={isSaving}
                className="w-full sm:w-auto"
              >
                <CalendarRange className="ml-1.5 size-4" />
                {isSaving ? "در حال تولید..." : "تولید برنامه در بازه"}
              </Button>
            </ResponsiveDialogFooter>
          </form>
        </ResponsiveDialogContent>
      </ResponsiveDialog>
    </div>
  )
}
