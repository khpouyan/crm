// components/clinic-dashboard/appointments/spaces-client.tsx
"use client"

import * as React from "react"
import { useRouter } from "next/navigation"
import { ClinicSpaceType } from "@prisma/client"
import {
  Building2,
  DoorOpen,
  Layers,
  Plus,
  Search,
  ShieldCheck,
  Users,
  Wrench,
  Edit3,
  Trash2,
  ArrowLeftRight,
  Ban,
  Save,
  CheckCircle2,
  XCircle,
} from "lucide-react"
import { toast } from "sonner"

import { Button } from "@/components/ui/button"
import { Input } from "@/components/ui/input"
import { Textarea } from "@/components/ui/textarea"
import { Checkbox } from "@/components/ui/checkbox"
import { Badge } from "@/components/ui/badge"
import {
  Card,
  CardContent,
  CardDescription,
  CardHeader,
  CardTitle,
} from "@/components/ui/card"
import {
  Select,
  SelectContent,
  SelectItem,
  SelectTrigger,
  SelectValue,
} from "@/components/ui/select"
import {
  ResponsiveDialog,
  ResponsiveDialogTrigger,
  ResponsiveDialogContent,
  ResponsiveDialogHeader,
  ResponsiveDialogTitle,
  ResponsiveDialogDescription,
  ResponsiveDialogFooter,
  ResponsiveDialogPanel,
  ResponsiveDialogDesktopOnly,
} from "@/components/ui/responsive-dialog"
import {
  Field,
  FieldGroup,
  FieldLabel,
  FieldDescription,
} from "@/components/ui/field"
import {
  upsertClinicSpaceAction,
  moveClinicSpaceTypeAction,
  toggleClinicSpaceRuleAction,
  deleteClinicSpaceAction,
} from "@/app/clinic-dashboard/(dashboard)/settings/appointments/spaces/actions"
import { cn } from "@/lib/utils"

export interface ClinicServiceSummary {
  id: string
  title: string
  subServicesCount: number
}

export interface ClinicSpaceItem {
  id: string
  name: string
  spaceType: ClinicSpaceType
  capacity: number
  allowOverlap: boolean
  requiresPrivacy: boolean
  equipmentNotes: string | null
  description: string | null
  isActive: boolean
  sortOrder: number
  allowedServices: Array<{ id: string; title: string }>
}

interface SpacesKanbanClientProps {
  clinicName: string
  initialSpaces: ClinicSpaceItem[]
  services: ClinicServiceSummary[]
}

const FA_DIGITS = ["۰", "۱", "۲", "۳", "۴", "۵", "۶", "۷", "۸", "۹"]
function toFa(val: number | string): string {
  return String(val).replace(/\d/g, (d) => FA_DIGITS[Number(d)])
}

interface KanbanColumnConfig {
  type: ClinicSpaceType
  title: string
  subtitle: string
  badgeLabel: string
  icon: React.ElementType
  headerTone: string
}

const KANBAN_COLUMNS: KanbanColumnConfig[] = [
  {
    type: ClinicSpaceType.ROOM,
    title: "اتاق‌های کلینیک",
    subtitle: "اتاق لیزر، تزریق، مشاوره و معاینه داخل مرکز",
    badgeLabel: "اتاق",
    icon: DoorOpen,
    headerTone: "border-primary/30 bg-primary/5 text-primary",
  },
  {
    type: ClinicSpaceType.MULTI_UNIT,
    title: "فضاهای چندیونیتی / چندتخته",
    subtitle: "اتاق‌های دارای چند یونیت یا صندلی برای حضور هم‌زمان پرسنل",
    badgeLabel: "چندیونیتی",
    icon: Layers,
    headerTone:
      "border-amber-500/30 bg-amber-500/10 text-amber-700 dark:text-amber-300",
  },
  {
    type: ClinicSpaceType.HOSPITAL,
    title: "بیمارستان / اتاق عمل / آنکال",
    subtitle: "مراکز بیمارستانی، اتاق عمل و شیفت‌های آنکال جراحی",
    badgeLabel: "بیمارستان",
    icon: Building2,
    headerTone:
      "border-emerald-500/30 bg-emerald-500/10 text-emerald-700 dark:text-emerald-300",
  },
]

function getSpaceTypeLabel(type: ClinicSpaceType): string {
  return KANBAN_COLUMNS.find((c) => c.type === type)?.badgeLabel || "اتاق"
}

export function SpacesKanbanClient({
  clinicName,
  initialSpaces,
  services,
}: SpacesKanbanClientProps) {
  const router = useRouter()
  const [spaces, setSpaces] = React.useState<ClinicSpaceItem[]>(initialSpaces)
  const [searchQuery, setSearchQuery] = React.useState("")
  const [serviceFilter, setServiceFilter] = React.useState<string>("ALL")
  const [draggedSpaceId, setDraggedSpaceId] = React.useState<string | null>(null)
  const [dragOverColumn, setDragOverColumn] =
    React.useState<ClinicSpaceType | null>(null)

  const [isSubmitting, setIsSubmitting] = React.useState(false)
  const [busySpaceId, setBusySpaceId] = React.useState<string | null>(null)

  const [dialogOpen, setDialogOpen] = React.useState(false)
  const [editingSpaceId, setEditingSpaceId] = React.useState<string | null>(null)
  const [name, setName] = React.useState("")
  const [spaceType, setSpaceType] = React.useState<ClinicSpaceType>(
    ClinicSpaceType.ROOM
  )
  const [capacity, setCapacity] = React.useState("1")
  const [allowOverlap, setAllowOverlap] = React.useState(false)
  const [requiresPrivacy, setRequiresPrivacy] = React.useState(false)
  const [equipmentNotes, setEquipmentNotes] = React.useState("")
  const [description, setDescription] = React.useState("")
  const [isActive, setIsActive] = React.useState(true)
  const [selectedServiceIds, setSelectedServiceIds] = React.useState<string[]>([])

  React.useEffect(() => {
    setSpaces(initialSpaces)
  }, [initialSpaces])

  const openCreateDialog = (
    defaultType: ClinicSpaceType = ClinicSpaceType.ROOM
  ) => {
    setEditingSpaceId(null)
    setName("")
    setSpaceType(defaultType)
    setCapacity(defaultType === ClinicSpaceType.MULTI_UNIT ? "2" : "1")
    setAllowOverlap(defaultType === ClinicSpaceType.MULTI_UNIT)
    setRequiresPrivacy(defaultType === ClinicSpaceType.HOSPITAL)
    setEquipmentNotes("")
    setDescription("")
    setIsActive(true)
    setSelectedServiceIds([])
    setDialogOpen(true)
  }

  const openEditDialog = (space: ClinicSpaceItem) => {
    setEditingSpaceId(space.id)
    setName(space.name)
    setSpaceType(space.spaceType)
    setCapacity(String(space.capacity))
    setAllowOverlap(space.allowOverlap)
    setRequiresPrivacy(space.requiresPrivacy)
    setEquipmentNotes(space.equipmentNotes || "")
    setDescription(space.description || "")
    setIsActive(space.isActive)
    setSelectedServiceIds(space.allowedServices.map((s) => s.id))
    setDialogOpen(true)
  }

  const toggleServiceSelection = (serviceId: string) => {
    setSelectedServiceIds((prev) =>
      prev.includes(serviceId)
        ? prev.filter((id) => id !== serviceId)
        : [...prev, serviceId]
    )
  }

  const handleSaveSpace = async (e: React.FormEvent) => {
    e.preventDefault()
    setIsSubmitting(true)
    try {
      const parsedCapacity = Math.max(1, Number(capacity) || 1)
      const res = await upsertClinicSpaceAction({
        id: editingSpaceId || undefined,
        name,
        spaceType,
        capacity: parsedCapacity,
        allowOverlap,
        requiresPrivacy,
        equipmentNotes,
        description,
        isActive,
        allowedServiceIds: selectedServiceIds,
      })

      if (!res.success) {
        toast.error(res.error || "خطا در ذخیره مکان")
        return
      }

      toast.success(
        editingSpaceId
          ? "تغییرات مکان ذخیره شد."
          : "مکان جدید به کانبان اضافه شد."
      )
      setDialogOpen(false)
      router.refresh()
    } finally {
      setIsSubmitting(false)
    }
  }

  const handleMoveToColumn = async (
    spaceId: string,
    targetType: ClinicSpaceType
  ) => {
    const targetSpace = spaces.find((s) => s.id === spaceId)
    if (!targetSpace || targetSpace.spaceType === targetType) return

    setBusySpaceId(spaceId)
    setSpaces((prev) =>
      prev.map((item) =>
        item.id === spaceId ? { ...item, spaceType: targetType } : item
      )
    )

    try {
      const res = await moveClinicSpaceTypeAction(spaceId, targetType)
      if (!res.success) {
        toast.error(res.error || "خطا در جابه‌جایی مکان")
        setSpaces(initialSpaces)
        return
      }
      toast.success(
        `مکان «${targetSpace.name}» به ستون «${getSpaceTypeLabel(targetType)}» منتقل شد.`
      )
      router.refresh()
    } finally {
      setBusySpaceId(null)
    }
  }

  const handleQuickRuleToggle = async (
    space: ClinicSpaceItem,
    field: "isActive" | "allowOverlap" | "requiresPrivacy"
  ) => {
    const nextVal = !space[field]
    setBusySpaceId(space.id)
    setSpaces((prev) =>
      prev.map((item) =>
        item.id === space.id ? { ...item, [field]: nextVal } : item
      )
    )

    try {
      const res = await toggleClinicSpaceRuleAction(space.id, field, nextVal)
      if (!res.success) {
        toast.error(res.error || "خطا در تغییر وضعیت")
        setSpaces(initialSpaces)
        return
      }
      toast.success("قانون مکان به‌روزرسانی شد.")
      router.refresh()
    } finally {
      setBusySpaceId(null)
    }
  }

  const handleDeleteSpace = async (space: ClinicSpaceItem) => {
    setBusySpaceId(space.id)
    try {
      const res = await deleteClinicSpaceAction(space.id)
      if (!res.success) {
        toast.error(res.error || "خطا در حذف مکان")
        return
      }
      setSpaces((prev) => prev.filter((item) => item.id !== space.id))
      toast.success(`مکان «${space.name}» حذف شد.`)
      router.refresh()
    } finally {
      setBusySpaceId(null)
    }
  }

  const filteredSpaces = React.useMemo(() => {
    const q = searchQuery.trim().toLowerCase()
    return spaces.filter((sp) => {
      if (serviceFilter !== "ALL") {
        const hasService = sp.allowedServices.some((s) => s.id === serviceFilter)
        if (!hasService) return false
      }

      if (!q) return true

      const inName = sp.name.toLowerCase().includes(q)
      const inEquip = (sp.equipmentNotes || "").toLowerCase().includes(q)
      const inServices = sp.allowedServices.some((s) =>
        s.title.toLowerCase().includes(q)
      )
      return inName || inEquip || inServices
    })
  }, [spaces, searchQuery, serviceFilter])

  return (
    <div className="space-y-6 p-4 md:p-0" dir="rtl">
      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
        <div>
          <h1 className="text-xl font-bold tracking-tight sm:text-2xl">
            مدیریت مکان‌ها (نمای کانبان)
          </h1>
          <p className="mt-1 text-xs text-muted-foreground sm:text-sm">
            تعریف اتاق‌ها، فضای چندیونیتی و بیمارستان‌های مرکز «{clinicName}» به همراه سرویس‌های مجاز، ظرفیت، هم‌پوشانی و پرایوسی
          </p>
        </div>

        <Button
          type="button"
          onClick={() => openCreateDialog(ClinicSpaceType.ROOM)}
          className="w-full sm:w-auto"
        >
          <Plus className="ml-1.5 size-4" />
          افزودن مکان جدید
        </Button>
      </div>

      <Card>
        <CardContent className="p-3 sm:p-4">
          <div className="flex flex-col gap-3 sm:flex-row sm:items-center">
            <div className="relative flex-1">
              <Search className="pointer-events-none absolute top-1/2 right-3 size-4 -translate-y-1/2 text-muted-foreground" />
              <Input
                value={searchQuery}
                onChange={(e) => setSearchQuery(e.target.value)}
                placeholder="جستجو بر اساس نام مکان، سرویس یا تجهیزات..."
                className="pr-9"
              />
            </div>

            <div className="w-full sm:w-60">
              <Select
                value={serviceFilter}
                onValueChange={(val) => setServiceFilter(val || "ALL")}
              >
                <SelectTrigger className="bg-background">
                  <SelectValue>
                    {serviceFilter === "ALL"
                      ? "همه سرویس‌ها"
                      : services.find((s) => s.id === serviceFilter)?.title ||
                        "همه سرویس‌ها"}
                  </SelectValue>
                </SelectTrigger>
                <SelectContent>
                  <SelectItem value="ALL">همه سرویس‌ها</SelectItem>
                  {services.map((srv) => (
                    <SelectItem key={srv.id} value={srv.id}>
                      {srv.title}
                    </SelectItem>
                  ))}
                </SelectContent>
              </Select>
            </div>
          </div>
        </CardContent>
      </Card>

      <div className="grid grid-cols-1 gap-4 lg:grid-cols-3">
        {KANBAN_COLUMNS.map((col) => {
          const ColIcon = col.icon
          const colSpaces = filteredSpaces.filter((s) => s.spaceType === col.type)
          const isDropTarget = dragOverColumn === col.type

          return (
            <div
              key={col.type}
              onDragOver={(e) => {
                e.preventDefault()
                if (dragOverColumn !== col.type) {
                  setDragOverColumn(col.type)
                }
              }}
              onDragLeave={() => {
                if (dragOverColumn === col.type) {
                  setDragOverColumn(null)
                }
              }}
              onDrop={(e) => {
                e.preventDefault()
                setDragOverColumn(null)
                const idToMove = draggedSpaceId
                setDraggedSpaceId(null)
                if (idToMove) {
                  void handleMoveToColumn(idToMove, col.type)
                }
              }}
              className={cn(
                "flex min-h-96 flex-col rounded-2xl border bg-muted/25 p-3 transition-colors",
                isDropTarget && "border-primary bg-primary/10"
              )}
            >
              <div
                className={cn(
                  "mb-3 flex items-start justify-between gap-2 rounded-xl border p-3",
                  col.headerTone
                )}
              >
                <div className="space-y-0.5">
                  <div className="flex items-center gap-2">
                    <ColIcon className="size-4 shrink-0" />
                    <h2 className="text-sm font-bold">{col.title}</h2>
                    <Badge
                      variant="secondary"
                      className="h-5 px-1.5 text-[11px]"
                    >
                      {toFa(colSpaces.length)}
                    </Badge>
                  </div>
                  <p className="text-[11px] opacity-80">{col.subtitle}</p>
                </div>

                <Button
                  type="button"
                  variant="ghost"
                  size="icon-xs"
                  onClick={() => openCreateDialog(col.type)}
                  title={`افزودن به ${col.title}`}
                >
                  <Plus className="size-4" />
                </Button>
              </div>

              <div className="flex flex-1 flex-col gap-3">
                {colSpaces.length === 0 ? (
                  <div className="flex flex-1 flex-col items-center justify-center rounded-xl border border-dashed bg-background/50 p-6 text-center text-xs text-muted-foreground">
                    <p>مکانی در این ستون ثبت نشده است.</p>
                    <Button
                      type="button"
                      variant="link"
                      size="xs"
                      className="mt-1"
                      onClick={() => openCreateDialog(col.type)}
                    >
                      افزودن اولین {col.badgeLabel}
                    </Button>
                  </div>
                ) : (
                  colSpaces.map((space) => {
                    const isBusy = busySpaceId === space.id
                    return (
                      <Card
                        key={space.id}
                        draggable
                        onDragStart={() => setDraggedSpaceId(space.id)}
                        onDragEnd={() => {
                          setDraggedSpaceId(null)
                          setDragOverColumn(null)
                        }}
                        className={cn(
                          "cursor-grab border transition-all active:cursor-grabbing",
                          !space.isActive && "opacity-65",
                          draggedSpaceId === space.id && "scale-[0.99] opacity-60"
                        )}
                      >
                        <CardHeader className="p-3.5 pb-2">
                          <div className="flex items-start justify-between gap-2">
                            <div className="space-y-1">
                              <div className="flex flex-wrap items-center gap-1.5">
                                <CardTitle className="text-sm font-bold">
                                  {space.name}
                                </CardTitle>
                                {!space.isActive && (
                                  <Badge
                                    variant="outline"
                                    className="h-5 text-[10px] text-muted-foreground"
                                  >
                                    غیرفعال
                                  </Badge>
                                )}
                              </div>
                              {space.description && (
                                <CardDescription className="line-clamp-2 text-xs">
                                  {space.description}
                                </CardDescription>
                              )}
                            </div>

                            <div className="flex items-center gap-1">
                              <Button
                                type="button"
                                variant="ghost"
                                size="icon-xs"
                                disabled={isBusy}
                                onClick={() => openEditDialog(space)}
                                title="ویرایش مکان و سرویس‌ها"
                              >
                                <Edit3 className="size-3.5" />
                              </Button>
                              <Button
                                type="button"
                                variant="ghost"
                                size="icon-xs"
                                disabled={isBusy}
                                className="text-destructive hover:bg-destructive/10"
                                onClick={() => void handleDeleteSpace(space)}
                                title="حذف مکان"
                              >
                                <Trash2 className="size-3.5" />
                              </Button>
                            </div>
                          </div>
                        </CardHeader>

                        <CardContent className="space-y-3 p-3.5 pt-1">
                          <div className="flex flex-wrap gap-1.5 text-[11px]">
                            <Badge
                              variant="secondary"
                              className="gap-1 font-normal"
                            >
                              <Users className="size-3" />
                              ظرفیت: {toFa(space.capacity)} هم‌زمان
                            </Badge>

                            <button
                              type="button"
                              disabled={isBusy}
                              onClick={() =>
                                void handleQuickRuleToggle(space, "allowOverlap")
                              }
                              title="برای تغییر اجازه هم‌پوشانی کلیک کنید"
                            >
                              <Badge
                                variant={
                                  space.allowOverlap ? "default" : "outline"
                                }
                                className="cursor-pointer gap-1 font-normal"
                              >
                                {space.allowOverlap ? (
                                  <CheckCircle2 className="size-3" />
                                ) : (
                                  <XCircle className="size-3" />
                                )}
                                {space.allowOverlap
                                  ? "هم‌پوشانی مجاز"
                                  : "بدون هم‌پوشانی"}
                              </Badge>
                            </button>

                            <button
                              type="button"
                              disabled={isBusy}
                              onClick={() =>
                                void handleQuickRuleToggle(
                                  space,
                                  "requiresPrivacy"
                                )
                              }
                              title="برای تغییر شرط پرایوسی/اختصاصی کلیک کنید"
                            >
                              <Badge
                                variant={
                                  space.requiresPrivacy
                                    ? "destructive"
                                    : "outline"
                                }
                                className="cursor-pointer gap-1 font-normal"
                              >
                                <ShieldCheck className="size-3" />
                                {space.requiresPrivacy
                                  ? "پرایوسی / تک‌مراجع"
                                  : "پرایوسی اختیاری"}
                              </Badge>
                            </button>
                          </div>

                          {space.equipmentNotes && (
                            <div className="flex items-start gap-1.5 rounded-lg bg-muted/60 p-2 text-[11px] text-muted-foreground">
                              <Wrench className="mt-0.5 size-3.5 shrink-0 text-foreground/70" />
                              <span className="line-clamp-2">
                                {space.equipmentNotes}
                              </span>
                            </div>
                          )}

                          <div className="space-y-1.5">
                            <span className="block text-[11px] font-semibold text-muted-foreground">
                              سرویس‌های مجاز در این مکان (
                              {toFa(space.allowedServices.length)}):
                            </span>
                            {space.allowedServices.length > 0 ? (
                              <div className="flex flex-wrap gap-1">
                                {space.allowedServices.map((srv) => (
                                  <Badge
                                    key={srv.id}
                                    variant="outline"
                                    className="bg-background text-[11px] font-normal"
                                  >
                                    {srv.title}
                                  </Badge>
                                ))}
                              </div>
                            ) : (
                              <span className="block text-[11px] text-muted-foreground/80">
                                هنوز سرویسی به این مکان اختصاص داده نشده است.
                              </span>
                            )}
                          </div>

                          <div className="flex items-center justify-between gap-2 border-t pt-2.5">
                            <div className="flex items-center gap-1">
                              <ArrowLeftRight className="size-3.5 text-muted-foreground" />
                              <Select
                                value={space.spaceType}
                                disabled={isBusy}
                                onValueChange={(val) => {
                                  if (val) {
                                    void handleMoveToColumn(
                                      space.id,
                                      val as ClinicSpaceType
                                    )
                                  }
                                }}
                              >
                                <SelectTrigger className="h-7 w-28 bg-background text-[11px]">
                                  <SelectValue>
                                    {getSpaceTypeLabel(space.spaceType)}
                                  </SelectValue>
                                </SelectTrigger>
                                <SelectContent>
                                  {KANBAN_COLUMNS.map((c) => (
                                    <SelectItem key={c.type} value={c.type}>
                                      {c.badgeLabel}
                                    </SelectItem>
                                  ))}
                                </SelectContent>
                              </Select>
                            </div>

                            <Button
                              type="button"
                              variant="ghost"
                              size="xs"
                              disabled={isBusy}
                              onClick={() =>
                                void handleQuickRuleToggle(space, "isActive")
                              }
                              className="text-[11px]"
                            >
                              {space.isActive ? "غیرفعال کردن" : "فعال کردن"}
                            </Button>
                          </div>
                        </CardContent>
                      </Card>
                    )
                  })
                )}
              </div>
            </div>
          )
        })}
      </div>

      <ResponsiveDialog
        open={dialogOpen}
        onOpenChange={(open) => !isSubmitting && setDialogOpen(open)}

function getCurrentSystemTimeHHmm(): string {
  const now = new Date()
  const hh = String(now.getHours()).padStart(2, "0")
  const mm = String(now.getMinutes()).padStart(2, "0")
  return `${hh}:${mm}`
}

function parseHHmmToPickerValue(hhmm: string): TimePickerValue {
  const clean = normalizePersianDigits(hhmm || "").trim()
  const [h, m] = clean.split(":").map(Number)
  return {
    hour: Number.isFinite(h) ? Math.min(23, Math.max(0, h)) : 9,
    minute: Number.isFinite(m) ? Math.min(59, Math.max(0, m)) : 0,
  }
}

function normalizeTypedHHmm(raw: string, fallback: string): string {
  const clean = normalizePersianDigits(raw || "").trim()
  const match = /^(\d{1,2}):(\d{1,2})$/.exec(clean)
  if (!match) return fallback
  const h = Number(match[1])
  const m = Number(match[2])
  if (
    !Number.isInteger(h) ||
    !Number.isInteger(m) ||
    h < 0 ||
    h > 23 ||
    m < 0 ||
    m > 59
  ) {
    return fallback
  }
  return `${String(h).padStart(2, "0")}:${String(m).padStart(2, "0")}`
}

function jalaliStringToDate(jStr: string): Date | null {
  const norm = normalizePersianDigits(jStr || "").trim()
  const match = /^(\d{4})[/-](\d{1,2})[/-](\d{1,2})$/.exec(norm)
  if (!match) return null
  const jy = Number(match[1])
  const jm = Number(match[2])
  const jd = Number(match[3])
  if (jy < 1300 || jy > 1500 || jm < 1 || jm > 12 || jd < 1 || jd > 31) {
    return null
  }
  const g = toGregorian(jy, jm, jd)
  const check = toJalali(g)
  if (check.jy !== jy || check.jm !== jm || check.jd !== jd) return null
  return g
}

function dateToJalaliString(d: Date): string {
  const j = toJalali(d)
  return `${j.jy}/${String(j.jm).padStart(2, "0")}/${String(j.jd).padStart(
    2,
    "0"
  )}`
}

function parseJalaliParts(jStr: string): { jy: number; jm: number; jd: number } {
  const norm = normalizePersianDigits(jStr).trim()
  const [y, m, d] = norm.split(/[/-]/).map(Number)
  const now = toJalali(new Date())
  return {
    jy: y && y >= 1300 && y <= 1500 ? y : now.jy,
    jm: m && m >= 1 && m <= 12 ? m : now.jm,
    jd: d && d >= 1 && d <= 31 ? d : now.jd,
  }
}

function getMonthDaysCount(jy: number, jm: number): number {
  if (jm <= 6) return 31
  if (jm <= 11) return 30
  const g30 = toGregorian(jy, 12, 30)
  const j30 = toJalali(g30)
  return j30.jy === jy && j30.jm === 12 && j30.jd === 30 ? 30 : 29
}

function resolveHolidayTitle(gDate: Date): string | null {
  const hLib = persianHolidaysLib as {
    getHolidaysInRange?: (
      start: Date,
      end: Date
    ) => Array<{ date: Date; title?: string; isHoliday?: boolean }>
  }
  if (typeof hLib.getHolidaysInRange === "function") {
    const startOfDay = new Date(
      gDate.getFullYear(),
      gDate.getMonth(),
      gDate.getDate(),
      0,
      0,
      0
    )
    const endOfDay = new Date(
      gDate.getFullYear(),
      gDate.getMonth(),
      gDate.getDate(),
      23,
      59,
      59
    )
    const list = hLib.getHolidaysInRange(startOfDay, endOfDay)
    const found = list.find((item) => item.isHoliday !== false)
    if (found) return found.title || "تعطیل رسمی"
  }
  return null
}

        return false
      }
      if (
        filterSubServiceId !== "ALL" &&
        appt.subServiceId !== filterSubServiceId
      ) {
        return false
      }
      if (
        filterTherapistId !== "ALL" &&
        appt.therapistId !== filterTherapistId
      ) {
        return false
      }
      if (filterStatus !== "ALL" && appt.status !== filterStatus) {
        return false
      }
      if (q) {
        const fileNumText =
          appt.fileNumber !== null && appt.fileNumber !== undefined
            ? String(appt.fileNumber)
            : ""
        const hay = `${normalizePersianDigits(
          appt.clientName
        )} ${normalizePersianDigits(appt.clientPhone)} ${normalizePersianDigits(
          fileNumText
        )} ${appt.serviceTitle}`.toLowerCase()
        if (!hay.includes(q)) return false
      }
      return true
    })

    appointments,
    filterServiceId,
    filterSubServiceId,
    filterTherapistId,
    filterStatus,
    searchQuery,

          dialogMode === "reschedule" ||
          dialogMode === "replace_client"

            : undefined,
        clientId: formClientId,

        status: formStatus,
        allowManualOverride:

        overrideReason: allowOverride ? overrideReason : undefined,
        notes: formNotes,

    } finally {
      setIsSubmitting(false)
    }

        jalaliStr: null,
        isFriday: false,
        holidayTitle: null,
        items: [],
      })

    for (let d = 1; d <= monthDaysCount; d++) {
      const jStr = `${parsedSel.jy}/${String(parsedSel.jm).padStart(
        2,
        "0"

        (a) => normalizePersianDigits(a.jalaliDate).trim() === jStr
      )
      arr.push({
        day: d,
        jalaliStr: jStr,
        isFriday: weekday === 6,
        holidayTitle,
        items: dayItems,
      })

    firstWeekday,
    monthDaysCount,
    parsedSel.jy,
    parsedSel.jm,
    filteredAppointments,

  return (

        <div>
          <h1 className="text-xl font-bold tracking-tight sm:text-2xl">
            {activeTab === "daily" && "روزانه"}

            {activeTab === "table" && "جدولی"}
            {activeTab === "logs" && "گزارش رخدادها"}

            مدیریت نوبت‌دهی دستی و اتوماتیک ({clinicName})

        <div className="flex flex-wrap items-center gap-2">
          <Button
            type="button"
            variant="outline"
            size="sm"
            onClick={() =>
              openInlineWizard(selectedDate, getCurrentSystemTimeHHmm())

          >
            <LayoutPanelTop className="ml-1.5 size-4 text-primary" />

          <Button
            type="button"
            size="sm"
            onClick={() =>
              openModalWizard(selectedDate, getCurrentSystemTimeHHmm())

        {filtersOpen && (
          <CardContent className="grid grid-cols-1 gap-3 p-4 sm:grid-cols-2 lg:grid-cols-5">
            <Field>

              <DatePicker

              <FieldLabel>خدمات</FieldLabel>
              <Select

                              <Badge variant="outline" className="text-[10px]">
                                {appt.bookingSource ===
                                AppointmentBookingSource.AUTO
                                  ? "رزرو اتوماتیک"
                                  : "رزرو دستی"}
                              </Badge>
                            </div>
                          </div>

                    </div>
                  </div>
                )

            </CardContent>

            <CardContent className="p-3 sm:p-4">
              <div className="mb-2 flex items-center justify-between">

                  {JALALI_MONTH_NAMES[parsedSel.jm - 1]} {toFa(parsedSel.jy)}

                    {toFa(filteredAppointments.length)}

              <div className="mb-2 hidden grid-cols-7 gap-2 text-center text-xs font-bold text-muted-foreground md:grid">
                {WEEKDAYS.map((w, idx) => (
                  <div
                    key={w}

                      "rounded-lg bg-muted/50 py-2",

                    )}
                  >
                    {w}
                  </div>
                ))}
              </div>

              <div className="grid grid-cols-1 gap-2 sm:grid-cols-2 md:grid-cols-7">
                {monthlyCells.map((cell, idx) => {
                  if (!cell.day || !cell.jalaliStr) {
                    return (
                      <div
                        key={`empty-${idx}`}
                        className="hidden min-h-24 rounded-xl border border-dashed bg-muted/10 md:block"
                      />
                    )
                  }

                  const isSelectedDay =
                    normalizePersianDigits(selectedDate).trim() === cell.jalaliStr

                  return (
                    <button
                      key={cell.jalaliStr}

                      onClick={() => handleDateChange(cell.jalaliStr!)}
                      className={cn(
                        "flex min-h-24 flex-col justify-between rounded-xl border bg-background p-2.5 text-right transition-colors hover:border-primary/50",
                        isSelectedDay && "border-primary ring-1 ring-primary",
                        (cell.isFriday || cell.holidayTitle) &&
                          "border-destructive/30 bg-destructive/5"
                      )}
                    >
                      <div className="flex w-full items-center justify-between">
                        <span
                          className={cn(
                            "text-xs font-bold",
                            cell.isFriday || cell.holidayTitle
                              ? "text-destructive"
                              : "text-foreground"
                          )}
                        >
                          {toFa(cell.day)}
                        </span>
                        {cell.items.length > 0 && (
                          <Badge variant="default" className="h-5 text-[10px]">
                            {toFa(cell.items.length)} نوبت

                      {cell.holidayTitle && (
                        <div className="mt-1 truncate text-[10px] font-medium text-destructive">
                          {cell.holidayTitle}

                      )}

                        {cell.items.slice(0, 2).map((item) => (
                          <div
                            key={item.id}
                            className="truncate rounded bg-muted/60 px-1.5 py-0.5 text-[10px]"
                          >
                            {toFa(item.startTime)} — {item.clientName}

                        ))}

                    </button>
                  )

              </div>
            </CardContent>
          </Card>
        </TabsContent>

        <TabsContent value="table" className="space-y-3">
          <div className="flex items-center justify-between text-xs text-muted-foreground">
            <span>
              تعداد کل نوبت‌ها:{" "}
              <strong className="text-foreground">
                {toFa(filteredAppointments.length)}
              </strong>
            </span>
          </div>

          <div className="overflow-x-auto rounded-xl border bg-card">
            <Table>
              <TableHeader>
                <TableRow className="bg-muted/40">
                  <TableHead className="text-right">ردیف</TableHead>
                  <TableHead className="text-right">تاریخ</TableHead>
                  <TableHead className="text-right">ساعت</TableHead>
                  <TableHead className="text-right">مراجعه‌کننده</TableHead>
                  <TableHead className="text-right">شماره موبایل</TableHead>
                  <TableHead className="text-right">خدمات</TableHead>
                  <TableHead className="text-right">ریز خدمات</TableHead>
                  <TableHead className="text-right">وضعیت</TableHead>
                  <TableHead className="text-right">درمانگر</TableHead>
                  <TableHead className="text-right">روش ثبت</TableHead>
                  <TableHead className="text-left">عملیات</TableHead>
                </TableRow>
              </TableHeader>
              <TableBody>
                {filteredAppointments.length === 0 ? (
                  <TableRow>
                    <TableCell
                      colSpan={11}
                      className="py-8 text-center text-sm text-muted-foreground"
                    >
                      نوبتی با این فیلترها یافت نشد.
                    </TableCell>
                  </TableRow>
                ) : (
                  filteredAppointments.map((appt, index) => {

                    return (
                      <TableRow key={appt.id}>
                        <TableCell>{toFa(index + 1)}</TableCell>
                        <TableCell dir="ltr" className="text-right font-mono">
                          {toFa(appt.jalaliDate)}
                        </TableCell>
                        <TableCell dir="ltr" className="text-right font-mono">
                          {toFa(appt.startTime)}
                        </TableCell>
                        <TableCell className="font-medium">
                          {appt.clientName}
                          {appt.fileNumber
                            ? ` - ${toFa(appt.fileNumber)}`

                          <Badge variant="outline" className="text-[11px]">

                              </DropdownMenuItem>

                                onClick={() =>
                                  void handleStatusChange(
                                    appt.id,
                                    AppointmentStatus.CANCELED
                                  )
                                }
                              >
                                <RotateCcw className="ml-2 size-3.5" />
                                انتقال به لیست رزرو / لغو
                              </DropdownMenuItem>
                              <DropdownMenuSeparator />
                              <DropdownMenuItem
                                onClick={() => openActionDialog(appt, "edit")}
                              >
                                <Edit3 className="ml-2 size-3.5" />
                                ویرایش
                              </DropdownMenuItem>
                              <DropdownMenuItem
                                className="text-destructive"
                                onClick={() => void handleDeleteAppt(appt.id)}
                              >
                                <Trash2 className="ml-2 size-3.5" />
                                حذف
                              </DropdownMenuItem>
                            </DropdownMenuContent>
                          </DropdownMenu>
                        </TableCell>
                      </TableRow>
                    )
                  })
                )}
              </TableBody>
            </Table>
          </div>
        </TabsContent>

        <TabsContent value="logs" className="space-y-3">
          <div className="overflow-x-auto rounded-xl border bg-card">
            <Table>
              <TableHeader>
                <TableRow className="bg-muted/40">
                  <TableHead className="text-right">تاریخ</TableHead>
                  <TableHead className="text-right">ساعت نوبت</TableHead>
                  <TableHead className="text-right">ثبت‌کننده</TableHead>
                  <TableHead className="text-right">نوع ثبت</TableHead>
                  <TableHead className="text-right">توضیحات</TableHead>
                </TableRow>
              </TableHeader>
              <TableBody>
                {filteredAppointments.length === 0 ? (
                  <TableRow>
                    <TableCell
                      colSpan={5}
                      className="py-8 text-center text-sm text-muted-foreground"
                    >
                      رخدادی ثبت نشده است.
                    </TableCell>
                  </TableRow>
                ) : (
                  filteredAppointments.map((appt) => (
                    <TableRow key={`log-${appt.id}`}>
                      <TableCell dir="ltr" className="text-right font-mono">
                        {toFa(appt.jalaliDate)}
                      </TableCell>
                      <TableCell dir="ltr" className="text-right font-mono">
                        {toFa(appt.startTime)} - {toFa(appt.blockEndTime)}
                      </TableCell>
                      <TableCell>{appt.createdByName}</TableCell>
                      <TableCell>
                        {appt.bookingSource === AppointmentBookingSource.AUTO
                          ? "نوبت اتوماتیک"
                          : "نوبت دستی"}
                      </TableCell>
                      <TableCell>
                        نوبت برای <strong>{appt.clientName}</strong> در بخش{" "}
                        <strong>{appt.serviceTitle}</strong>
                        {appt.subServiceTitle
                          ? ` (${appt.subServiceTitle})`
                          : ""}{" "}
                        با درمانگر <strong>{appt.therapistName}</strong>.
                        {appt.isManualOverride && appt.overrideReason
                          ? ` [هماهنگی ویژه: ${appt.overrideReason}]`
                          : ""}
                      </TableCell>
                    </TableRow>
                  ))
                )}
              </TableBody>
            </Table>
          </div>
        </TabsContent>
      </Tabs>

      {/* ویزارد هم به‌صورت مودال برای ثبت سریع یا ویرایش/جابه‌جایی/جایگزینی مراجع */}
      <ResponsiveDialog
        open={dialogOpen}
        onOpenChange={(open) => !isSubmitting && setDialogOpen(open)}
        drawerPosition="bottom"
      >
        <ResponsiveDialogTrigger
          className="sr-only"
          tabIndex={-1}
          aria-hidden="true"
        />
        <ResponsiveDialogContent className="sm:max-w-2xl">
          <ResponsiveDialogHeader>
            <ResponsiveDialogTitle>
              {dialogMode === "create" && "رزرو نوبت (ویزارد سه‌مرحله‌ای)"}
              {dialogMode === "edit" && "ویرایش نوبت"}
              {dialogMode === "reschedule" && "جابه‌جایی زمان نوبت"}
              {dialogMode === "replace_client" && "جایگزین کردن مراجع‌کننده"}
            </ResponsiveDialogTitle>
            <ResponsiveDialogDescription>
              مراحل انتخاب مراجع، خدمت/پرسنل/اتاق و تعیین زمان نوبت را گام‌به‌گام تکمیل کنید.
            </ResponsiveDialogDescription>
          </ResponsiveDialogHeader>

          <ResponsiveDialogPanel className="no-scrollbar max-h-[72vh] overflow-y-auto pt-2">
            <div className="px-5">{renderWizardBody()}</div>
          </ResponsiveDialogPanel>

          <ResponsiveDialogFooter className="gap-2">
            <ResponsiveDialogDesktopOnly>
              <Button
                type="button"
                variant="outline"
                disabled={isSubmitting}
                onClick={() => setDialogOpen(false)}
              >
                بستن
              </Button>
            </ResponsiveDialogDesktopOnly>
          </ResponsiveDialogFooter>
        </ResponsiveDialogContent>
      </ResponsiveDialog>
    </div>
  )
}

// components/clinic-dashboard/appointments/client-combobox.tsx
"use client"

import * as React from "react"
import { AlertTriangle, Check, UserPlus, X } from "lucide-react"
import { toast } from "sonner"

import { Button } from "@/components/ui/button"
import { Input } from "@/components/ui/input"
import { Badge } from "@/components/ui/badge"
import { Checkbox } from "@/components/ui/checkbox"
import {
  Field,
  FieldError,
  FieldGroup,
  FieldLabel,
} from "@/components/ui/field"
import {
  Tooltip,
  TooltipContent,
  TooltipProvider,
  TooltipTrigger,
} from "@/components/ui/tooltip"
import {
  Combobox,
  ComboboxClear,
  ComboboxContent,
  ComboboxEmpty,
  ComboboxIcon,
  ComboboxInput,
  ComboboxInputGroup,
  ComboboxItem,
  ComboboxList,
} from "@/components/ui/combobox"
import {
  quickCreateClientAction,
  type CreatedClientSummary,
} from "@/app/clinic-dashboard/(dashboard)/calendar/actions"
import { isValidNationalId } from "@/lib/national-id"
import { normalizeIranPhone, isValidIranPhone } from "@/lib/iranian-mobile"
import { normalizePersianDigits } from "@/lib/normalize-persian-digits"
import { cn } from "@/lib/utils"

export interface ClientComboboxItem {
  id: string
  fullName: string
  phone: string
  nationalId?: string | null
  nickname?: string | null
  fileNumber?: number | null
}

interface ClientComboboxProps {
  clients: ClientComboboxItem[]
  value: string
  disabled?: boolean
  onChange: (clientId: string) => void
  onClientCreated?: (client: CreatedClientSummary) => void
  quickCreateOpen?: boolean
  onQuickCreateOpenChange?: (open: boolean) => void
}

const FA_DIGITS = ["۰", "۱", "۲", "۳", "۴", "۵", "۶", "۷", "۸", "۹"]
function toFa(val: number | string): string {
  return String(val).replace(/\d/g, (d) => FA_DIGITS[Number(d)])
}

export function ClientCombobox({
  clients,
  value,
  disabled = false,
  onChange,
  onClientCreated,
  quickCreateOpen: controlledQuickOpen,
  onQuickCreateOpenChange,
}: ClientComboboxProps) {
  const [searchText, setSearchText] = React.useState("")
  const deferredSearch = React.useDeferredValue(searchText)

  const [internalQuickOpen, setInternalQuickOpen] = React.useState(false)
  const isQuickOpen = controlledQuickOpen ?? internalQuickOpen
  const setQuickOpen = (next: boolean) => {
    if (controlledQuickOpen === undefined) {
      setInternalQuickOpen(next)
    }
    onQuickCreateOpenChange?.(next)
  }

  const [newFullName, setNewFullName] = React.useState("")
  const [newNickname, setNewNickname] = React.useState("")
  const [newPhone, setNewPhone] = React.useState("")
  const [noPhone, setNoPhone] = React.useState(false)
  const [newNationalId, setNewNationalId] = React.useState("")
  const [noNationalId, setNoNationalId] = React.useState(false)
  const [isCreating, setIsCreating] = React.useState(false)
  const [createError, setCreateError] = React.useState<string | null>(null)

  const selectedClient = React.useMemo(
    () => clients.find((c) => c.id === value) || null,
    [clients, value]
  )

  React.useEffect(() => {
    if (selectedClient && !searchText) {
      const nickPart = selectedClient.nickname
        ? ` (${selectedClient.nickname})`
        : ""
      setSearchText(
        `${selectedClient.fullName}${nickPart}${
          selectedClient.phone ? ` - ${toFa(selectedClient.phone)}` : ""
        }`
      )
    }
  }, [selectedClient, searchText])

  const duplicateNameSet = React.useMemo(() => {
    const counts = new Map<string, number>()
    for (const c of clients) {
      const key = (c.fullName || "").trim().toLowerCase()
      if (!key) continue
      counts.set(key, (counts.get(key) || 0) + 1)
    }
    const dups = new Set<string>()
    for (const [key, count] of counts.entries()) {
      if (count > 1) dups.add(key)
    }
    return dups
  }, [clients])

  const visibleClients = React.useMemo(() => {
    const q = normalizePersianDigits(deferredSearch).trim().toLowerCase()
    if (!q) return clients.slice(0, 20)
    const matches: ClientComboboxItem[] = []
    for (const c of clients) {
      const name = (c.fullName || "").toLowerCase()
      const nick = (c.nickname || "").toLowerCase()
      const phone = normalizePersianDigits(c.phone || "")
      const natId = normalizePersianDigits(c.nationalId || "")
      const fileNo = normalizePersianDigits(String(c.fileNumber ?? ""))
      if (
        name.includes(q) ||
        nick.includes(q) ||
        phone.includes(q) ||
        natId.includes(q) ||
        fileNo.includes(q)
      ) {
        matches.push(c)
        if (matches.length >= 25) break
      }
    }
    return matches
  }, [clients, deferredSearch])

  // بررسی نام مشابه در پرونده‌های کلینیک
  const matchingClients = React.useMemo(() => {
    const targetName = newFullName.trim().toLowerCase()
    if (!targetName) return []
    return clients.filter(
      (c) => (c.fullName || "").trim().toLowerCase() === targetName
    )
  }, [clients, newFullName])

  const hasDuplicateName = matchingClients.length > 0

  const normalizedPhoneForCheck = React.useMemo(
    () => normalizeIranPhone(newPhone),
    [newPhone]
  )
  const cleanNationalIdForCheck = React.useMemo(
    () => normalizePersianDigits(newNationalId).replace(/\D/g, ""),
    [newNationalId]
  )

  const isPhoneInvalid = React.useMemo(() => {
    if (noPhone) return false
    if (!newPhone.trim()) return false
    return !isValidIranPhone(newPhone)
  }, [noPhone, newPhone])

  const isNationalIdInvalid = React.useMemo(() => {
    if (noNationalId) return false
    if (!cleanNationalIdForCheck) return false
    return !isValidNationalId(cleanNationalIdForCheck)
  }, [noNationalId, cleanNationalIdForCheck])

  const duplicatePhoneClient = React.useMemo(() => {
    if (noPhone || !normalizedPhoneForCheck) return null
    return (
      clients.find(
        (c) =>
          c.phone && normalizeIranPhone(c.phone) === normalizedPhoneForCheck
      ) || null
    )
  }, [clients, noPhone, normalizedPhoneForCheck])

  const duplicateNationalIdClient = React.useMemo(() => {
    if (noNationalId || !cleanNationalIdForCheck) return null
    return (
      clients.find(
        (c) =>
          c.nationalId &&
          normalizePersianDigits(c.nationalId).replace(/\D/g, "") ===
            cleanNationalIdForCheck
      ) || null
    )
  }, [clients, noNationalId, cleanNationalIdForCheck])

  const handleOpenQuickCreate = () => {
    const trimmed = normalizePersianDigits(searchText).trim()
    setCreateError(null)
    setNoPhone(false)
    setNoNationalId(false)
    if (/^\d{10}$/.test(trimmed)) {
      setNewNationalId(trimmed)
      setNewPhone("")
      setNewFullName("")
    } else if (/^09\d{9}$/.test(trimmed) || /^9\d{9}$/.test(trimmed)) {
      setNewPhone(normalizeIranPhone(trimmed))
      setNewNationalId("")
      setNewFullName("")
    } else {
      setNewFullName(searchText.trim())
      setNewPhone("")
      setNewNationalId("")
    }
    setNewNickname("")
    setQuickOpen(true)
  }

  const handleCreateClient = async (options?: {
    skipPhone?: boolean
    skipNationalId?: boolean
  }) => {
    setCreateError(null)
    if (!newFullName.trim()) {
      setCreateError("علت: وارد کردن نام و نام خانوادگی مراجع الزامی است.")
      return
    }

    const effectiveNoPhone = options?.skipPhone ? true : noPhone
    const effectiveNoNatId = options?.skipNationalId ? true : noNationalId

    if (!effectiveNoPhone && newPhone.trim() && !isValidIranPhone(newPhone)) {
      setCreateError("شماره موبایل وارد شده معتبر نیست.")
      return
    }

    if (
      !effectiveNoNatId &&
      cleanNationalIdForCheck &&
      !isValidNationalId(cleanNationalIdForCheck)
    ) {
      setCreateError("کد ملی وارد شده معتبر نیست.")
      return
    }

    const finalPhone = effectiveNoPhone
      ? null
      : newPhone.trim()
      ? normalizeIranPhone(newPhone)
      : null

    const finalNationalId = effectiveNoNatId
      ? null
      : cleanNationalIdForCheck || null

    setIsCreating(true)
    try {
      const res = await quickCreateClientAction({
        fullName: newFullName.trim(),
        phone: finalPhone,
        nationalId: finalNationalId,
        nickname: hasDuplicateName ? newNickname.trim() || null : null,
      })

      if (!res.success || !res.client) {
        const msg = res.error || "علت: امکان ثبت پرونده مراجع وجود ندارد."
        setCreateError(msg)
        toast.error(msg)
        return
      }

      onClientCreated?.(res.client)
      onChange(res.client.id)
      const nickPart = res.client.nickname ? ` (${res.client.nickname})` : ""
      setSearchText(
        `${res.client.fullName}${nickPart}${
          res.client.phone ? ` - ${toFa(res.client.phone)}` : ""
        }`
      )
      setQuickOpen(false)
      toast.success(`پرونده مراجع «${res.client.fullName}» ایجاد و انتخاب شد.`)
    } finally {
      setIsCreating(false)
    }
  }

  return (
    <TooltipProvider>
      <div className="space-y-2.5" dir="rtl">
        <div className="flex flex-col gap-2 sm:flex-row sm:items-center">
          <div className="flex-1">
            <Combobox
              value={selectedClient}
              disabled={disabled}
              onValueChange={(nextClient) => {
                if (nextClient) {
                  onChange(nextClient.id)
                  const nickPart = nextClient.nickname
                    ? ` (${nextClient.nickname})`
                    : ""
                  setSearchText(
                    `${nextClient.fullName}${nickPart}${
                      nextClient.phone ? ` - ${toFa(nextClient.phone)}` : ""
                    }`
                  )
                } else {
                  onChange("")
                  setSearchText("")
                }
              }}
            >
              <ComboboxInputGroup className="h-10 bg-background">
                <ComboboxInput
                  disabled={disabled}
                  value={searchText}
                  onChange={(e) => {
                    setSearchText(e.target.value)
                    if (value) onChange("")
                  }}
                  placeholder="تایپ کنید: کد ملی، شماره تماس، نام یا نام مستعار..."
                />
                {searchText && !disabled && (
                  <ComboboxClear
                    onClick={() => {
                      setSearchText("")
                      onChange("")
                    }}
                  />
                )}
                <ComboboxIcon />
              </ComboboxInputGroup>

              <ComboboxContent align="start">
                <ComboboxList>
                  {visibleClients.length === 0 ? (
                    <ComboboxEmpty className="py-4 text-xs">
                      مراجعی با این مشخصات یافت نشد. از دکمه «ثبت مراجع جدید» پرونده بسازید.
                    </ComboboxEmpty>
                  ) : (
                    visibleClients.map((client) => {
                      const hasDupInList = duplicateNameSet.has(
                        (client.fullName || "").trim().toLowerCase()
                      )
                      return (
                        <ComboboxItem
                          key={client.id}
                          value={client}
                          className="py-2 text-xs"
                        >
                          <div className="flex w-full flex-col gap-0.5">
                            <div className="flex flex-wrap items-center gap-1.5">
                              <span className="font-bold text-foreground">
                                {client.fullName}
                              </span>
                              {(client.nickname || hasDupInList) && (
                                <Badge
                                  variant={
                                    client.nickname ? "secondary" : "outline"
                                  }
                                  className="h-4 px-1.5 text-[10px]"
                                >
                                  {client.nickname
                                    ? `نام مستعار: ${client.nickname}`
                                    : "هم‌نام"}
                                </Badge>
                              )}
                              {client.fileNumber !== null &&
                                client.fileNumber !== undefined && (
                                  <span className="text-[10px] text-muted-foreground">
                                    پرونده #{toFa(client.fileNumber)}
                                  </span>
                                )}
                            </div>
                            <div className="flex flex-wrap items-center gap-3 text-[11px] text-muted-foreground">
                              <span>
                                موبایل:{" "}
                                <strong dir="ltr" className="font-mono">
                                  {client.phone
                                    ? toFa(client.phone)
                                    : "ثبت نشده"}
                                </strong>
                              </span>
                              <span>
                                کد ملی:{" "}
                                <strong dir="ltr" className="font-mono">
                                  {client.nationalId
                                    ? toFa(client.nationalId)
                                    : "ثبت نشده"}
                                </strong>
                              </span>
                            </div>
                          </div>
                        </ComboboxItem>
                      )
                    })
                  )}
                </ComboboxList>
              </ComboboxContent>
            </Combobox>
          </div>

          {!disabled && (
            <Tooltip>
              <TooltipTrigger
                render={
                  <Button
                    type="button"
                    variant="outline"
                    size="sm"
                    onClick={() =>
                      isQuickOpen
                        ? setQuickOpen(false)
                        : handleOpenQuickCreate()
                    }
                    className="h-10 shrink-0"
                  />
                }
              >
                {isQuickOpen ? (
                  <>
                    <X className="ml-1.5 size-4" />
                    بستن ثبت سریع
                  </>
                ) : (
                  <>
                    <UserPlus className="ml-1.5 size-4 text-primary" />
                    ثبت مراجع جدید
                  </>
                )}
              </TooltipTrigger>
              <TooltipContent side="top" className="text-xs">
                تشکیل پرونده جدید برای مراجع در همین بخش
              </TooltipContent>
            </Tooltip>
          )}
        </div>

        {/* ردیف ثبت سریع مراجع جدید در زیر موارد انتخابی */}
        {isQuickOpen && !disabled && (
          <div className="space-y-3 rounded-xl border border-primary/30 bg-muted/25 p-3.5">
            <div className="flex items-center justify-between">
              <span className="text-xs font-bold text-primary">
                ثبت سریع پرونده مراجع جدید
              </span>
              <span className="text-[11px] text-muted-foreground">
                پس از ثبت، به‌صورت خودکار انتخاب می‌شود
              </span>
            </div>

            <FieldGroup>
              <Field>
                <FieldLabel htmlFor="qc-fullname">
                  نام و نام خانوادگی *
                </FieldLabel>
                <Input
                  id="qc-fullname"
                  value={newFullName}
                  onChange={(e) => setNewFullName(e.target.value)}
                  placeholder="مثلاً: سارا محمدی"
                  className="h-9 bg-background text-xs"
                />
              </Field>

              {/* هشدار زرد مراجع با نام مشابه + فیلد نام مستعار فقط در صورت تکرار نام */}
              {hasDuplicateName && (
                <>
                  <div className="space-y-1.5 rounded-xl border border-amber-400/70 bg-amber-400/15 p-3 text-xs text-amber-950 dark:border-amber-400/50 dark:bg-amber-400/15 dark:text-amber-100">
                    <div className="flex items-center gap-1.5 font-bold">
                      <AlertTriangle className="size-4 shrink-0 text-amber-600 dark:text-amber-300" />
                      <span>مراجع با نام مشابه در سیستم ثبت شده است</span>
                    </div>
                    <p className="leading-relaxed text-amber-900/90 dark:text-amber-200/90">
                      پرونده‌های هم‌نام:{" "}
                      {matchingClients
                        .slice(0, 3)
                        .map((m) =>
                          [
                            m.fullName,
                            m.fileNumber ? `پرونده #${toFa(m.fileNumber)}` : null,
                            m.phone ? toFa(m.phone) : null,
                            m.nickname ? `نام مستعار: ${m.nickname}` : null,
                          ]
                            .filter(Boolean)
                            .join(" — ")
                        )
                        .join(" | ")}
                    </p>
                  </div>

                  <Field>
                    <FieldLabel htmlFor="qc-nickname">
                      نام مستعار (اختیاری)
                    </FieldLabel>
                    <Input
                      id="qc-nickname"
                      value={newNickname}
                      onChange={(e) => setNewNickname(e.target.value)}
                      placeholder="مثلاً: سارا - پاسداران یا سارا ۲"
                      className="h-9 border-amber-400/60 bg-background text-xs focus-visible:border-amber-500"
                    />
                  </Field>
                </>
              )}

              <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
                {/* شماره تماس + بدون شماره */}
                <Field data-invalid={isPhoneInvalid || undefined}>
                  <FieldLabel htmlFor="qc-phone">
                    شماره تماس {!noPhone && "*"}
                  </FieldLabel>
                  <Input
                    id="qc-phone"
                    dir="ltr"
                    type="tel"
                    disabled={noPhone}
                    aria-invalid={isPhoneInvalid}
                    value={newPhone}
                    onChange={(e) => setNewPhone(e.target.value)}
                    onBlur={() => {
                      if (!noPhone && newPhone.trim()) {
                        setNewPhone(normalizeIranPhone(newPhone))
                      }
                    }}
                    placeholder="09123456789"
                    className="h-9 bg-background text-xs"
                  />
                  {isPhoneInvalid && (
                    <FieldError>شماره موبایل وارد شده معتبر نیست.</FieldError>
                  )}
                  {duplicatePhoneClient && (
                    <FieldError className="text-[11px] font-medium text-amber-700 dark:text-amber-300">
                      این شماره در پرونده «
                      {duplicatePhoneClient.fullName || "بدون نام"}»
                      {duplicatePhoneClient.fileNumber
                        ? ` (#${toFa(duplicatePhoneClient.fileNumber)})`
                        : ""}{" "}
                      ثبت شده است.
                    </FieldError>
                  )}
                  <Field orientation="horizontal" className="pt-1">
                    <Checkbox
                      id="qc-no-phone"
                      checked={noPhone}
                      onCheckedChange={(checked) => {
                        const isChecked = Boolean(checked)
                        setNoPhone(isChecked)
                        if (isChecked) setNewPhone("")
                      }}
                    />
                    <FieldLabel
                      htmlFor="qc-no-phone"
                      className="cursor-pointer text-xs font-normal"
                    >
                      بدون شماره
                    </FieldLabel>
                  </Field>
                </Field>

                {/* کد ملی + فاقد کد ملی */}
                <Field data-invalid={isNationalIdInvalid || undefined}>
                  <FieldLabel htmlFor="qc-nationalid">
                    کد ملی {!noNationalId && "*"}
                  </FieldLabel>
                  <Input
                    id="qc-nationalid"
                    dir="ltr"
                    maxLength={10}
                    disabled={noNationalId}
                    aria-invalid={isNationalIdInvalid}
                    value={newNationalId}
                    onChange={(e) => setNewNationalId(e.target.value)}
                    placeholder="۱۰ رقم کد ملی"
                    className="h-9 bg-background text-xs"
                  />
                  {isNationalIdInvalid && (

function getJalaliMonthDaysCount(jy: number, jm: number): number {
  if (jm <= 6) return 31
  if (jm <= 11) return 30
  const g30 = toGregorian(jy, 12, 30)
  const j30 = toJalali(g30)
  return j30.jy === jy && j30.jm === 12 && j30.jd === 30 ? 30 : 29
}

function getHolidayTitleForDate(gDate: Date): string | null {
  const hLib = persianHolidaysLib as {
    getHolidaysInRange?: (
      start: Date,
      end: Date
    ) => Array<{ date: Date; title?: string; isHoliday?: boolean }>
  }
  if (typeof hLib.getHolidaysInRange === "function") {
    const startOfDay = new Date(
      gDate.getFullYear(),
      gDate.getMonth(),
      gDate.getDate(),
      0,
      0,
      0
    )
    const endOfDay = new Date(
      gDate.getFullYear(),
      gDate.getMonth(),
      gDate.getDate(),
      23,
      59,
      59
    )
    const list = hLib.getHolidaysInRange(startOfDay, endOfDay)
    const found = list.find((item) => item.isHoliday !== false)
    if (found) return found.title || "تعطیل رسمی"
  }
  return null
}

        reqEnd <= timeToMinutes(sh.endTime)
    )

    if (!insideClinicShift) {
      const allowedStr = dayPol.shifts
        .map((s) => `${s.startTime} تا ${s.endTime}`)
        .join(" و ")
      toast.error(
        `بازه انتخاب‌شده خارج از ساعات کاری کلینیک در این روز است. بازه‌های مجاز کلینیک: ${toFa(
          allowedStr
        )}`
      )
      return
    }

    setIsSaving(true)
    try {
      const res = await upsertStaffScheduleAction({
        id: editingScheduleId || undefined,
        clinicUserId: activeStaff.id,
        spaceId: spaceId === "NONE" ? null : spaceId,
        jalaliDate: normDate,
        startTime,
        endTime,
        status: scheduleStatus,
        blocksClinicBook,
        note: scheduleNote,
        isActive: scheduleIsActive,
        allowedServiceIds: scheduleServiceIds,
      })

      if (!res.success) {
        toast.error(res.error || "خطا در ثبت برنامه حضور")
        return
      }

      toast.success("برنامه حضور روزانه ذخیره شد.")
      setScheduleDialogOpen(false)
      router.refresh()
    } finally {
      setIsSaving(false)
    }
  }

  const openBulkGeneratorDialog = () => {
    const lastDay = getJalaliMonthDaysCount(selectedYear, selectedMonth)
    setBulkFromDate(formatJalaliKey(selectedYear, selectedMonth, 1))
    setBulkToDate(formatJalaliKey(selectedYear, selectedMonth, lastDay))
    setBulkWeekdays([0, 2])
    setBulkStartTime("09:00")
    setBulkEndTime("17:00")
    setBulkStatus(StaffScheduleStatus.IN_CLINIC)
    setBulkSpaceId(spaces[0]?.id || "NONE")
    setBulkBlocksClinic(false)
    setBulkNote("")
    setBulkServiceIds(activeStaff.services.map((s) => s.serviceId))
    setBulkReplaceExisting(false)
    setBulkDialogOpen(true)
  }

  const toggleBulkWeekday = (dayIdx: number) => {
    setBulkWeekdays((prev) =>
      prev.includes(dayIdx)
        ? prev.filter((d) => d !== dayIdx)
        : [...prev, dayIdx]
    )
  }

  const toggleBulkService = (srvId: string) => {
    setBulkServiceIds((prev) =>
      prev.includes(srvId)
        ? prev.filter((id) => id !== srvId)
        : [...prev, srvId]
    )
  }

  const handleGenerateBulk = async (e: React.FormEvent) => {
    e.preventDefault()
    setIsSaving(true)
    try {
      const res = await generateBulkStaffSchedulesAction({
        clinicUserId: activeStaff.id,
        fromJalaliDate: normalizePersianDigits(bulkFromDate),
        toJalaliDate: normalizePersianDigits(bulkToDate),
        weekdays: bulkWeekdays,
        spaceId: bulkSpaceId === "NONE" ? null : bulkSpaceId,
        startTime: bulkStartTime,
        endTime: bulkEndTime,
        status: bulkStatus,
        blocksClinicBook: bulkBlocksClinic,
        note: bulkNote,
        allowedServiceIds: bulkServiceIds,
        replaceOnTargetDays: bulkReplaceExisting,
      })

      if (!res.success) {
        toast.error(res.error || "خطا در تولید برنامه بازه‌ای")
        return
      }

      const skippedMsg =
        res.skippedCount && res.skippedCount > 0
          ? ` (${toFa(res.skippedCount)} روز به دلیل تعطیلی یا خارج بودن از ساعات کاری کلینیک اعمال نشد)`
          : ""

      toast.success(
        `برنامه حضور برای ${toFa(
          res.createdCount || 0
        )} روز در بازه انتخاب‌شده ایجاد شد.${skippedMsg}`
      )
      setBulkDialogOpen(false)
      router.refresh()
    } finally {
      setIsSaving(false)
    }
  }

  const handleDeleteSchedule = async (slotId: string) => {
    setIsSaving(true)
    try {
      const res = await deleteStaffScheduleAction(slotId)
      if (!res.success) {
        toast.error(res.error || "خطا در حذف شیفت")
        return
      }
      toast.success("بازه حضور حذف شد.")
      router.refresh()
    } finally {
      setIsSaving(false)
    }
  }

  const monthSlotsCount = calendarCells.reduce(
    (acc, cell) => acc + cell.slots.length,
    0
  )

  return (
    <div className="space-y-6 p-4 md:p-0" dir="rtl">
      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
        <div>
          <h1 className="text-xl font-bold tracking-tight sm:text-2xl">
            زمان‌بندی ماهانه، مکان و سرویس‌های پرسنل
          </h1>
          <p className="mt-1 text-xs text-muted-foreground sm:text-sm">
            ثبت حضور، آنکال، مقیم و شیفت بیمارستان صرفاً در روزها و ساعات کاری مجاز کلینیک ({clinicName})
          </p>
        </div>

        {isAdmin && staffProfiles.length > 0 && (
          <div className="w-full sm:w-72">
            <Select
              value={activeStaff.id}
              onValueChange={(val) => val && setSelectedStaffId(val)}
            >
              <SelectTrigger className="bg-background">
                <SelectValue>
                  {activeStaff.fullName} ({activeStaff.jobTitle})
                </SelectValue>
              </SelectTrigger>
              <SelectContent>
                {staffProfiles.map((member) => (
                  <SelectItem key={member.id} value={member.id}>
                    {member.fullName} — {member.jobTitle}
                  </SelectItem>
                ))}
              </SelectContent>
            </Select>
          </div>
        )}
      </div>

      <Card>
        <CardContent className="flex flex-col gap-3 p-4 sm:flex-row sm:items-center sm:justify-between">
          <div className="flex items-center gap-3">
            <div className="flex size-10 items-center justify-center rounded-xl bg-primary/10 text-primary">
              <UserCheck className="size-5" />
            </div>
            <div>
              <div className="flex flex-wrap items-center gap-2">
                <h2 className="text-base font-bold">{activeStaff.fullName}</h2>
                <Badge variant="secondary">{activeStaff.jobTitle}</Badge>
                {activeStaff.specialty && (
                  <Badge variant="outline">{activeStaff.specialty}</Badge>
                )}
              </div>
              <p className="mt-0.5 text-xs text-muted-foreground">
                سرویس‌های مجاز:{" "}
                <strong className="text-foreground">
                  {toFa(activeStaff.services.length)} سرویس
                </strong>{" "}
                | شیفت‌های ثبت‌شده در{" "}
                <strong className="text-foreground">
                  {JALALI_MONTHS[selectedMonth - 1]} {toFa(selectedYear)}
                </strong>
                :{" "}
                <strong className="text-foreground">
                  {toFa(monthSlotsCount)} مورد
                </strong>
              </p>
            </div>
          </div>

          <div className="flex flex-col gap-2 sm:flex-row">
            <Button
              type="button"
              variant="outline"
              onClick={openBulkGeneratorDialog}
              className="w-full sm:w-auto"
            >
              <CalendarRange className="ml-1.5 size-4" />
              اعمال بازه‌ای / چندماهه / سالانه
            </Button>
          </div>
        </CardContent>
      </Card>

      <Tabs defaultValue="schedules" dir="rtl" className="space-y-4">
        <TabsList className="grid w-full max-w-md grid-cols-2 [&_[data-slot=tabs-indicator]]:hidden">
          <TabsTrigger
            value="schedules"
            className="data-[active]:bg-primary data-[active]:text-primary-foreground data-[active]:shadow-xs"
          >
            <CalendarClock className="ml-1.5 size-4" />
            تقویم حضور ماهانه و آنکال
          </TabsTrigger>
          <TabsTrigger
            value="services"
            className="data-[active]:bg-primary data-[active]:text-primary-foreground data-[active]:shadow-xs"
          >
            <Clock className="ml-1.5 size-4" />
            زمان‌بندی سرویس‌های مجاز
          </TabsTrigger>
        </TabsList>

        <TabsContent value="schedules" className="space-y-4">
          <Card>
            <CardHeader className="flex flex-col gap-3 border-b p-4 sm:flex-row sm:items-center sm:justify-between">
              <div className="flex items-center gap-2">
                <Button
                  type="button"
                  variant="outline"
                  size="icon-xs"
                  onClick={handlePrevMonth}
                  title="ماه قبل"
                >
                  <ChevronRight className="size-4" />
                </Button>

                <div className="flex items-center gap-2">
                  <Select
                    value={String(selectedMonth)}
                    onValueChange={(v) => setSelectedMonth(Number(v ?? 1))}
                  >
                    <SelectTrigger className="w-32 bg-background">
                      <SelectValue>
                        {JALALI_MONTHS[selectedMonth - 1]}
                      </SelectValue>
                    </SelectTrigger>
                    <SelectContent>
                      {JALALI_MONTHS.map((mName, idx) => (
                        <SelectItem key={mName} value={String(idx + 1)}>
                          {mName}
                        </SelectItem>
                      ))}
                    </SelectContent>
                  </Select>

                  <Select
                    value={String(selectedYear)}
                    onValueChange={(v) =>
                      setSelectedYear(Number(v ?? initialYm.year))
                    }
                  >
                    <SelectTrigger className="w-24 bg-background">
                      <SelectValue>{toFa(selectedYear)}</SelectValue>
                    </SelectTrigger>
                    <SelectContent>
                      {[
                        initialYm.year - 1,
                        initialYm.year,
                        initialYm.year + 1,
                        initialYm.year + 2,
                      ].map((yr) => (
                        <SelectItem key={yr} value={String(yr)}>
                          {toFa(yr)}
                        </SelectItem>
                      ))}
                    </SelectContent>
                  </Select>
                </div>

                <Button
                  type="button"
                  variant="outline"
                  size="icon-xs"
                  onClick={handleNextMonth}
                  title="ماه بعد"
                >
                  <ChevronLeft className="size-4" />
                </Button>
              </div>

              <Button
                type="button"
                variant="ghost"
                size="xs"
                onClick={handleGoCurrentMonth}
              >
                برو به ماه جاری
              </Button>
            </CardHeader>

            <CardContent className="p-3 sm:p-4">
              <div className="mb-2 hidden grid-cols-7 gap-2 text-center text-xs font-bold text-muted-foreground md:grid">
                {WEEKDAYS.map((w) => (
                  <div
                    key={w.index}
                    className={cn(
                      "rounded-lg bg-muted/50 py-2",
                      w.index === 6 && "text-destructive"
                    )}
                  >
                    {w.label}
                  </div>
                ))}
              </div>

              <div className="grid grid-cols-1 gap-2.5 sm:grid-cols-2 md:grid-cols-7">
                {calendarCells.map((cell, idx) => {
                  if (!cell.day || !cell.key) {
                    return (
                      <div
                        key={`empty-${idx}`}
                        className="hidden min-h-32 rounded-xl border border-dashed bg-muted/10 md:block"
                      />
                    )
                  }

                  const isFriday = cell.weekday === 6
                  const isOfficialHoliday = Boolean(cell.holidayTitle)
                  const dayLabel =
                    WEEKDAYS.find((w) => w.index === cell.weekday)?.label || ""

                  return (
                    <div
                      key={cell.key}
                      className={cn(
                        "flex min-h-32 flex-col justify-between rounded-xl border bg-background p-2.5 transition-colors",
                        cell.isDayBlockedForStaff
                          ? "border-destructive/30 bg-destructive/5"
                          : "hover:border-primary/40"
                      )}
                    >
                      <div>
                        <div className="mb-1.5 flex items-center justify-between gap-1">
                          <div className="flex items-center gap-1.5">
                            <span
                              className={cn(
                                "inline-flex size-6 items-center justify-center rounded-full text-xs font-bold",
                                isFriday ||
                                  isOfficialHoliday ||
                                  cell.isDayBlockedForStaff
                                  ? "bg-destructive/15 text-destructive"
                                  : "bg-primary/10 text-primary"
                              )}
                            >
                              {toFa(cell.day)}
                            </span>
                            <span className="text-[11px] font-medium text-muted-foreground md:hidden">
                              {dayLabel}
                            </span>
                          </div>

                          {!cell.isDayBlockedForStaff ? (
                            <Button
                              type="button"
                              variant="ghost"
                              size="icon-xs"
                              onClick={() =>
                                openCreateScheduleForDate(cell.key!)
                              }
                              title={`ثبت شیفت برای ${toFa(cell.key)}`}
                            >
                              <Plus className="size-3.5" />
                            </Button>
                          ) : (
                            <span
                              className="inline-flex items-center text-destructive/80"
                              title="تعطیل یا فاقد ساعت کاری ثبت‌شده در کلینیک"
                            >
                              <Lock className="size-3.5" />
                            </span>
                          )}
                        </div>

                        {cell.holidayTitle && (
                          <div className="mb-1 rounded-md bg-destructive/10 px-1.5 py-0.5 text-[10px] font-medium text-destructive">
                            {cell.holidayTitle}
                          </div>
                        )}

                        {!cell.clinicDay ? (
                          <div className="mb-1.5 rounded-md bg-muted px-1.5 py-1 text-[10px] text-muted-foreground">
                            ساعت کاری کلینیک ثبت نشده
                          </div>
                        ) : cell.clinicDay.isClinicHoliday ? (
                          <div className="mb-1.5 rounded-md bg-destructive/10 px-1.5 py-1 text-[10px] font-medium text-destructive">
                            تعطیل کلینیک{" "}
                            {cell.clinicDay.holidayNote
                              ? `(${cell.clinicDay.holidayNote})`
                              : ""}
                          </div>
                        ) : cell.clinicDay.shifts.length > 0 ? (
                          <div className="mb-1.5 space-y-0.5 rounded-md bg-emerald-500/10 px-1.5 py-1 text-[10px] text-emerald-800 dark:text-emerald-300">
                            <div className="font-semibold">ساعت کلینیک:</div>
                            {cell.clinicDay.shifts.map((sh, sIdx) => (
                              <div key={sIdx} dir="ltr" className="font-mono">
                                {toFa(sh.startTime)} - {toFa(sh.endTime)}
                              </div>
                            ))}
                          </div>
                        ) : null}

                        <div className="space-y-1.5">
                          {cell.slots.map((slot) => {
                            const meta = STATUS_META[slot.status]
                            return (
                              <div
                                key={slot.id}
                                className="space-y-1 rounded-lg border bg-muted/35 p-1.5 text-[11px]"
                              >
                                <div className="flex items-center justify-between gap-1">
                                  <Badge
                                    variant={meta.badgeVariant}
                                    className="h-4 px-1.5 text-[10px]"
                                  >
                                    {meta.shortLabel}
                                  </Badge>
                                  <div className="flex items-center gap-0.5">
                                    <button
                                      type="button"
                                      disabled={isSaving}
                                      onClick={() => openEditSchedule(slot)}
                                      className="rounded p-0.5 hover:bg-background"
                                      title="ویرایش"
                                    >
                                      <Edit3 className="size-3" />
                                    </button>
                                    <button
                                      type="button"
                                      disabled={isSaving}
                                      onClick={() =>
                                        void handleDeleteSchedule(slot.id)
                                      }
                                      className="rounded p-0.5 text-destructive hover:bg-destructive/10"
                                      title="حذف"
                                    >
                                      <Trash2 className="size-3" />
                                    </button>
                                  </div>
                                </div>

                                <div
                                  dir="ltr"
                                  className="font-mono text-[10px] font-semibold"
                                >
                                  {toFa(slot.startTime)} - {toFa(slot.endTime)}
                                </div>

                                {slot.space && (
                                  <div className="flex items-center gap-1 truncate text-[10px] text-muted-foreground">
                                    {slot.space.spaceType ===
                                    ClinicSpaceType.HOSPITAL ? (
                                      <Hospital className="size-3 shrink-0 text-emerald-600" />
                                    ) : (
                                      <DoorOpen className="size-3 shrink-0 text-primary" />
                                    )}
                                    <span className="truncate">
                                      {slot.space.name}
                                    </span>
                                  </div>
                                )}

                                {slot.blocksClinicBook && (
                                  <div className="flex items-center gap-0.5 text-[10px] font-medium text-destructive">
                                    <AlertTriangle className="size-2.5 shrink-0" />
                                    <span>نوبت مطب بسته</span>
                                  </div>
                                )}
                              </div>
                            )
                          })}
                        </div>
                      </div>
                    </div>
                  )
                })}
              </div>
            </CardContent>
          </Card>
        </TabsContent>

        <TabsContent value="services" className="space-y-4">
          {activeStaff.services.length === 0 ? (
            <Card>
              <CardContent className="p-6 text-center text-sm text-muted-foreground">
                هنوز هیچ سرویسی به این پرسنل اختصاص داده نشده است. ابتدا از بخش «مدیریت پرسنل» خدمات مجاز این فرد را مشخص کنید.
              </CardContent>
            </Card>
          ) : (
            <div className="grid grid-cols-1 gap-4 md:grid-cols-2 lg:grid-cols-3">
              {activeStaff.services.map((srv) => {
                const isSplit =
                  srv.timingMode === ServiceTimingMode.SPLIT_STAGE
                const totalCycleMinutes = isSplit
                  ? srv.prepBeforeMinutes +
                    srv.stage1Minutes +
                    srv.freeGapMinutes +
                    srv.stage2Minutes +
                    srv.restAfterMinutes
                  : srv.prepBeforeMinutes +
                    srv.durationMinutes +
                    srv.restAfterMinutes

                return (
                  <Card key={srv.serviceId} className="flex flex-col">
                    <CardHeader className="p-4 pb-2">
                      <div className="flex items-start justify-between gap-2">
                        <div className="space-y-1">
                          <CardTitle className="text-base font-bold">
                            {srv.serviceTitle}
                          </CardTitle>
                          <div className="flex flex-wrap gap-1.5">
                            <Badge
                              variant={isSplit ? "default" : "secondary"}
                              className="gap-1 text-[11px]"
                            >
                              {isSplit ? (
                                <>
                                  <Split className="size-3" />
                                  چندمرحله‌ای (دارای تایم آزاد میانی)
                                </>
                              ) : (
                                <>
                                  <Clock className="size-3" />
                                  تک‌مرحله‌ای استاندارد
                                </>
                              )}
                            </Badge>
                            {srv.isCustomized && (
                              <Badge variant="outline" className="text-[11px]">
                                شخصی‌سازی‌شده
                              </Badge>
                            )}
                          </div>
                        </div>

                        <Button
                          type="button"
                          variant="outline"
                          size="xs"
                          onClick={() => openEditServiceTiming(srv)}
                        >
                          <Edit3 className="ml-1 size-3.5" />
                          تنظیم زمان
                        </Button>
                      </div>
                    </CardHeader>

                    <CardContent className="flex-1 space-y-2.5 p-4 pt-2 text-xs">
                      <div className="space-y-1.5 rounded-xl border bg-muted/20 p-3">
                        <div className="flex items-center justify-between">
                          <span className="text-muted-foreground">
                            حضور مراجع قبل از شروع:
                          </span>
                          <span className="font-semibold">
                            {toFa(srv.prepBeforeMinutes)} دقیقه
                          </span>
                        </div>

                        {isSplit ? (
                          <>
                            <div className="flex items-center justify-between">
                              <span className="text-muted-foreground">
                                مرحله ۱ (مثلاً خون‌گیری):
                              </span>
                              <span className="font-semibold">
                                {toFa(srv.stage1Minutes)} دقیقه
                              </span>
                            </div>
                            <div className="flex items-center justify-between text-emerald-700 dark:text-emerald-300">
                              <span className="flex items-center gap-1 font-medium">
                                <Sparkles className="size-3" />
                                تایم آزاد میانی (قابل رزرو برای دیگران):
                              </span>
                              <span className="font-bold">
                                {toFa(srv.freeGapMinutes)} دقیقه
                              </span>
                            </div>
                            <div className="flex items-center justify-between">
                              <span className="text-muted-foreground">
                                مرحله ۲ (مثلاً تزریق نهایی):
                              </span>
                              <span className="font-semibold">
                                {toFa(srv.stage2Minutes)} دقیقه
                              </span>
                            </div>
                          </>
                        ) : (
                          <div className="flex items-center justify-between">
                            <span className="text-muted-foreground">
                              مدت انجام سرویس:
                            </span>
                            <span className="font-semibold">
                              {toFa(srv.durationMinutes)} دقیقه
                            </span>
                          </div>
                        )}

                        <div className="flex items-center justify-between border-t pt-1.5">
                          <span className="flex items-center gap-1 text-muted-foreground">
                            <Coffee className="size-3" />
                            زمان استراحت / پاکسازی:
                          </span>
                          <span className="font-semibold">
                            {toFa(srv.restAfterMinutes)} دقیقه
                          </span>
                        </div>
                      </div>

                      <div className="flex items-center justify-between text-[11px] text-muted-foreground">
                        <span>کل بازه چرخه نوبت:</span>
                        <strong className="text-foreground">
                          {toFa(totalCycleMinutes)} دقیقه
                        </strong>
                      </div>

                      {srv.requiresPrivacy && (
                        <div className="flex items-center gap-1.5 rounded-lg bg-amber-500/10 px-2.5 py-1.5 text-[11px] font-medium text-amber-800 dark:text-amber-300">
                          <ShieldCheck className="size-3.5 shrink-0" />
                          نیازمند فضای خالی / رعایت کامل پرایوسی مراجع
                        </div>
                      )}
                    </CardContent>
                  </Card>
                )
              })}
            </div>
          )}
        </TabsContent>
      </Tabs>

      {/* مودال تنظیم زمان‌بندی سرویس */}
      <ResponsiveDialog
        open={timingDialogOpen}
        onOpenChange={(open) => !isSaving && setTimingDialogOpen(open)}
        drawerPosition="bottom"
      >
        <ResponsiveDialogTrigger
          className="sr-only"
          tabIndex={-1}
          aria-hidden="true"
        />
        <ResponsiveDialogContent className="sm:max-w-lg">
          <ResponsiveDialogHeader>
            <ResponsiveDialogTitle>
              تنظیم زمان‌بندی «{editingService?.serviceTitle}»
            </ResponsiveDialogTitle>
            <ResponsiveDialogDescription>
              مدت انجام سرویس، زمان حضور مراجع قبل از شروع، استراحت و در صورت نیاز تایم آزاد میانی (مانند PRP) را مشخص کنید.
            </ResponsiveDialogDescription>
          </ResponsiveDialogHeader>

          <form onSubmit={handleSaveServiceTiming} className="contents">
            <ResponsiveDialogPanel className="no-scrollbar max-h-[70vh] overflow-y-auto pt-2">
              <div className="px-5">
                <FieldGroup>
                  <Field>
                    <FieldLabel>مدل زمان‌بندی سرویس</FieldLabel>
                    <Select
                      value={timingMode}
                      onValueChange={(val) =>
                        setTimingMode(
                          (val as ServiceTimingMode) ||
                            ServiceTimingMode.SINGLE_STAGE
                        )
                      }
                    >
                      <SelectTrigger className="bg-background">

const FA_DIGITS = ["۰", "۱", "۲", "۳", "۴", "۵", "۶", "۷", "۸", "۹"]
function toFa(val: number | string): string {
  return String(val).replace(/\d/g, (d) => FA_DIGITS[Number(d)])
}

function getShiftModelMeta(cat: ShiftCategory) {
  return (
    SHIFT_MODEL_OPTIONS.find((m) => m.value === cat) || SHIFT_MODEL_OPTIONS[4]
  )
}

function getShiftModelLabel(cat: ShiftCategory, customLabel?: string): string {
  if (customLabel?.trim()) return customLabel.trim()
  return getShiftModelMeta(cat).label
}

function formatSingleClockCompact(hhmm: string): string {
  const clean = normalizePersianDigits(hhmm || "").trim()
  const match = /^(\d{1,2}):(\d{2})$/.exec(clean)
  if (!match) return toFa(clean)
  const h = Number(match[1])
  const m = Number(match[2])
  if (m === 0) return toFa(h)
  return `${toFa(h)}:${toFa(String(m).padStart(2, "0"))}`
}

function formatCompactShiftRange(
  category: ShiftCategory,
  startTime: string,
  endTime: string
): string {
  if (category === "ALL_DAY") return "۲۴ ساعته"
  return `${formatSingleClockCompact(startTime)} - ${formatSingleClockCompact(
    endTime
  )}`
}

function createShiftByCategory(
  category: ShiftCategory,
  index: number
): WorkingShiftInput {
  const found = getShiftModelMeta(category)
  return {
    id: `shift-${Date.now()}-${index}-${Math.random().toString(36).slice(2, 6)}`,
    category,
    label: found.label,
    startTime: found.defaultStart,
    endTime: found.defaultEnd,
  }
}

function getJalaliMonthDaysCount(jy: number, jm: number): number {
  if (jm <= 6) return 31
  if (jm <= 11) return 30
  const g30 = toGregorian(jy, 12, 30)
  const j30 = toJalali(g30)
  return j30.jy === jy && j30.jm === 12 && j30.jd === 30 ? 30 : 29
}

function toLocalDateIso(gDate: Date): string {
  const y = gDate.getFullYear()
  const m = String(gDate.getMonth() + 1).padStart(2, "0")
  const d = String(gDate.getDate()).padStart(2, "0")
  return `${y}-${m}-${d}`
}

function parseTimeString(timeStr: string): TimePickerValue {
  const clean = normalizePersianDigits(timeStr || "").trim()
  const match = /^(\d{1,2}):(\d{1,2})$/.exec(clean)
  if (!match) return { hour: 9, minute: 0 }
  const hour = Math.min(23, Math.max(0, Number(match[1])))
  const minute = Math.min(59, Math.max(0, Number(match[2])))
  return { hour, minute }
}

function normalizeTypedTime(raw: string, fallback: string): string {
  const clean = normalizePersianDigits(raw || "").trim()
  const match = /^(\d{1,2}):(\d{1,2})$/.exec(clean)
  if (!match) return fallback
  const hour = Number(match[1])
  const minute = Number(match[2])
  if (
    !Number.isInteger(hour) ||
    !Number.isInteger(minute) ||
    hour < 0 ||
    hour > 23 ||
    minute < 0 ||
    minute > 59
  ) {
    return fallback
  }
  return `${String(hour).padStart(2, "0")}:${String(minute).padStart(2, "0")}`
}

interface ShiftTimeFieldProps {
  pickerKey: string
  label: string
  timeValue: string
  disabled?: boolean
  drawerTitle: string
  onChange: (nextTime: string) => void
}

function ShiftTimeField({
  pickerKey,
  label,
  timeValue,
  disabled = false,
  drawerTitle,
  onChange,
}: ShiftTimeFieldProps) {
  const [open, setOpen] = React.useState(false)
  const [text, setText] = React.useState(() => toFa(timeValue))

  React.useEffect(() => {
    setText(toFa(timeValue))
  }, [timeValue, pickerKey])

  const commitTyped = (raw: string) => {
    const normalized = normalizeTypedTime(raw, timeValue)
    onChange(normalized)
    setText(toFa(normalized))
  }

  const parsedValue = React.useMemo(
    () => parseTimeString(timeValue),
    [timeValue]
  )

  return (
    <div className="flex flex-1 items-center gap-1.5">
      <span className="shrink-0 text-xs text-muted-foreground">{label}</span>
      <ResponsiveTimePicker
        open={!disabled && open}
        onOpenChange={(nextOpen) => {
          if (!disabled) setOpen(nextOpen)
        }}
      >
        <InputGroup className="h-9 bg-background">
          <InputGroupInput
            dir="ltr"
            inputMode="numeric"
            disabled={disabled}
            placeholder="۰۹:۰۰"
            value={text}
            onChange={(e) => setText(e.target.value)}
            onBlur={(e) => commitTyped(e.target.value)}
            onKeyDown={(e) => {
              if (e.key === "Enter") {
                e.preventDefault()
                commitTyped(text)
              }
            }}
            className="text-right text-xs tabular-nums"
          />
          <InputGroupAddon align="inline-end">
            <ResponsiveTimePickerTrigger
              render={
                <InputGroupButton
                  type="button"
                  size="icon-xs"
                  variant="ghost"
                  disabled={disabled}
                  aria-label={`انتخاب ${label}`}
                >
                  <Clock className="size-4" />
                </InputGroupButton>
              }
            />
          </InputGroupAddon>
        </InputGroup>

        <ResponsiveTimePickerContent
          value={parsedValue}
          onValueChange={(next) => {
            const formatted = `${String(next.hour).padStart(2, "0")}:${String(
              next.minute
            ).padStart(2, "0")}`
            onChange(formatted)
            setText(toFa(formatted))
          }}
          hourFormat="24"
          digits="fa"
          drawerTitle={drawerTitle}
          drawerConfirmLabel="تایید ساعت"
          onDrawerConfirm={() => setOpen(false)}
        />
      </ResponsiveTimePicker>
    </div>
  )
}

                            onValueChange={(val) =>
                              handleBulkShiftCategoryChange(
                                shift.id,
                                (val as ShiftCategory) || "CUSTOM"
                              )
                            }
                          >
                            <SelectTrigger className="h-9 bg-background text-xs">
                              <SelectValue>
                                <span className="inline-flex items-center gap-1.5">
                                  <ShiftIcon
                                    className={cn("size-3.5", meta.textClass)}
                                  />
                                  <span>{currentModelLabel}</span>
                                </span>
                              </SelectValue>
                            </SelectTrigger>
                            <SelectContent>
                              {SHIFT_MODEL_OPTIONS.map((opt) => {
                                const OptIcon = opt.icon
                                return (
                                  <SelectItem key={opt.value} value={opt.value}>
                                    <span className="inline-flex items-center gap-1.5">
                                      <OptIcon
                                        className={cn(
                                          "size-3.5",
                                          opt.textClass
                                        )}
                                      />
                                      <span>{opt.label}</span>
                                    </span>
                                  </SelectItem>
                                )
                              })}
                            </SelectContent>
                          </Select>
                        </div>

                        <div className="flex flex-1 flex-col gap-2 sm:flex-row sm:items-center sm:gap-3">
                          <ShiftTimeField
                            pickerKey={`bulk-start-${shift.id}`}
                            label="از"
                            timeValue={isAllDay ? "00:00" : shift.startTime}
                            disabled={isAllDay}
                            drawerTitle="انتخاب ساعت شروع"
                            onChange={(nextTime) =>
                              setBulkShifts((prev) =>
                                prev.map((item) =>
                                  item.id === shift.id
                                    ? { ...item, startTime: nextTime }
                                    : item
                                )
                              )
                            }
                          />

                          <ShiftTimeField
                            pickerKey={`bulk-end-${shift.id}`}
                            label="تا"
                            timeValue={isAllDay ? "23:59" : shift.endTime}
                            disabled={isAllDay}
                            drawerTitle="انتخاب ساعت پایان"
                            onChange={(nextTime) =>
                              setBulkShifts((prev) =>
                                prev.map((item) =>
                                  item.id === shift.id
                                    ? { ...item, endTime: nextTime }
                                    : item
                                )
                              )
                            }
                          />
                        </div>

                        {bulkShifts.length > 1 && (
                          <Button
                            type="button"
                            variant="ghost"
                            size="icon"
                            className="size-8 shrink-0 self-end text-destructive hover:bg-destructive/10 sm:self-auto"
                            onClick={() =>
                              setBulkShifts((prev) =>
                                prev.filter((item) => item.id !== shift.id)
                              )
                            }
                            title="حذف بازه"
                          >
                            <Trash2 className="size-4" />
                          </Button>
                        )}
                      </div>
                    )
                  })}
                </div>
              </Field>

              <div className="flex flex-col gap-3 border-t pt-4 sm:flex-row sm:items-center sm:justify-between">
                <label className="flex cursor-pointer items-center gap-2 text-xs text-muted-foreground">
                  <Checkbox
                    checked={bulkOverwrite}
                    onCheckedChange={(v) => setBulkOverwrite(Boolean(v))}
                  />
                  <span>
                    بازنویسی روزهایی که در این ماه قبلاً تنظیم یا استثنا ثبت کرده‌ام
                  </span>
                </label>

                <Button
                  type="button"
                  disabled={isSaving}
                  onClick={handleApplyBulkToMonth}
                >
                  <Save className="ml-2 size-4" />
                  {isSaving ? "در حال اعمال..." : "اعمال روی کل ماه"}
                </Button>
              </div>
            </FieldGroup>
          </CardContent>
        </Card>

        <div className="flex flex-wrap items-center gap-4 rounded-lg border bg-card px-4 py-3 text-xs sm:rounded-xl">
          <span className="font-semibold text-foreground">راهنمای تقویم:</span>
          <div className="flex items-center gap-1.5">
            <span className="font-bold text-destructive">{toFa(15)}</span>
            <span className="text-muted-foreground">
              تعطیل عمومی و رسمی (فقط عدد تاریخ قرمز)
            </span>
          </div>
          <div className="flex items-center gap-1.5">
            <span className="inline-block size-3.5 rounded-sm border border-destructive bg-destructive/20 sm:rounded" />
            <span className="text-muted-foreground">
              تعطیل خود کلینیک (کل باکس روز قرمز)
            </span>
          </div>
          <div className="flex items-center gap-1.5 text-muted-foreground">
            <Clock className="size-3.5 text-primary" />
            <span>
              برای استثنا یا تغییر ساعت هر روز، روی باکس همان روز کلیک کنید
            </span>
          </div>
        </div>

        {/* تقویم ماهانه با اسکرول عرضی ظریف در موبایل */}
        <Card className="overflow-hidden rounded-lg sm:rounded-xl">
          <CardContent className="p-2 sm:p-4">
            <div className="overflow-x-auto pb-2 [scrollbar-color:hsl(var(--border))_transparent] [scrollbar-width:thin] [&::-webkit-scrollbar]:h-1.5 [&::-webkit-scrollbar-thumb]:rounded-full [&::-webkit-scrollbar-thumb]:bg-border [&::-webkit-scrollbar-track]:bg-transparent">
              <div className="min-w-[680px] md:min-w-0">
                <div className="grid grid-cols-7 gap-1.5 pb-2 text-center sm:gap-2">
                  {JALALI_WEEKDAYS.map((weekday, idx) => (
                    <div
                      key={weekday}
                      className={cn(
                        "rounded-md bg-muted/70 py-2 text-xs font-bold sm:rounded-lg",
                        idx === 6 && "text-destructive"
                      )}
                    >
                      {weekday}
                    </div>
                  ))}
                </div>

                <div className="grid grid-cols-7 gap-1.5 sm:gap-2">
                  {monthGridCells.map((cell) => {
                    if (!cell.jd || !cell.dateIso || !cell.gDate) {
                      return (
                        <div
                          key={cell.key}
                          className="min-h-24 rounded-md border border-dashed bg-muted/20 sm:min-h-28 sm:rounded-xl"
                        />
                      )
                    }

                    const dayRecord = recordsByIso[cell.dateIso]
                    const officialList = holidaysByIso[cell.dateIso] || []
                    const isOfficialHoliday =
                      officialList.length > 0 || cell.weekdayIndex === 6
                    const isClinicHoliday = Boolean(dayRecord?.isClinicHoliday)
                    const shifts = !isClinicHoliday
                      ? dayRecord?.shifts || []
                      : []

                    const holidayTooltipText = [
                      ...officialList.map((h) => h.title),
                      cell.weekdayIndex === 6 ? "جمعه" : null,
                      isClinicHoliday
                        ? dayRecord?.holidayNote || "تعطیل کلینیک"
                        : null,
                    ]
                      .filter(Boolean)
                      .join(" • ")

                    return (
                      <button
                        key={cell.key}
                        type="button"
                        onClick={() => openDayModal(cell)}
                        className={cn(
                          "flex min-h-28 flex-col justify-between rounded-md border p-2 text-right transition-all sm:min-h-28 sm:rounded-xl sm:p-2.5",
                          isClinicHoliday
                            ? "border-destructive bg-destructive/15 text-destructive hover:bg-destructive/20"
                            : shifts.length > 0
                            ? "border-border/80 bg-card hover:border-primary hover:shadow-xs"
                            : "bg-card/70 hover:border-muted-foreground/40 hover:bg-muted/30"
                        )}
                      >
                        <div className="flex w-full items-start justify-between gap-1">
                          {holidayTooltipText ? (
                            <Tooltip>
                              <TooltipTrigger
                                render={
                                  <span
                                    className={cn(
                                      "text-sm font-extrabold sm:text-base",
                                      isClinicHoliday || isOfficialHoliday
                                        ? "text-destructive"
                                        : "text-foreground"
                                    )}
                                  >
                                    {toFa(cell.jd)}
                                  </span>
                                }
                              />
                              <TooltipContent
                                side="top"
                                className="max-w-56 text-xs"
                              >
                                {holidayTooltipText}
                              </TooltipContent>
                            </Tooltip>
                          ) : (
                            <span className="text-sm font-bold text-foreground sm:text-base">
                              {toFa(cell.jd)}
                            </span>
                          )}

                          {isClinicHoliday && (
                            <Badge
                              variant="destructive"
                              className="h-5 px-1.5 text-[10px]"
                            >
                              تعطیل مرکز
                            </Badge>
                          )}
                        </div>

                        <div className="mt-1.5 w-full space-y-1.5">
                          {isClinicHoliday ? (
                            <p className="line-clamp-2 text-[11px] font-medium text-destructive">
                              {dayRecord?.holidayNote || "تعطیل کلینیک"}
                            </p>
                          ) : shifts.length > 0 ? (
                            shifts.map((s) => {
                              const meta = getShiftModelMeta(s.category)
                              const ShiftIcon = meta.icon
                              const shiftTitle = getShiftModelLabel(
                                s.category,
                                s.label
                              )
                              const compactRange = formatCompactShiftRange(
                                s.category,
                                s.startTime,
                                s.endTime
                              )
                              return (
                                <div
                                  key={s.id}
                                  className={cn(
                                    "flex flex-col items-start gap-0.5 rounded border px-1.5 py-1 text-[10px] font-medium md:flex-row md:items-center md:justify-between md:gap-1 md:rounded-md md:py-0.5 md:text-[11px]",
                                    meta.pillClass
                                  )}
                                >
                                  <span className="inline-flex items-center gap-1 font-bold">
                                    <ShiftIcon className="size-3 shrink-0" />
                                    <span>{shiftTitle}:</span>
                                  </span>
                                  <span
                                    dir="ltr"
                                    className="font-mono text-[10px] font-semibold md:truncate md:text-[11px]"
                                  >
                                    {compactRange}
                                  </span>
                                </div>
                              )
                            })
                          ) : (
                            <span className="block text-[10px] text-muted-foreground">
                              بدون ساعت ثبت‌شده
                            </span>
                          )}
                        </div>
                      </button>
                    )
                  })}
                </div>
              </div>
            </div>
          </CardContent>
        </Card>

        <ResponsiveDialog
          open={Boolean(editingDay)}
          onOpenChange={(open) => !open && setEditingDay(null)}
          drawerPosition="bottom"
        >
          <ResponsiveDialogTrigger
            className="sr-only"
            tabIndex={-1}
            aria-hidden="true"
          />
          <ResponsiveDialogContent className="sm:max-w-lg">
            <ResponsiveDialogHeader>
              <ResponsiveDialogTitle>
                {editingDay
                  ? `تنظیم ساعات روز ${formatJalali(editingDay.gDate, {
                      weekday: true,
                    })}`
                  : "تنظیم روز"}
              </ResponsiveDialogTitle>
              <ResponsiveDialogDescription>
                {editingDay?.officialHolidays.length
                  ? `مناسبت رسمی: ${editingDay.officialHolidays
                      .map((h) => h.title)
                      .join("، ")}`
                  : "ساعات کاری این روز یا تعطیل بودن کلینیک را تعیین کنید."}
              </ResponsiveDialogDescription>
            </ResponsiveDialogHeader>

            <form onSubmit={handleSaveSingleDay} className="contents">
              <ResponsiveDialogPanel className="no-scrollbar max-h-[70vh] overflow-y-auto pt-2">
                <div className="px-5">
                  <FieldGroup>
                    <Field orientation="horizontal">
                      <Checkbox
                        id="dayIsClinicHoliday"
                        checked={dayIsClinicHoliday}
                        onCheckedChange={(v) =>
                          setDayIsClinicHoliday(Boolean(v))
                        }
                      />
                      <div className="space-y-0.5">
                        <FieldLabel
                          htmlFor="dayIsClinicHoliday"
                          className="cursor-pointer font-semibold text-destructive"
                        >
                          تعطیل بودن کلینیک در این روز (کل باکس قرمز می‌شود)
                        </FieldLabel>
                        <FieldDescription className="text-[11px]">
                          با فعال بودن این گزینه، کل روز برای کلینیک تعطیل در نظر گرفته می‌شود.
                        </FieldDescription>
                      </div>
                    </Field>

                    {dayIsClinicHoliday ? (
                      <Field>
                        <FieldLabel htmlFor="dayHolidayNote">
                          علت / عنوان تعطیلی کلینیک
                        </FieldLabel>
                        <Input
                          id="dayHolidayNote"
                          value={dayHolidayNote}
                          onChange={(e) => setDayHolidayNote(e.target.value)}
                          placeholder="مثلاً: ضدعفونی و تعمیرات، یا تعطیل اختصاصی"
                        />
                      </Field>
                    ) : (
                      <Field>
                        <div className="flex items-center justify-between gap-2">
                          <FieldLabel>بازه / شیفت‌های این روز</FieldLabel>
                          <Button
                            type="button"
                            variant="outline"
                            size="xs"
                            onClick={() =>
                              setDayShifts((prev) => [
                                ...prev,
                                createShiftByCategory(
                                  "CUSTOM",
                                  prev.length + 1
                                ),
                              ])
                            }
                          >
                            <Plus className="ml-1 size-3.5" />
                            افزودن بازه (+)
                          </Button>
                        </div>

                        <div className="mt-2 space-y-2.5">
                          {dayShifts.map((shift, idx) => {
                            const isAllDay = shift.category === "ALL_DAY"
                            const meta = getShiftModelMeta(shift.category)
                            const currentModelLabel = getShiftModelLabel(
                              shift.category,
                              shift.label
                            )
                            const ShiftIcon = meta.icon
                            return (
                              <div
                                key={shift.id}
                                className="space-y-2.5 rounded-lg border bg-muted/20 p-3 sm:rounded-xl"
                              >
                                <div className="flex items-center justify-between gap-2">
                                  <span className="text-xs font-semibold">
                                    بازه {toFa(idx + 1)}
                                  </span>
                                  {dayShifts.length > 1 && (
                                    <Button
                                      type="button"
                                      variant="ghost"
                                      size="icon-xs"
                                      className="text-destructive"
                                      onClick={() =>
                                        setDayShifts((prev) =>
                                          prev.filter((s) => s.id !== shift.id)
                                        )
                                      }
                                    >
                                      <Trash2 className="size-3.5" />
                                    </Button>
                                  )}
                                </div>

                                <Select
                                  value={shift.category}
                                  onValueChange={(val) =>
                                    handleDayShiftCategoryChange(
                                      shift.id,
                                      (val as ShiftCategory) || "CUSTOM"
                                    )
                                  }
                                >
                                  <SelectTrigger className="h-9 bg-background text-xs">
                                    <SelectValue>
                                      <span className="inline-flex items-center gap-1.5">
                                        <ShiftIcon
                                          className={cn(
                                            "size-3.5",
                                            meta.textClass
                                          )}
                                        />
                                        <span>{currentModelLabel}</span>
                                      </span>
                                    </SelectValue>
                                  </SelectTrigger>
                                  <SelectContent>
                                    {SHIFT_MODEL_OPTIONS.map((opt) => {
                                      const OptIcon = opt.icon
                                      return (
                                        <SelectItem
                                          key={opt.value}
                                          value={opt.value}
                                        >
                                          <span className="inline-flex items-center gap-1.5">
                                            <OptIcon
                                              className={cn(
                                                "size-3.5",
                                                opt.textClass
                                              )}
                                            />
                                            <span>{opt.label}</span>
                                          </span>
                                        </SelectItem>
                                      )
                                    })}
                                  </SelectContent>
                                </Select>

                                <div className="grid grid-cols-1 gap-2.5 sm:grid-cols-2">
                                  <ShiftTimeField
                                    pickerKey={`day-start-${shift.id}`}
                                    label="از"
                                    timeValue={
                                      isAllDay ? "00:00" : shift.startTime
                                    }
                                    disabled={isAllDay}
                                    drawerTitle="انتخاب ساعت شروع"
                                    onChange={(nextTime) =>
                                      setDayShifts((prev) =>
                                        prev.map((item) =>
                                          item.id === shift.id
                                            ? { ...item, startTime: nextTime }
                                            : item
                                        )
                                      )
                                    }
                                  />

                                  <ShiftTimeField
                                    pickerKey={`day-end-${shift.id}`}
                                    label="تا"
                                    timeValue={
                                      isAllDay ? "23:59" : shift.endTime
                                    }
                                    disabled={isAllDay}
                                    drawerTitle="انتخاب ساعت پایان"
                                    onChange={(nextTime) =>
                                      setDayShifts((prev) =>
                                        prev.map((item) =>
                                          item.id === shift.id
                                            ? { ...item, endTime: nextTime }
                                            : item
                                        )
                                      )
                                    }
                                  />
                                </div>
                              </div>
                            )
                          })}
                        </div>
                      </Field>
                    )}

                    {editingDay && recordsByIso[editingDay.dateIso] && (
                      <Field>
                        <Button
                          type="button"
                          variant="ghost"
                          size="sm"
                          disabled={isSaving}
                          onClick={handleResetSingleDay}
                          className="w-full justify-center text-xs text-muted-foreground hover:text-foreground"
                        >
                          <RotateCcw className="ml-1.5 size-3.5" />
                          حذف تنظیم/استثنای ثبت‌شده برای این روز
                        </Button>
                      </Field>
                    )}
                  </FieldGroup>
                </div>
              </ResponsiveDialogPanel>

              <ResponsiveDialogFooter className="gap-2">
                <ResponsiveDialogDesktopOnly>
                  <Button
                    type="button"
                    variant="outline"
                    onClick={() => setEditingDay(null)}
                    disabled={isSaving}
                  >
                    <Ban className="ml-1.5 size-4" />
                    انصراف
                  </Button>
                </ResponsiveDialogDesktopOnly>

                <Button
                  type="submit"
                  disabled={isSaving}
                  className="w-full sm:w-auto"
                >
                  <Save className="ml-1.5 size-4" />
                  {isSaving ? "در حال ذخیره..." : "ذخیره این روز"}
                </Button>
              </ResponsiveDialogFooter>
            </form>
          </ResponsiveDialogContent>
        </ResponsiveDialog>
      </div>
    </TooltipProvider>
  )
}
